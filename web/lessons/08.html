  • 字符串方法不改变原值：s.upper() 返回一个新串，s 本身没变。

Python# bmi.py —— 计算 BMI

weight = 70    # 体重，单位千克（整数）
height = 1.75  # 身高，单位米（小数）

# BMI = 体重 / 身高平方，** 是幂运算符
bmi = weight / height ** 2

# 用 f-string 输出，:.2f 表示保留两位小数
print(f"你的身高是 {height} 米，体重是 {weight} 千克，BMI 是 {bmi:.2f}。")

# 顺带解释一下数值的含义
if bmi < 18.5:
    print("偏瘦")
elif bmi < 24:
    print("正常范围")
elif bmi < 28:
    print("偏重")
else:
    print("肥胖")

Python# text_pipeline.py —— 字符串方法接力，全过程不修改原变量

text = "   python is FUN   "

# ① 原样输出：普通 print 看不出首尾空格，用 repr() 可以看清
print(repr(text))

# ② 只去掉两端空白
print(repr(text.strip()))

# ③ 再转小写
print(text.strip().lower())

# ④ 再每个单词首字母大写
print(text.strip().title())

# ⑤ 处理完之后的长度
print(len(text.strip()))

# 验证：原字符串自始至终没有被改动
print("原变量仍然是：", repr(text))

# 如果想让 text 本身保存处理结果，必须显式赋值
# text = text.strip().title()
# print(text)

Python# report.py —— 列对齐报表

products = [
    ("机械键盘", 499.5, 12),
    ("人体工学鼠标", 129.0, 156),
    ("4K 显示器", 2399.0, 7),
]

# 表头：名称左对齐 12，单价右对齐 10，库存居中 6
print(f"{'商品名称':<10}|{'':>11}|{'库存':^6}|")
print("-" * 32)

# 遍历列表，按格式输出每一行
for name, price, stock in products:
    print(f"{name:<12}|{price:>10.2f}|{stock:^6}|")

# 计算库存总件数
total_stock = 0
for _, _, stock in products:
    total_stock += stock
print("-" * 32)
print(f"库存合计：{total_stock} 件")

提示：for 循环和元组解包属于单元 3、单元 7 的正式内容。如果还没学到，可以只写打印部分，或先用 5 行重复代码代替循环——这在练习中是完全可以接受的。

参考答案 1-8

Python# truthy.py —— bool 转换的真相表

values = [0, 1, 0.0, -1, "", "0", "False", [], [0], None]

# Python 判断"真/假"的唯一规则：空或零为假，其余为真
for value in values:
    print(f"bool({value!r:>10}) = {bool(value)}")

print()
print("=== 逐条解释 ===")
print("bool(0)      = False —— 整数零被视为假")
print("bool(1)      = True  —— 非零整数为真")
print("bool(0.0)    = False —— 浮点零同样为假")
print("bool(-1)     = True  —— 负数也是非零，为真")
print('bool("")     = False —— 空字符串为假')
print('bool("0")   = True  —— 字符串 \"0\" 非空，内容不是 \"\" 所以为真')
print('bool("False") = True —— 同上，非空字符串一律为真')
print("bool([])     = False —— 空列表为假")
print("bool([0])    = True  —— 列表里有一个元素，不为空，为真")
print("bool(None)   = False —— None 表示\"什么都没有\"，为假")

{value!r:>10} 中的 !r 表示用 repr() 形式显示（带引号，能看清空字符串），:>10 表示右对齐占 10 格。

参考答案 1-9

Python# clean_names.py —— 去前缀 / 去后缀 / 按扩展名筛选
files = ["report_2026_q1.csv", "report_2026_q2.csv", "notes.txt", "report_2026_q3.csv"]

noprefix = [f.removeprefix("report_") for f in files]
print("去前缀：" + " | ".join(noprefix))          # notes.txt 原样保留

nosuffix = [f.removesuffix(".csv").removesuffix(".txt") for f in files]
print("去后缀：" + " | ".join(nosuffix))

csv_only = [f for f in files if f.endswith(".csv")]
print("CSV  ：" + " | ".join(csv_only))

输出：

示例去前缀：2026_q1.csv | 2026_q2.csv | notes.txt | 2026_q3.csv
去后缀：report_2026_q1 | report_2026_q2 | notes | report_2026_q3
CSV  ：report_2026_q1.csv | report_2026_q2.csv | report_2026_q3.csv

两种写法都能去掉后缀，各有适用场景：removesuffix(".csv").removesuffix(".txt") 明确知道有哪几种后缀；f.rsplit(".", 1)[0] 则通吃任意扩展名，但会把 archive.tar.gz 变成 archive.tar。真实项目里更稳的是 os.path.splitext(f)[0]（单元 9 会讲 os）。

参考答案 1-10

Python# kv.py —— 一行配置拆成字典，再按 key 排序输出
raw = "name=张三;age=18;city=北京"

d = {}
for pair in raw.split(";"):
    k, v = pair.split("=", 1)          # 只拆第一个 =，值里再有 = 也不会被切碎
    d[k] = v

print(d)
print(" | ".join(f"{k}={v}" for k, v in sorted(d.items())))
print(f"{d['name']} 今年 {int(d['age'])} 岁")

输出：

示例{'name': '张三', 'age': '18', 'city': '北京'}
age=18 | city=北京 | name=张三
张三 今年 18 岁

注意 d["age"] 是字符串 "18"，f"{d['age']} 岁" 不会做加法，所以"18 岁"这类输出不需要转类型；一旦要参与计算（d["age"] + 1）就必须先 int()。
