Python# odd_even.py —— 判断奇偶，只用一个 if

number = 7

# 一个 if + else 就够了，不需要两个独立的 if
if number % 2 == 0:
    print(f"{number} 是偶数")
else:
    print(f"{number} 是奇数")

# 注意：% 是取余运算符，任意数都能用它做奇偶判断
#   正奇数 % 2 == 1，正偶数 % 2 == 0
#   负数也能正确处理：-7 % 2 == 1  （Python 的 % 总是非负）
print("-7 是奇数吗：", -7 % 2 == 1)      # True

# 反过来说：不要写if number % 2: 这种写法，
# 虽然结果对（0 是假、1 是真），但可读性差，别人看不出你的意图。

    max_value = b

    max_value = c

Python# leap_year.py —— 闰年判断

# 规则：能被 400 整除，或者能被 4 整除但不能被 100 整除
def is_leap(year):
    if year % 400 == 0:
        return True
    elif year % 4 == 0 and year % 100 != 0:
        return True
    else:
        return False


for year in [2000, 1900, 2024, 2023, 2400]:
    if is_leap(year):
        print(f"{year} 是闰年")
    else:
        print(f"{year} 不是闰年")

# 输出：
# 2000 是闰年
# 1900 不是闰年
# 2024 是闰年
# 2023 不是闰年
# 2400 是闰年

# 纯 if 版本（不用函数，直接循环判断）
print("--- 不用函数的版本 ---")
for year in [2000, 1900, 2024, 2023]:
    if year % 4 == 0 and year % 100 != 0:
        print(f"{year} 是闰年")
    elif year % 400 == 0:
        print(f"{year} 是闰年")
    else:
        print(f"{year} 不是闰年")

# 注意：上面的顺序也正确，因为能被 400 整除的数也一定被 4 整除，

Python# short_circuit.py —— 短路求值的安全价值

data = [10, 20, 30]

# ===== 安全版：先问长度，再问内容 =====
if len(data) > 0 and data[0] == 10:
    print("安全版：列表非空，且第一个数是 10")
else:
    print("安全版：条件不满足")

# 换成空列表再试一次
empty = []

# ===== 安全版在空列表上依然安全 =====
if len(empty) > 0 and empty[0] == 10:
    print("空列表：不应该走到这里")
else:
    print("安全版处理空列表：条件不满足，程序继续正常运行")

# ===== 危险版：先访问元素 =====
# 下面这行如果真的运行，会崩溃（IndexError: list index out of range）
# if empty[0] == 10 and len(empty) > 0:
#     print("危险版：条件满足")

# 解释为什么崩溃：
# Python 执行 and 时是从左往右的。它必须先算出左边 empty[0] == 10 的结果，
# 但 empty 是空列表，取第0 个元素直接 IndexError，
# 右边的 len(empty) > 0 根本没机会被计算。

# ===== 更简洁的安全写法：利用空列表本身为假 =====
# Python 认为空列表、0、空字符串、None 都是"假"，所以可以直接判断
if empty and empty[0] == 10:
    print("第一个数是 10")
else:
    print("列表为空，无需检查（短路了，没访问 empty[0]）")

Python# triangle.py —— 三角形判定

def classify_triangle(a, b, c):
    """根据三条边判断三角形类型。"""
    # 第一道关：是不是三角形。必须放在最前面，
    # 否则 1, 2, 100 这种数据会被误判成"普通三角形"
    if a + b <= c or a + c <= b or b + c <= a:
        return "不是三角形"
    elif a == b == c:
        # 三个都相等才是等边。等腰要放在等边后面判断，
        # 否则等边三角形也会被"任意两边相等"命中
        return "等边三角形"
    elif a == b or b == c or a == c:
        return "等腰三角形"
    else:
        return "普通三角形"


# 测试
test_cases = [
    (3, 3, 3),
    (5, 5, 8),
    (7, 10, 5),
    (1, 2, 3),
    (1, 2, 10),
]

for a, b, c in test_cases:
    print(f"边长 {a}, {b}, {c} -> {classify_triangle(a, b, c)}")

# 输出：
# 边长 3, 3, 3 -> 等边三角形
# 边长 5, 5, 8 -> 等腰三角形
# 边长 7, 10, 5 -> 普通三角形
# 边长 1, 2, 3 -> 不是三角形
# 边长 1, 2, 10 -> 不是三角形

Python# door.py —— 忽略大小写与空格的门禁判断
members = ["alice", "bob", "carol"]
visitors = ["Alice", "dave", "BOB", "", "carol"]

for raw in visitors:
    name = raw.strip()                  # 先去首尾空格
    if not name:                        # 空串 → 无效，提前结束本轮
        print("无效输入")
        continue
    if name.lower() in members:         # 统一小写再比
        print(f"准入: {name}")
    else:
        print(f"拒绝: {name}")

输出：

示例准入: Alice
拒绝: dave
准入: BOB
无效输入
准入: carol

注意 continue 的用法：空名字不进入后面的判断，避免 "" in members 这种「逻辑上对、语义上没意义」的比较。先排除无效输入，再处理正常情况，这个顺序能让后面的代码少一层缩进。

六、本单元验收自测

1. 简答题：下面代码输出什么？

Pythonage = 20
if age >= 18:
    print("成年")
else:
    print("未成年")
print("结束")

要点：输出两行——成年 和 结束。if 为真时只执行 if 块内的代码，else 块被完全跳过；print("结束") 缩进在外层，两个分支都会走到之后执行它。

2. 判断题：if 5 &gt; 3 &amp;&amp; 2 &gt; 4: print("A") else: print("B") 这段写法（用 &amp;&amp;）在 Python 里能正常运行。

要点：错误。Python 用 and 不用 &&。正确写法：if 5 > 3 and 2 > 4:。而 2 > 4 为假，所以实际输出 B。

3. 简答题：为什么下面这段代码在 score = 85 时输出"D"，而作者期望的是"B"？

Pythonscore = 85
if score >= 60:
    level = "D"
elif score >= 80:
    level = "B"
else:
    level = "F"

要点：因为 if 链是"从上往下、命中即停"。score >= 60 为真（85 ≥ 60），立刻执行 level = "D"，后面的 elif 和 else 全部跳过。修复方法：把条件最严格的放最前面——if score >= 80: B / elif score >= 60: D / else: F。

4. 简答题：为什么下面这段代码在 items 为空列表时会崩溃？写出两种修复方式。

Pythonif items[0] == "apple" and len(items) > 0:
    print("第一个是苹果")

要点：and 从左往右求值，Python 必须先算出 items[0] == "apple"，而空列表取第 0 个元素立刻抛 IndexError，右边的长度检查根本没机会执行。修复一：交换顺序写成 if len(items) > 0 and items[0] == "apple"，让便宜的安全检查在前。修复二：利用空列表为假，直接写 if items and items[0] == "apple"。

5. 判断题：if value == None: 和 if value is None: 两种写法应该优先用哪个？为什么？

要点：优先用 is None。None 是表示"没有值"的唯一对象，用 is 表达的是"身份/状态"判断，语义准确且速度快。如果某个自定义类重写了 __eq__ 允许自己等于 None，== None 就会意外成立；而 is None 永远只在真的是 None 时成立。

七、常见错误清单（务必逐条自查）

错误 1：IndentationError: expected an indented block

症状：if 后面那行没缩进。 原因：Python 用缩进表示代码块，if/for/def 后面必须缩进一层。 改法：统一 4 个空格（不要混 Tab）；VS Code 里开 Render Whitespace 能看见空格与 Tab 的区别。

错误 2：if x = 1: 或 elif 写成 else if

症状：SyntaxError。 原因：条件用 ==；Python 没有 else if，是 elif。 改法：if x == 1:；多分支写 if / elif / else。

错误 3：if answer == "Yes" 对 "yes" 不成立

症状：用户输入 yes 却走了 else。 原因：字符串比较区分大小写。 改法：比较前统一 answer.strip().lower()；需要更宽松就用 answer.lower() in ("y", "yes")。

错误 4：elif 顺序写反，永远进第一个分支

症状：分数 95 只拿到「及格」。 原因：if score > 60: ... elif score > 90: ...——90 分也满足 > 60，第一个分支就截住了。 改法：从窄到宽排：先 >= 90，再 >= 60，最后 else。

错误 5：if items: 判空却把 0 当成空

症状：数量是 0 时走了「没有数据」分支，但 0 是合法值。 原因：0、""、[]、None 在条件里都为假。 改法：要区分「没有」和「是 0」就显式判：if items is None:、if len(items) == 0:。
