2.5 关卡 2 的五条硬指标

先讲「为什么是这五条」，第八节再展开验收细则。五条分别对应五种能力：

# 指标 考的是
1 三个项目全部跑通、推到公开仓库，各有 README + requirements.txt 可交付：别人拿到能跑起来
2 「学习笔记」已部署上线，公网可访问，错误页已定制 能上线：处理过真实环境的坑
3 能用自己的函数写出 pytest 测试并跑过（不是抄教材的） 可验证：不靠肉眼保证质量
4 接手一个没写过的 500 行 Python 项目，能读懂结构并定位一个 bug 可阅读：能维护别人的代码
5 能说清每个项目「解决了什么问题、还能怎么扩展」 有判断力：知道东西值不值、往哪走

★ 为什么第 5 条也算硬指标：前四条「做过」就能证明，只有第 5 条考「想清楚」。有人照抄教程做出三个项目然后寸步难行，也有人做出三个项目且清楚下一步加什么。区别不在做过多少，在有没有判断。 面试和实际工作里后者才真正值钱。

常见状态与补法：项目能跑但没推仓库 → 练习 21-1；推了但没 README → 必含「是什么 / 怎么装 / 怎么跑」三段；「学习笔记」只在本地跑 → 回单元 20；测试全是抄教材的 → 练习 21-2；看别人的代码就慌 → 按 2.4 五步 + 示例 5 练；说不出能怎么扩展 → 每个项目写三句「未来想法」。

「还差哪一条」这种判断最容易自我欺骗。写成一张布尔表，缺口就没法赖掉：

Python# 五条硬指标自评：把「感觉差不多了」变成一张缺口清单
INDICATORS = {
    1: "三个项目跑通 + 公开仓库 + README + requirements.txt",
    2: "学习笔记已部署，公网可访问，404/500 已定制",
    3: "自己的函数写出 pytest 测试并跑绿（≥3 条）",
    4: "读懂没写过的 500 行项目并定位一个 bug",
    5: "能说清每个项目解决什么问题、还能怎么扩展",
}

# 自评：True = 已达标；一条一条对，别客气
MY_STATUS = {1: True, 2: False, 3: True, 4: False, 5: True}


def report(status, indicators=INDICATORS):
    done = [n for n in indicators if status.get(n)]
    gaps = [n for n in indicators if not status.get(n)]
    return done, gaps


done, gaps = report(MY_STATUS)
print(f"已达标 {len(done)}/5：{done}")
print(f"还差 {len(gaps)} 条：")
for number in gaps:
    print(f"  [{number}] {INDICATORS[number]}")

print("\n补法（对应练习）：")
FIX = {1: "练习 21-1（补 README / requirements / 推仓库）",
       2: "回单元 20，按 2.7 三大坑排查后重新 push",
       3: "练习 21-2（挑 3 个纯逻辑函数，各 2 条断言）",
       4: "练习 21-4（2.4 五步 + 示例 5 的 explorer.py）",
       5: "每个项目 README 加一节「未来想法」，写清要动哪几个文件"}
for number in gaps:
    print(f"  [{number}] {FIX[number]}")

print("\n★ 只有第 5 条考「想清楚」，也只有它没法靠照抄过关。")

这份自评里 3/5 达标、缺第 2、4 条——缺口清单比「差不多了」有用得多，因为它直接对应到两个练习上。建议每完成一个练习就改一次 MY_STATUS，重跑一遍，看着数字往上走。

指标 3 最冤的一种翻车方式是命名写错：pytest 一条都不报错，只显示 collected 0 items，你以为「测试通过了」。提前查一遍命名，比事后猜省事：

Python# pytest 收不到测试？先查命名
from pathlib import Path

DEMO = {
    "test_game_functions.py": "def test_update_score():\n    assert True\n\n"
                              "def check_collision():\n    assert True\n",
    "game_functions.py": "def update_score(score, hits):\n    return score + hits\n",
    "tests/test_models.py": "class TestTopic:\n    def test_str(self):\n        assert True\n",
}


def audit_tests(root):
    """返回 (测试文件数, 会被收走的函数, 收不走的函数, 收不走的函数名)。"""
    files = collected = ignored = 0
    bad_names = []
    for path in sorted(Path(root).rglob("*.py")):
        if not (path.name.startswith("test_") or "tests" in path.parts):
            continue
        files += 1
        for line in path.read_text(encoding="utf-8").splitlines():
            stripped = line.strip()
            if not stripped.startswith("def "):
                continue
            name = stripped[4:].split("(")[0]
            if name.startswith("test_"):
                collected += 1
            else:
                ignored += 1                       # 在测试文件里，但 pytest 不认
                bad_names.append(f"{path.name}: {name}")
    return files, collected, ignored, bad_names


root = Path("data/tests_demo")
for name, text in DEMO.items():
    path = root / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding="utf-8")

files, collected, ignored, bad = audit_tests(root)
print(f"测试文件 {files} 个，pytest 会收走 {collected} 个 test_ 函数，"
      f"另有 {ignored} 个在测试文件里但名字不以 test_ 开头：")
for name in bad:
    print("  ", name)
print("\n★ pytest 的规则：文件名 test_*.py 或 *_test.py，函数名 test_* ，类名 Test*")
print("   名字写错不报错，只显示 collected 0 items —— 最容易白忙半天。")

import shutil
shutil.rmtree("data", ignore_errors=True)

输出把一个「看起来写了测试」的文件戳穿了：test_game_functions.py 里 def check_collision(): 名字里虽有 test 文件、却没有 test_ 前缀，pytest 根本不收。这个函数里的断言永远不会执行，也永远不会失败——比报错更危险。

示例 1：项目目录体检脚本（project_doctor.py）

对应 2.4 的「工程件齐全」检查。放在项目根目录跑一遍，就知道缺什么。

Python# project_doctor.py —— 检查一个项目该有的工程件是否齐全
# 用法：python project_doctor.py [项目根目录，默认为当前目录]
import sys
from pathlib import Path

REQUIRED = ["README.md", "requirements.txt", ".gitignore"]
IGNORE_KEYWORDS = ["__pycache__", "venv", "db.sqlite3"]
SECRET_NAMES = ["SECRET_KEY", "PASSWORD", "API_KEY", "TOKEN"]
SKIP_DIRS = {".git", "__pycache__", "venv", ".venv", "env", "node_modules"}
FILE_LIMIT = 300


def read_text(path):
    """读文件；不存在或编码不对就返回空字符串。"""
    try:
        return path.read_text(encoding="utf-8")
    except (UnicodeDecodeError, OSError):
        return ""


def check_requirements(root):
    """依赖清单有没有锁版本。返回 None 表示通过。"""
    path = root / "requirements.txt"
    if not path.exists():
        return "requirements.txt 不存在"
    unpinned = [ln.strip() for ln in read_text(path).splitlines()
                if ln.strip() and not ln.strip().startswith("#")
                and "==" not in ln]
    return f"有 {len(unpinned)} 个依赖没锁版本，例如：{unpinned[:3]}" if unpinned else None


def check_gitignore(root):
    """.gitignore 有没有覆盖关键条目。"""
    path = root / ".gitignore"
    if not path.exists():
        return ".gitignore 不存在"
    missing = [k for k in IGNORE_KEYWORDS if k not in read_text(path)]
    return f".gitignore 缺少：{missing}" if missing else None


def report(root):
    """跑完全部检查并打印报告，返回问题总数。"""
    files = [p for p in root.rglob("*.py")
             if not any(part in SKIP_DIRS for part in p.parts)]
    problems = 0
    print(f"体检对象：{root.resolve()}\n【必备工程件】")
    for name in REQUIRED:
        ok = (root / name).exists()
        print(f"  [{'OK' if ok else '缺'}] {name}")
        problems += 0 if ok else 1

    print("\n【内容检查】")
    for label, result in [("依赖清单", check_requirements(root)),
                          ("忽略清单", check_gitignore(root))]:
        print(f"  [{'OK' if result is None else '问题'}] {label}"
              + ("检查通过" if result is None else f"：{result}"))
        problems += 0 if result is None else 1

    print(f"\n【代码扫描】（共 {len(files)} 个 .py 文件）")
    # 扫硬编码密钥：排除从环境变量读的、名单字面量、没赋值的行
    secrets = []
    for path in files:
        for lineno, line in enumerate(read_text(path).splitlines(), 1):
            if not any(k in line for k in SECRET_NAMES):
                continue
            if "os.environ" in line or "getenv" in line:
                continue
            if "=" not in line or "[" in line or "{" in line:
                continue
            if len(line.split("=", 1)[1].strip()) > 8:
                secrets.append(f"{path}:{lineno}")
    long_files = [f"{p}（{len(read_text(p).splitlines())} 行）" for p in files
                  if len(read_text(p).splitlines()) > FILE_LIMIT]
    for label, found in [("硬编码密钥检查", secrets),
                         (f"{FILE_LIMIT} 行检查", long_files)]:
        print(f"  [{'OK' if not found else '问题'}] {label}"
              + ("" if not found else f"：{found}"))
        problems += len(found)

    print()
    print(f"共发现 {problems} 个必须修的问题。" if problems
          else "体检通过：工程件齐全，可以推仓库 / 部署了。")
    return problems


if __name__ == "__main__":
    report(Path(sys.argv[1]) if len(sys.argv) > 1 else Path("."))

示例 2：requirements.txt 自动生成（make_requirements.py）

思路：扫项目里所有 import 语句，把「Python 自带的」剔掉，剩下的就是第三方依赖。

Python# make_requirements.py —— 扫描 import 语句，列出项目用到的第三方包
# 用法：python make_requirements.py [项目根目录]
# 局限：它只猜「需要哪些包」，版本号要用 pip freeze 补上
import ast
import sys
import sysconfig
from pathlib import Path

# import 名和 PyPI 包名不一致的映射（最常踩的几个）
ALIASES = {"PIL": "Pillow", "cv2": "opencv-python",
           "sklearn": "scikit-learn", "bs4": "beautifulsoup4",
           "yaml": "PyYAML", "django_bootstrap5": "django-bootstrap5"}
INDIRECT = {"setuptools", "pip", "wheel"}
SKIP_DIRS = {".git", "__pycache__", "venv", ".venv", "env",
             "node_modules", "data", "tests"}


def known_names():
    """标准库 + 内置模块 + 已知别名，都是"不用写进 requirements"的。"""
    stdlib = Path(sysconfig.get_paths()["stdlib"])
    names = {p.stem for p in stdlib.glob("*.py")} | set(sys.builtin_module_names)
    names |= {p.name for p in stdlib.glob("*")
              if p.is_dir() and (p / "__init__.py").exists()}
    return names | set(ALIASES) | INDIRECT


def collect_imports(root, known):
    """遍历所有 .py 文件，返回 {顶层模块名: [出现位置]}（已滤掉标准库和本地模块）。"""
    found = {}
    for path in sorted(root.rglob("*.py")):
        if any(part in SKIP_DIRS for part in path.parts):
            continue
        try:
            tree = ast.parse(path.read_text(encoding="utf-8"))
        except (UnicodeDecodeError, SyntaxError):
            continue
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                names = [x.name.split(".")[0] for x in node.names]
            elif isinstance(node, ast.ImportFrom):
                # level > 0 是相对导入，属项目内部模块，不算依赖
                names = ([node.module.split(".")[0]]
                         if node.level == 0 and node.module else [])
            else:
                names = []
            for name in names:
                found.setdefault(name, []).append(str(path))
    return {n: p for n, p in found.items() if n not in known
            and not (root / f"{n}.py").exists()
            and not (root / n / "__init__.py").exists()}


def main():
    root = Path(sys.argv[1]) if len(sys.argv) > 1 else Path(".")
    imports = collect_imports(root, known_names())
    print(f"在 {root.resolve()} 中发现 {len(imports)} 个第三方依赖：")
    for name in sorted(imports):
        print(f"  {ALIASES.get(name, name):<22} ← {', '.join(imports[name][:2])}")
    print("\n下一步：在虚拟环境里执行 pip freeze，核对版本号后写成")
    print("        包名==版本号  的形式写进 requirements.txt。")


if __name__ == "__main__":
    main()

示例 3：代码规模与圈复杂度提示（code_metrics.py）

对应 2.3 节的「单文件不超 300 行」。用 ast 静态统计，不导入你的项目，所以不会有副作用。

Python# code_metrics.py —— 统计文件行数、函数长度和圈复杂度近似值
# 用法：python code_metrics.py [项目根目录]
# 只用标准库，不会导入你的项目代码
import ast
import sys
from pathlib import Path

SKIP_DIRS = {".git", "__pycache__", "venv", ".venv", "env", "node_modules", "data"}
FILE_LIMIT, FUNC_LIMIT, COMPLEXITY_LIMIT = 300, 50, 8
BRANCH_NODES = (ast.If, ast.IfExp, ast.While, ast.For, ast.AsyncFor,
                ast.ExceptHandler, ast.With, ast.AsyncWith, ast.Assert,
                ast.comprehension)


def count_decisions(node):
    """数一遍「会改变控制流」的节点数量，作为圈复杂度的近似。"""
    total = 0
    for child in ast.walk(node):
        if isinstance(child, BRANCH_NODES):
            total += 1
        elif isinstance(child, ast.BoolOp):
            total += max(0, len(child.values) - 1)   # a and b：每多一项 +1
        elif isinstance(child, ast.Match):
            total += len(child.cases)
    return total


def analyze_source(source, filename):
    """解析一个文件，返回 {文件, 总行数, 函数列表}（按复杂度降序）。"""
    result = {"文件": filename, "总行数": len(source.splitlines()), "函数": []}
    try:
        tree = ast.parse(source)
    except SyntaxError as error:
        result["错误"] = str(error)
        return result
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            start = node.lineno
            length = getattr(node, "end_lineno", start) - start + 1
            result["函数"].append({
                "名字": node.name, "行号": start, "长度": length,
                "复杂度": count_decisions(node) + 1})   # 圈复杂度 = 决策点 + 1
    result["函数"].sort(key=lambda f: f["复杂度"], reverse=True)
    return result


def report(root):
    """扫描并打印报告，返回值得重构的处数。"""
    reports = []
    for path in sorted(root.rglob("*.py")):
        if any(part in SKIP_DIRS for part in path.parts):
            continue
        try:
            reports.append(analyze_source(path.read_text(encoding="utf-8"),
                                          str(path)))
        except UnicodeDecodeError:
            continue
    if not reports:
        print(f"{root} 下没找到 .py 文件")
        return 0
    print(f"共 {len(reports)} 个文件、{sum(r['总行数'] for r in reports)} 行、"
          f"{sum(len(r['函数']) for r in reports)} 个函数\n")
    problems = 0
    print(f"【超过 {FILE_LIMIT} 行的文件（建议拆分）】")
    big = [r for r in reports if r["总行数"] > FILE_LIMIT]
    for r in sorted(big, key=lambda x: x["总行数"], reverse=True):
        print(f"  {r['文件']}：{r['总行数']} 行")
    if not big:
        print("  无")
    problems += len(big)

    print("\n【复杂度最高的 5 个函数】")
    all_funcs = [(r["文件"], f) for r in reports for f in r["函数"]]
    for filename, func in sorted(all_funcs, key=lambda x: x[1]["复杂度"],
                                 reverse=True)[:5]:
        flag = (" ← 超标" if func["复杂度"] > COMPLEXITY_LIMIT else "") \
               + (" ← 太长" if func["长度"] > FUNC_LIMIT else "")
        print(f"  复杂度{func['复杂度']:>3}  {func['长度']:>4} 行  "
              f"{func['名字']}()  ({filename}:{func['行号']}){flag}")
        if flag:
            problems += 1
    print(f"\n共 {problems} 处值得重构。先挑「复杂度最高」的那个函数下手。"
          if problems else "\n全部在阈值以内，规模控制得不错。")
    return problems


if __name__ == "__main__":
    report(Path(sys.argv[1]) if len(sys.argv) > 1 else Path("."))

示例 4：统一日志配置模板（logging_setup.py）

三个项目都缺一个东西：出问题时能回头查。print 只适合调试，正式项目要写日志。这个模板可以直接抄进任何项目。

Python# logging_setup.py —— 项目通用的日志配置模板
# 用法：from logging_setup import setup_logging
import logging
import logging.handlers
from pathlib import Path

LOG_FORMAT = "%(asctime)s | %(levelname)-7s | %(name)-12s | %(message)s"
DATE_FORMAT = "%Y-%m-%d %H:%M:%S"


def setup_logging(name="my_project", level="INFO", log_file="data/app.log"):
    """配置并返回一个 logger。

    装两个 handler：① RotatingFileHandler 写文件，单文件超 1MB 就轮换、最多留 3 份，
    防止日志把磁盘写满；② StreamHandler 让终端也能看到，方便开发。
    """
    logger = logging.getLogger(name)
    logger.handlers.clear()      # 重复调用时先清空，否则日志会打印好几遍
    logger.setLevel(getattr(logging, level.upper(), logging.INFO))
    logger.propagate = False     # 不冒泡到 root，避免重复输出
    formatter = logging.Formatter(LOG_FORMAT, datefmt=DATE_FORMAT)

    path = Path(log_file)
    path.parent.mkdir(parents=True, exist_ok=True)
    for handler in (logging.handlers.RotatingFileHandler(
                        path, maxBytes=1_000_000, backupCount=3,
                        encoding="utf-8"),
                    logging.StreamHandler()):
        handler.setFormatter(formatter)
        logger.addHandler(handler)
    return logger


def main():
    """最小可跑演示：建 logger → 记三条不同级别的日志。"""
    logger = setup_logging(name="demo", log_file="data/app.log")
    logger.info("程序启动")
    logger.warning("这是一条警告：磁盘占用偏高")
    logger.error("这是一条错误：连不上数据库")
    logger.info("程序正常结束，日志已写入 data/app.log")
    # 跑完可以删掉 data/ 目录


if __name__ == "__main__":
    main()

示例 5：陌生项目结构速读（explorer.py）

把 2.4 节的第 2–4 步做成一个脚本：标出每个文件的角色、读依赖清单、给出建议阅读顺序。

Python# explorer.py —— 30 秒速读一个陌生项目
# 用法：python explorer.py [项目根目录]
import sys
from pathlib import Path

SKIP_DIRS = {".git", "__pycache__", "venv", ".venv", "env", "node_modules",
             ".idea", ".vscode", "static", "migrations"}
TEXT_SUFFIXES = {".py", ".txt", ".md", ".cfg", ".toml", ".yaml", ".yml",
                 ".json", ".ini", ".env"}
ENTRY_MARKERS = ('if __name__ == "__main__"', "def main(")
READ_ORDER = ["项目说明", "依赖清单", "★ 程序入口", "★ Django 入口",
              "配置文件", "业务逻辑", "测试"]


def read_text(path):
    """读文件；编码不对就返回空字符串。"""
    try:
        return path.read_text(encoding="utf-8")
    except (UnicodeDecodeError, OSError):
        return ""


def guess_role(path, source):
    """根据文件名和内容猜这个文件的角色。"""
    name = path.name
    if name == "manage.py":
        return "★ Django 入口"
    if name in ("settings.py", "config.py", ".env"):
        return "配置文件"
    if name == "requirements.txt":
        return "依赖清单"
    if name.startswith("test_") or "tests" in path.parts:
        return "测试"
    if name.lower().startswith("readme"):
        return "项目说明"
    if any(m in source for m in ENTRY_MARKERS):
        return "★ 程序入口"
    return "业务逻辑" if "def " in source or "class " in source else "数据/其他"


def scan_files(root):
    """★ 第 2–4 步：所有源码文件按角色归类，并打印依赖清单。"""
    print("\n【文件角色】")
    roles = {}
    for path in sorted(root.rglob("*")):
        if not path.is_file() or path.suffix not in TEXT_SUFFIXES:
            continue
        if any(part in SKIP_DIRS for part in path.parts):
            continue
        rel = str(path.relative_to(root)).replace("\\", "/")
        if rel == "requirements.txt":
            continue                      # 依赖清单单独打印
        roles.setdefault(guess_role(path, read_text(path)), []).append(rel)
    for role in sorted(roles):
        print(f"\n{role}：")
        for name in roles[role][:8]:
            print(f"    {name}")

    print("\n【依赖清单】")
    lines = [ln.strip() for ln in read_text(root / "requirements.txt").splitlines()]
    print("\n".join(f"  {ln}" for ln in lines if ln and not ln.startswith("#"))
          or "  没有 requirements.txt —— 这本身就是一个工程问题")
    return roles


def main():
    root = Path(sys.argv[1]) if len(sys.argv) > 1 else Path("data/unknown_project")
    if not root.exists():
        print(f"目录不存在：{root}（可以先随便传一个已有目录试试）")
        return
    print(f"速读目标：{root.resolve()}\n")
    roles = scan_files(root)
    print("\n【建议阅读顺序】")
    for role in READ_ORDER:
        if role in roles:
            print(f"  {role}：{roles[role][0]}")


if __name__ == "__main__":
    main()

示例 6：三期项目各抽一个纯函数，用 assert 自测（three_projects_core.py）

2.3 节说过「逻辑与 I/O 分离」的实际收益是「能写测试」。这个示例就是那个收益的成品：把三个项目里最该被测试的四段逻辑（飞船边界、矩形相交、按年统计、数据归属）各抽成一个纯函数，不 import pygame、不连数据库，然后用 assert 当场自测。这四条断言换一种写法就是 pytest 用例（练习 21-2 要求你做的正是这件事）。

Python# three_projects_core.py —— 三期项目各抽一个纯函数，用 assert 自测
from datetime import datetime


# ===== 项目 1：飞船移动的边界钳制（不碰 pygame 也能测）=====
def move_ship(x, speed, width, screen_width, moving_right):
    """返回移动后的 x；碰到左右边缘就停下，不许飞出屏幕。"""
    x += speed if moving_right else -speed
    if x + width >= screen_width:
        return screen_width - width
    if x <= 0:
        return 0
    return x


# ===== 项目 1：子弹和外星人的矩形相交 =====
def rects_hit(a, b):
    """a、b 是 (x, y, w, h)；返回 True 表示两个矩形有重叠。"""
    return (a[0] < b[0] + b[2] and a[0] + a[2] > b[0]
            and a[1] < b[1] + b[3] and a[1] + a[3] > b[1])


# ===== 项目 2：CSV 数据按年统计最高温 =====
def yearly_max(rows, column="TMAX"):
    """rows 是 (日期字符串, 温度的字符串或 None) 的列表。"""
    best = {}
    for date_text, value in rows:
        try:
            high = int(value)
        except (TypeError, ValueError):
            continue                       # 缺失值跳过，但要记数见调用方
        year = datetime.strptime(date_text, "%Y-%m-%d").year
        if year not in best or high > best[year]:
            best[year] = high
    return dict(sorted(best.items()))


# ===== 项目 3：数据归属判断（Django 的 is_owner 纯逻辑版）=====
def is_owner(resource, user):
    """resource 是 {"owner": "alice"} 这样的字典；未登录一律 False。"""
    if user is None or not user.get("is_authenticated"):
        return False
    return resource.get("owner") == user["username"]


# ================= 以下全是测试，没有一行需要外部库 =================
ship = 0

print("\n=== 矩形相交（子弹 3x15，外星人 50x40）===")
bullet = (400, 500, 3, 15)
for alien in [(400, 520, 50, 40), (400, 480, 50, 40), (300, 500, 50, 40)]:
    print(f"  子弹{bullet} vs 外星人{alien} -> {rects_hit(bullet, alien)}")

三条解读：

  1. move_ship 的两个边界返回的是「贴边后的坐标」而不是 True/False：调用方拿到值直接赋给 ship.rect.x，不需要再判断一次。这是「纯函数」的典型形态——输入状态、输出新状态，不修改任何外部对象，所以能反复调用、能随便断言。
  2. rects_hit 的四个不等号方向最容易写反：a[0] < b[0] + b[2]（a 的左 < b 的右）与 a[0] + a[2] > b[0]（a 的右 > b 的左）必须同时成立才算横向重叠，纵向同理。少写一个就把「相交」判成「只要挨着就算」，游戏里表现为「子弹离外星人还有半个身位就爆炸」。
  3. 把 yearly_max 的 try/except 去掉会怎样：int(None) 抛 TypeError、int("") 抛 ValueError，整个函数在第 4 行数据上崩掉——而这份数据里确实有两个坏值。这正是单元 16 讲的「真实数据有缺失」，也是「测试必须覆盖空值」的原因：assert yearly_max([]) == {} 这一条就专门守这个边界。
