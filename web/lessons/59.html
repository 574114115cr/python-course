示例 1：知识自查清单（可运行的版本）

Python# self_check.py —— 用代码把前 11 章串起来

def check_basics():
    """检查基础语法：变量、类型、格式化。"""
    name = "小明"
    age = 18
    height = 1.75
    is_student = True

    # 类型转换
    age_str = str(age)
    age_int = int(age_str)

    # f-string格式化
    message = f"{name} 今年 {age} 岁，身高 {height:.2f} 米"
    return message, age_int, is_student


def check_collections():
    """检查列表和字典操作。"""
    # 列表
    fruits = ["苹果", "香蕉"]
    fruits.append("橙子")
    fruits.insert(1, "葡萄")
    fruits.remove("香蕉")
    first = fruits[0]
    last = fruits[-1]
    sliced = fruits[1:3]
    reversed_list = fruits[::-1]

    # 列表推导式
    numbers = [1, 2, 3, 4, 5]
    evens = [n for n in numbers if n % 2 == 0]
    squares = [n ** 2 for n in numbers]

    # 字典
    scores = {"语文": 90}
    scores["数学"] = 85
    got = scores.get("英语", 0)
    keys = list(scores.keys())
    items = list(scores.items())

    # 元组
    point = (3, 5)
    x, y = point

    return {
        "列表": fruits, "首元素": first, "末元素": last,
        "切片": sliced, "反转": reversed_list,
        "偶数": evens, "平方": squares,
        "字典": scores, "默认值": got,
        "键": keys, "键值对": items, "元组": point, "x": x,
    }


def check_control_flow():
    """检查条件与循环。"""
    # if / elif / else
    def grade(score):
        if score >= 90:
            return "A"
        elif score >= 80:
            return "B"
        elif score >= 60:
            return "C"
        else:
            return "F"

    # for + range
    total = 0
    for i in range(1, 6):
        total += i

    # while + break
    count = 0
    while True:
        count += 1
        if count >= 3:
            break

    # continue
    skipped = []
    for i in range(6):
        if i % 2 == 0:
            continue
        skipped.append(i)

    return {
        "等级": [grade(s) for s in [95, 85, 65, 45]],
        "1到5之和": total,
        "循环次数": count,
        "奇数": skipped,
    }


def check_functions():
    """检查函数与参数。"""
    def calc(*args, **kwargs):
        """可变参数。"""
        return sum(args), kwargs

    def with_default(a, b=10, *rest, **extra):
        """默认参数 + 可变参数。"""
        return a, b, rest, extra

    def pure_function(data):
        """不修改传入列表的纯函数。"""
        return data + [999]

    original = [1, 2, 3]
    result = pure_function(original)

    return {
        "可变参数": calc(1, 2, 3, x=1, y=2),
        "默认参数": with_default(1),
        "纯函数原数据": original,   # [1, 2, 3] 没被改
        "纯函数结果": result,       # [1, 2, 3, 999]
    }


# 展示
if __name__ == "__main__":
    for title, func in [
        ("基础语法", check_basics),
        ("集合类型", check_collections),
        ("流程控制", check_control_flow),
        ("函数", check_functions),
    ]:
        print("=" * 50)
        print(f"【{title}】")
        print("=" * 50)
        result = func()
        if isinstance(result, tuple):
            for i, item in enumerate(result, 1):
                print(f"  {i}. {item}")
        elif isinstance(result, dict):
            for key, value in result.items():
                print(f"  {key}: {value}")
        print()

示例 2：类 + 继承的完整复习

Python# class_review.py —— 类的全部要素复习

class Shape:
    """抽象基类。"""
    def __init__(self, name):
        self.name = name

    def area(self):
        """子类必须重写。"""
        raise NotImplementedError("子类必须重写 area()")

    def describe(self):
        """共有的方法，所有子类都能用。"""
        return f"{self.name}的面积是 {self.area():.2f}"


class Rectangle(Shape):
    """矩形。"""
    def __init__(self, width, height):
        super().__init__("矩形")
        self.width = width
        self.height = height

    def area(self):
        """重写父类方法。"""
        return self.width * self.height


class Circle(Shape):
    """圆形。"""
    PI = 3.14159

    def __init__(self, radius):
        super().__init__("圆形")
        self.radius = radius

    def area(self):
        return self.PI * self.radius ** 2


class Square(Rectangle):
    """正方形 —— 多层继承。"""
    def __init__(self, side):
        super().__init__(side, side)

    def describe(self):
        """重写父类的方法。"""
        return f"正方形，边长 {self.width}"


# 多态验证
shapes = [Rectangle(3, 4), Circle(2), Square(5)]

for shape in shapes:
    print(shape.describe())

print()

# 属性名与方法名不能重名（常见错误）
class BadDemo:
    """INTENTIONALLY-BROKEN 演示：属性名和方法名重名会互相覆盖"""
    def __init__(self):
        self.perimeter = 10      # 属性

    def perimeter(self):         # 同名方法覆盖了属性
        return 20

# 正确做法：给属性加前缀区分
class GoodDemo:
    def __init__(self, width, height):
        self._perimeter_value = 2 * (width + height)

    def perimeter(self):
        return self._perimeter_value


good = GoodDemo(3, 4)
print(f"正确写法：perimeter() = {good.perimeter()}")   # 14

示例 3：文件 + 异常完整复习

Python# file_review.py —— 文件与异常的完整复习

import json
import csv
import os
import tempfile


def file_basics_demo():
    """读写文件与三种模式。"""
    filename = "复习_文件.txt"

    # w：写入并清空
    with open(filename, "w", encoding="utf-8") as f:
        f.write("第一行\n")

    # a：追加
    with open(filename, "a", encoding="utf-8") as f:
        f.write("第二行\n")

    # r：读取全部
    with open(filename, "r", encoding="utf-8") as f:
        content = f.read()

    # 逐行读（推荐方式）
    lines = []
    with open(filename, "r", encoding="utf-8") as f:
        for line in f:
            lines.append(line.strip())

    os.remove(filename)
    return content, lines


def json_demo():
    """JSON 读写。"""
    filename = "复习_data.json"
    data = {"name": "小明", "scores": [90, 85, 92], "active": True}

    # 写
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

    # 读
    with open(filename, "r", encoding="utf-8") as f:
        loaded = json.load(f)

    os.remove(filename)
    return loaded, loaded == data


def csv_demo():
    """CSV 读写。"""
    filename = "复习_data.csv"
    rows = [["姓名", "成绩"], ["小明", 92], ["小红", 88]]

    # 写
    with open(filename, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        for row in rows:
            writer.writerow(row)

    # 读
    with open(filename, "r", newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        data = list(reader)

    os.remove(filename)
    return data


def exception_demo():
    """四种异常结构。"""
    results = []

    # try / except
    try:
        int("abc")
    except ValueError as error:
        results.append(f"捕获异常：{error}")

    # try / except / else
    try:
        value = int("42")
    except ValueError:
        results.append("不该走到这里")
    else:
        results.append(f"else 分支：转换成功 {value}")

    # try / finally
    def with_finally():
        try:
            return "try 的返回值"
        finally:
            results.append("finally 一定会执行")

    with_finally()

    # 主动抛出
    def validate(age):
        if age < 0:
            raise ValueError("年龄不能为负")
        return True

    try:
        validate(-1)
    except ValueError as error:
        results.append(f"主动抛出被捕获：{error}")

    return results


if __name__ == "__main__":
    print("【文件读写】")
    print(file_basics_demo())

    print("\n【JSON】")
    print(json_demo())

    print("\n【CSV】")
    print(csv_demo())

    print("\n【异常】")
    for item in exception_demo():
        print(" ", item)

Python# test_review.py —— 这是一个可以直接运行的 pytest 测试文件

import json
import os
import tempfile

import pytest


# ===== 被测函数 =====
def add(a, b):
    """两数相加。"""
    return a + b


def divide(a, b):
    """两数相除，除零抛异常。"""
    return a / b


class Account:
    """账户类。"""
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("金额必须大于 0")
        self.balance += amount
        return self.balance


class SavingsAccount(Account):
    """储蓄账户，比普通账户多一个利率。"""
    def __init__(self, owner, balance=0, rate=0.03):
        super().__init__(owner, balance)
        self.rate = rate

    def add_interest(self):
        """计算利息。"""
        interest = self.balance * self.rate
        self.balance += interest
        return interest


# ===== 测试函数 =====
def test_add():
    """正常用例。"""
    assert add(2, 3) == 5
    assert add(-1, 1) == 0


@pytest.mark.parametrize("a,b,expected", [
    (10, 2, 5.0),
    (9, 3, 3.0),
    (-10, 2, -5.0),
])
def test_divide(a, b, expected):
    """参数化测试。"""
    assert divide(a, b) == expected


def test_divide_by_zero():
    """异常用例。"""
    with pytest.raises(ZeroDivisionError):
        divide(10, 0)


# ===== 测试类 =====
class TestAccount:
    def test_default_balance(self):
        assert Account("小明").balance == 0

    def test_deposit(self):
        account = Account("小明", 100)
        account.deposit(50)
        assert account.balance == 150

    @pytest.mark.parametrize("amount", [0, -1, -100])
    def test_invalid_deposit(self, amount):
        """异常用例：非法金额。"""
        with pytest.raises(ValueError):
            Account("小明").deposit(amount)


class TestSavingsAccount:
    """测试继承。"""
    def test_inherits_account_attributes(self):
        account = SavingsAccount("小红", 1000)
        assert account.owner == "小红"
        assert account.balance == 1000

    def test_default_rate(self):
        assert SavingsAccount("小红").rate == 0.03

    def test_add_interest(self):
        account = SavingsAccount("小红", 1000, 0.05)
        interest = account.add_interest()
        assert interest == pytest.approx(50.0)
        assert account.balance == pytest.approx(1050.0)

    def test_is_account_instance(self):
        """子类对象也是父类实例。"""
        assert isinstance(SavingsAccount("小红"), Account)


# ===== 文件测试（用临时目录，不污染工作区）=====
def test_json_roundtrip():
    """测试 JSON 写入再读出，数据应该一致。"""
    # tempfile 提供临时目录，测试结束后自动清理
    with tempfile.TemporaryDirectory() as tmpdir:
        filename = os.path.join(tmpdir, "test.json")
        data = {"name": "小明", "scores": [90, 85]}

        # 写
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False)

        # 读
        with open(filename, "r", encoding="utf-8") as f:
            loaded = json.load(f)

    assert loaded == data
    assert loaded["name"] == "小明"


def test_read_nonexistent_file_raises():
    """读取不存在的文件应该抛 FileNotFoundError。"""
    with pytest.raises(FileNotFoundError):
        with open("根本不存在.json", "r", encoding="utf-8") as f:
            f.read()

示例 5：一个小而完整的项目（类 + JSON + 异常）

阶段 1 学的东西能不能串起来，用一个 40 行的记账本就看得出来：类负责状态，JSON 负责持久化， 异常负责「文件不在」「文件坏了」「输入非法」这三种现实情况。

Python# ledger.py —— 换成你自己的路径也能直接跑
import json
import os
import tempfile


class Ledger:
    def __init__(self, path):
        self.path = path
        self.items = []
        self.load()

    def load(self):
        try:
            with open(self.path, encoding="utf-8") as f:
                self.items = json.load(f)
        except FileNotFoundError:
            self.items = []          # 第一次运行：没有文件是正常的
        except json.JSONDecodeError:
            self.items = []          # 文件坏了：不崩，从空账本开始
            print("！账本文件损坏，已重置")

    def add(self, kind, amount, note=""):
        if kind not in ("收", "支"):
            raise ValueError("kind 只能是「收」或「支」")
        if amount <= 0:
            raise ValueError("金额必须大于 0")
        self.items.append({"kind": kind, "amount": amount, "note": note})

    def income(self):
        return sum(i["amount"] for i in self.items if i["kind"] == "收")

    def spend(self):
        return sum(i["amount"] for i in self.items if i["kind"] == "支")

    def balance(self):
        return self.income() - self.spend()

    def save(self):
        with open(self.path, "w", encoding="utf-8") as f:
            json.dump(self.items, f, ensure_ascii=False, indent=2)


path = os.path.join(tempfile.mkdtemp(), "ledger.json")
book = Ledger(path)
print("空账本：", book.items, "余额", book.balance())

book.add("收", 8000, "工资")
book.add("支", 1200, "房租")
book.add("支", 35.5, "午饭")
print(f"收入 {book.income()} 支出 {book.spend()} 余额 {book.balance()}")

book.save()
print("写入文件：", os.path.basename(path), os.path.getsize(path), "字节")

again = Ledger(path)             # 重新读一次，验证真的落盘了
print("重新读入条数：", len(again.items), "余额", again.balance())

for kind, amount in (("借", 100), ("支", 0)):
    try:
        again.add(kind, amount)
    except ValueError as e:
        print("拒绝：", e)

运行结果：

示例空账本： [] 余额 0
收入 8000 支出 1235.5 余额 6764.5
写入文件： ledger.json 225 字节
重新读入条数： 3 余额 6764.5
拒绝： kind 只能是「收」或「支」
拒绝： 金额必须大于 0

四个值得注意的设计点：

  • load() 在 __init__ 里调，所以「新建对象」永远等于「拿到当前磁盘状态」，调用方不用记得先加载。
  • 两个 except 分开写：FileNotFoundError 是正常的第一次运行，JSONDecodeError 是异常情况—— 虽然处理方式都是「空账本」，但后者要打印提示，合成一个 except 就把这两种情况混在一起了。
  • add() 主动抛 ValueError：与其让非法数据进账本、后面算错总额，不如在入口就拦住。
  • save() 用 ensure_ascii=False：否则文件里存的是 \u5de5\u8d44 这种转义，肉眼没法校对。

示例 6：给上面的项目写 6 条断言（不装 pytest 也能跑）

Python# check_ledger.py —— 断言就是最小的测试：跑通没报错，就说明这几条不变式成立
import json
import os
import tempfile


def load(path):
    try:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        return []


def add(items, kind, amount):
    if kind not in ("收", "支"):
        raise ValueError("kind")
    if amount <= 0:
        raise ValueError("amount")
    items.append({"kind": kind, "amount": amount})


work = tempfile.mkdtemp()
fresh = os.path.join(work, "a.json")
broken = os.path.join(work, "b.json")
open(broken, "w", encoding="utf-8").write("{不是 JSON")

assert load(fresh) == []                  # 文件不存在 → 空账本，不抛异常
assert load(broken) == []                 # 内容坏了 → 也是空账本
assert load(fresh) is not load(fresh)     # 每次都是新列表，不会共享

items = []
add(items, "收", 100)
add(items, "支", 30)
assert sum(i["amount"] for i in items if i["kind"] == "收") == 100
assert sum(i["amount"] for i in items if i["kind"] == "支") == 30

for kind, amount in (("借", 10), ("支", 0), ("支", -5)):
    try:
        add(items, kind, amount)
        raise AssertionError(f"应该拒绝 {kind} {amount}")
    except ValueError:
        pass

print(f"6 条断言全部通过，账本仍有 {len(items)} 条记录")

运行结果：

示例6 条断言全部通过，账本仍有 2 条记录

注意最后那个 raise AssertionError(...)：它写在 try 里、except ValueError 抓不到它， 所以只要 add() 没有拒绝非法输入，这条断言就会真的失败。这是「测异常」的固定套路—— 不是断言「没报错」，而是断言「该报的错报了」。

上一章用 pytest 写测试，这一章用 assert：两者检查的是同一件事， 区别只在 pytest 会告诉你哪一条失败、失败在第几行。没有 pytest 的环境里，assert 也够用。
