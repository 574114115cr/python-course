练习 6-8【嵌套 while：九九乘法表前 3 行】 用嵌套 while 循环输出九九乘法表的前 3 行：

代码1×1=1

提示：外层循环控制行号，内层循环控制每行有几个式子。 注意观察：内层循环的变量在每轮外层循环开始时必须重置，否则会出错——这是嵌套循环最经典的 bug。

Python# sum_to_n.py —— while 与 for 两种写法的对比

n = int(input("请输入一个正整数 n："))

# ===== while 写法 =====
total = 0
i = 1
while i <= n:
    total += i
    i += 1
print(f"while 写法：1 到 {n} 的和是 {total}")

# ===== for 写法（更简洁，因为次数已知）=====
total2 = sum(range(1, n + 1))
print(f"for 写法：1 到 {n} 的和是 {total2}")

# ===== 另一种 while 写法：用累加器技巧 =====
total3 = 0
counter = n
while counter > 0:
    total3 += counter
    counter -= 1
print(f"倒着加：{total3}")

# 验证几个固定答案
for n in [1, 5, 10, 100]:
    print(f"n={n} 时和为 {sum(range(1, n + 1))}")
# n=1 -> 1, n=5 -> 15, n=10 -> 55, n=100 -> 5050

Python# multiples.py —— 用 while 输出倍数

# 1）1–50 之间所有 7 的倍数
print("--- 7 的倍数 ---")
n = 1
while n <= 50:
    if n % 7 == 0:
        print(n)
    n += 1

# 更简洁的等价写法：用 while 变量本身就以 7 为步长
print("--- 7 的倍数（步长版）---")
n = 7
while n <= 50:
    print(n)
    n += 7

# 2）1–50 之间同时是 3 和 5 的倍数（即 15 的倍数）
print("--- 同时是 3 和 5 的倍数 ---")
n = 1
while n <= 50:
    if n % 3 == 0 and n % 5 == 0:
        print(n)
    n += 1
# 输出：15 30 45

# 也可以用"lcm"（最小公倍数）的思路：15 的倍数就是答案
print("--- 15 的倍数 ---")
n = 15
while n <= 50:
    print(n)
    n += 15

# 顺带做一个和：1–100 之间所有 7 的倍数之和
total = 0
n = 7
while n <= 100:
    total += n
    n += 7
print(f"7 的倍数之和：{total}")   # 7+14+...+98 = 735

Python# find_first.py —— 用 break 找第一个符合条件的数

numbers = [12, 5, 33, 7, 91, 26]

# ===== 写法一：用标志位（能表达"没找到"的情况）=====
target_value = 30
found = False
found_index = -1

for index, number in enumerate(numbers):
    if number > target_value:
        found_index = index
        found = True
        break                # 找到第一个就停

if found:
    print(f"第一个大于 {target_value} 的数是 {numbers[found_index]}，"
          f"下标是 {found_index}")
else:
    print(f"没有找到大于 {target_value} 的数")

# ===== 写法二：用一个哨兵值（-1）表达"没找到" =====
def find_first(items, limit):
    """返回第一个大于 limit 的元素及其下标，都找不到时返回 (None, -1)。"""
    for index, item in enumerate(items):
        if item > limit:
            return item, index
    return None, -1


result, index = find_first(numbers, 30)
if result is None:
    print("没有找到")
else:
    print(f"找到 {result}，下标 {index}")

# ===== 测试边界情况：全部都小于阈值 =====
print(find_first([1, 2, 3], 100))    # (None, -1)
print(find_first([1, 50, 3], 30))    # (50, 1)

Python# guess_number.py —— 完整的猜数字游戏

import random


def play_guess_number():
    """启动猜数字游戏主流程。"""
    # 生成 1–100 的随机整数
    secret = random.randint(1, 100)
    count = 0
    history = []             # 记录猜测历史，方便复盘

    print("=" * 40)
    print("欢迎来到猜数字游戏！")
    print("我已经想好了一个 1 到 100 之间的整数")
    print("输入 'quit' 可以随时退出")
    print("=" * 40)

    # ===== 标准骨架：while True + 唯一出口 break =====
    while True:
        raw = input("请输入你的猜测（1-100）：")

        # 退出通道：用户主动结束
        if raw in ["quit", "q", "exit"]:
            print("游戏已退出，答案是", secret)
            print("你的猜测记录：", history)
            break

        # 把字符串转成整数，转换失败就重新问
        try:
            guess = int(raw)
        except ValueError:
            print(f"'{raw}' 不是有效数字，请重新输入")
            continue

        # 超出范围提示
        if guess < 1 or guess > 100:
            print("请输入 1 到 100 之间的数字")
            continue

        count += 1
        history.append(guess)

        if guess < secret:
            print(f"太小了，再大一点（第 {count} 次猜测）")
        elif guess > secret:
            print(f"太大了，再小一点（第 {count} 次猜测）")
        else:
            print("=" * 40)
            print(f"恭喜你猜对了！答案就是 {secret}")
            print(f"你一共猜了 {count} 次")
            print(f"你的猜测记录：{history}")
            print("=" * 40)
            break                   # 猜对了，唯一的正常出口

    return count


# 游戏结束后统计一下平均猜测次数
counts = [play_guess_number()]
print(f"本次游戏共猜了 {counts[0]} 次")

Python# password_retry.py —— 用 while...else 实现重试机制

correct_password = "py2026"
max_attempts = 3
attempt = 0

while attempt < max_attempts:
    password = input(f"请输入密码（还可尝试 {max_attempts - attempt} 次）：")
    attempt += 1

    # 主动放弃
    if password == "exit":
        print("你选择了放弃本次输入")
        break

    if password == correct_password:
        print(f"密码正确！你在第 {attempt} 次尝试时成功登录")
        break
    else:
        print("密码错误")
else:
    # 只有循环正常走完（尝试满 3 次都没 break）才会到这里
    print("-" * 30)
    print("错误次数已达上限，账户被锁定")
    print("-" * 30)

# ===== 对比：如果不用 while...else，长什么样 =====
# 需要一个额外的标志位来记住"是不是正常走完的"
# attempt < max_attempts 这个循环结束后，
# 如果 attempt == max_attempts，说明三次都错了
attempt2 = 0
while attempt2 < max_attempts:
    pwd = input("密码：")
    if pwd == "exit":
        break
    if pwd == correct_password:
        print("成功")
        break
    attempt2 += 1

if attempt2 == max_attempts:
    print("账户锁定")
# 结论：while...else 把"是否正常结束"这件事表达得干净得多

Python# input_validation.py —— 读入一个有效的正整数

print("请输入一个正整数：")

while True:
    raw = input("> ")

    # ===== 方式一：try/except（推荐，单元 9 会正式讲）=====
    try:
        number = int(raw)
    except ValueError:
        print(f"'{raw}' 不是有效的整数，请重新输入")
        continue             # 回到循环开头重新要一次

    # 整数校验通过，再检查范围
    if number <= 0:
        print("请输入大于 0 的整数")
        continue

    # 到这里说明输入完全合法，用 break 跳出无限循环
    print(f"你输入的数是 {number}")
    break

print("程序继续往下执行，说明已经拿到了有效输入")

# ===== 方式二：不用 try，用 str.isdigit() =====
# 注意 isdigit() 的局限：
#   "12".isdigit()    -> True
#   "-12".isdigit()   -> False  （负号不是数字字符）
#   "1.5".isdigit()   -> False  （小数点不是数字字符）
print("\n--- 用 isdigit 的版本 ---")
while True:
    raw = input("> ")
    if not raw.isdigit():
        print("请输入纯数字")
        continue
    number = int(raw)
    if number <= 0:
        print("请输入大于 0 的整数")
        continue
    print(f"你输入的数是 {number}")
    break

Python# multiplication_table.py —— 嵌套 while 循环

# 完整九九乘法表
print("--- 完整九九乘法表 ---")
i = 1
while i <= 9:
    j = 1
    while j <= i:                     # 内层：每行输出 i 个式子
        print(f"{j}×{i}={j * i}", end="\t")
        j += 1
    print()# 换行，让下一个 i 另起一行
    i += 1

# 只输出前 3 行
print("\n--- 前 3 行 ---")
i = 1
while i <= 3:
    j = 1
    while j <= i:
        print(f"{j}×{i}={j * i}", end="\t")
        j += 1
    print()
    i += 1

# ===== 嵌套循环的经典 bug：内层变量没重置 =====
# 下面这段会输出错误结果
# i = 1
# while i <= 3:
#     j = 1
#     while j <= i:
#         print(j, end=" ")
#         j += 1
#     print()
#     # 忘记写 j = 1，第二轮 i=2 时 j 已经是 2，循环一次都不执行
#     i += 1

# 正确写法：内层循环的变量必须在每轮外层循环里重新初始化
print("\n--- 固定版---")
i = 1
while i <= 3:
    j = 1                 # 关键：每轮外层循环开始时重置
    while j <= i:
        print(f"{j}×{i}={j * i}", end="\t")
        j += 1
    print()
    i += 1

# 观察输出：
# 1×1=1
# 1×2=2   2×2=4
# 1×3=3   2×3=6   3×3=9

Python# clean_scores.py —— 两种「删干净」的写法
scores = [88, 0, 92, 0, 75, 0, 100]

# 写法一：while + remove（remove 一次只删一个，所以要用循环）
while 0 in scores:
    scores.remove(0)
print(f"有效成绩 {len(scores)} 个，平均 {sum(scores) / len(scores):.1f} 分")

# 写法二：遍历副本，验证结果一致
scores2 = [88, 0, 92, 0, 75, 0, 100]
for s in scores2[:]:            # [:] 造一份副本，删原列表不影响遍历
    if s == 0:
        scores2.remove(s)
print("两次结果一致：", scores == scores2)

输出：

示例有效成绩 4 个，平均 88.8 分
两次结果一致： True

如果写成 for s in scores2:（不加 [:]），就会命中 2.7 讲的「边遍历边删」问题：删掉第一个 0 后，后面元素前移，下标跳过一项，最后会剩一个 0。这是本单元最值得手动复现一次的 bug。

Python# signup.py —— while + input 收集，再按姓名排序输出
signups = {}

while True:
    name = input("姓名（输入 q 结束）：").strip()
    if name == "q":
        break
    if not name:                       # 防手滑直接回车
        continue
    if name in signups:
        print(f"{name} 已报名，将覆盖原选择")
    signups[name] = input("想学的课程：").strip()

print(f"共 {len(signups)} 人报名")
for name in sorted(signups):           # 按姓名字母序，不排序则按插入顺序
    print(f"{name} → {signups[name]}")

三个值得注意的地方：

  • input(...).strip() 顺手去掉首尾空格，否则用户多敲一个空格，「张三」和「张三 」会被当成两个人。
  • if not name: continue 处理「直接回车」的情况，避免把空姓名写进字典。
  • sorted(signups) 排的是键，所以直接得到按姓名排序的结果；要按课程排就写 sorted(signups.items(), key=lambda kv: kv[1])（单元 3 会讲 key）。
