✅ 验收 1：手写猜数字游戏

要求：合上所有资料，10 分钟内独立写出一个可运行的猜数字游戏。

必须包含： - random.randint(1, 100) 生成随机数 - while 循环持续接收猜测 - if 判断大了/小了/对了 - 猜对后用 break 跳出并输出猜测次数

参考实现：

Python# 关卡验收 1：猜数字游戏
import random

secret = random.randint(1, 100)
count = 0

print("我想了一个 1-100 的数字，猜猜是几？")

while True:
    raw = input("请输入你的猜测（q 退出）：")

    if raw in ["q", "quit"]:
        print(f"游戏结束，答案是 {secret}")
        break

    try:
        guess = int(raw)
    except ValueError:
        print("请输入有效数字")
        continue

    count += 1

    if guess < secret:
        print("太小了，再大一点")
    elif guess > secret:
        print("太大了，再小一点")
    else:
        print(f"恭喜猜对了！答案是 {secret}，你一共猜了 {count} 次")
        break

自检清单： - [ ] 能一次运行成功，没有语法错误 - [ ] 输入非数字不会崩溃 - [ ] 能通过输入 q 退出 - [ ] 猜测次数统计正确

✅ 验收 2：手写一个含继承的类

要求：独立设计一个至少三层的类结构（用到继承和 super()），并能说明每个类的职责。

参考实现：

Python# 关卡验收 2：含继承的三层类设计

class Vehicle:
    """交通工具基类。"""
    def __init__(self, name, price):
        self.name = name
        self.price = price

    def __str__(self):
        return f"{self.name}（{type(self).__name__}，{self.price} 元）"

    def describe(self):
        """基类的通用描述，子类可以重写。"""
        return f"这是一辆 {self.name}"


class Car(Vehicle):
    """汽车。"""
    def __init__(self, name, price, fuel_consumption):
        super().__init__(name, price)
        self.fuel_consumption = fuel_consumption    # 每百公里耗油量

    def fuel_cost(self, distance, price_per_liter=8):
        """计算行驶指定距离的油费。"""
        liters = distance / 100 * self.fuel_consumption
        return liters * price_per_liter


class ElectricCar(Car):
    """电动汽车 —— 继承 Car，但重写相关方法。"""
    def __init__(self, name, price, battery_capacity):
        super().__init__(name, price, fuel_consumption=0)
        self.battery_capacity = battery_capacity      # 电池容量 kWh
        self.charge = 100                             # 剩余电量百分比

    def fuel_cost(self, distance, price_per_liter=0):
        """★ 重写：电动车不用油，用电。"""
        electricity_per_100km = 15                   # 百公里耗电 15 度
        kwh = distance / 100 * electricity_per_100km
        return kwh * 0.6                             # 电价 0.6 元/度

    def describe(self):
        """★ 重写：用父类信息 + 自己的信息。"""
        return f"{super().describe()}，纯电动，电池 {self.battery_capacity} kWh"


# ===== 测试 =====
vehicles = [
    Car("燃油轿车", 120000, 7.5),
    ElectricCar("特斯拉 Model 3", 280000, 60),
    Car("越野车", 260000, 12.0),
]

print("--- 多态：同一段代码调用 describe() ---")
for v in vehicles:
    print(v.describe())

print("\n--- 费用对比（同一段代码，行为不同）---")
distance = 300
for v in vehicles:
    cost = v.fuel_cost(distance)
    print(f"{v.name}：行驶 {distance} 公里，成本 {cost:.2f} 元")

print("\n--- 验证继承关系 ---")
ec = ElectricCar("测试", 100000, 50)
print(f"是 Car 的实例吗：{isinstance(ec, Car)}")
print(f"是 Vehicle 的实例吗：{isinstance(ec, Vehicle)}")
print(f"继承自：{[c.__name__ for c in type(ec).__mro__]}")
print(f"重写的方法：ElectricCar.fuel_cost 是 Car.fuel_cost 吗？"
      f"{ElectricCar.fuel_cost is Car.fuel_cost}")

自检清单： - [ ] 至少三层（基类 + 两个子类） - [ ] 子类的 __init__ 调用了 super().__init__() - [ ] 至少一个方法被重写，且重写后行为确实不同 - [ ] 能用 isinstance 验证继承关系 - [ ] 能说明「为什么用继承而不是复制粘贴」

✅ 验收 3：用 pytest 写 3 条测试并全部通过

要求：给上面验收 2 的类（或你自己写的任何函数）写 3 条测试，用 pytest 运行并全部通过。

参考实现：

Python# test_vehicle.py —— 关卡验收 3：给验收 2 的类写测试

import pytest

from vehicle import Car, ElectricCar


class TestCar:
    """测试汽车类。"""

    def test_fuel_cost(self):
        """正常情况：300 公里、7.5L/100km、油价 8 元。"""
        car = Car("测试车", 100000, 7.5)
        # 300/100 * 7.5 = 22.5 升，22.5 * 8 = 180 元
        assert car.fuel_cost(300) == 180.0

    def test_zero_distance(self):
        """★ 边界：不走就不花钱。"""
        car = Car("测试车", 100000, 7.5)
        assert car.fuel_cost(0) == 0.0


class TestElectricCar:
    """测试电动汽车类。"""

    def test_inherits_from_car(self):
        """★ 验证继承关系。"""
        ec = ElectricCar("测试电动车", 200000, 60)
        assert isinstance(ec, Car)

    def test_fuel_cost_is_overridden(self):
        """★ 验证方法重写：电动车不走油费逻辑。"""
        car = Car("燃油车", 100000, 7.5)
        ec = ElectricCar("电动车", 200000, 60)

        # 同样跑 300 公里，电动车成本应该远低于燃油车
        car_cost = car.fuel_cost(300)
        electric_cost = ec.fuel_cost(300)

        assert electric_cost == 27.0      # 300/100*15*0.6 = 27
        assert electric_cost < car_cost

# 错误 类型 修复
