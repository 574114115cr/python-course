练习 8-3【学生类】 写 Student 类： - 属性：name、scores（成绩列表） - add_score(score)：添加一次成绩（不要用 append 直接在外部改） - average()：返回平均分（用 sum() 和 len()） - highest()：返回最高分 - is_pass()：平均分 ≥ 60 返回 True - __str__：输出"XXX 的平均分是 X.XX" 要求：注意副本语义——如果 add_score 里用 self.scores.append()，它是改自己的属性，没问题；但如果直接把 scores 列表传给构造函数，要考虑外部修改列表会影响对象。

class Dog:
    """一只狗。

    属性：
        name: 名字（字符串）
        age:  年龄（整数），默认 1岁
    """

    def __init__(self, name, age=1):
        """初始化一只狗。"""
        self.name = name
        self.age = age

    def bark(self):
        """让狗叫一声。"""
        print(f"{self.name} 说：汪汪汪！")

    def birthday(self):
        """过生日，年龄加一。"""
        self.age += 1
        print(f"🎂 {self.name} 生日快乐，现在 {self.age} 岁了！")

    def get_info(self):
        """返回包含基本信息的字典。"""
        return {
            "name": self.name,
            "age": self.age,
            "is_adult": self.age >= 3,
        }

    def __str__(self):
        """定义对象转成字符串时的样子。"""
        return f"{self.name} 是一只 {self.age} 岁的狗"

Python# rectangle_class.py —— 矩形类

class Rectangle:
    """矩形。"""
    def __init__(self, width, height):
        self.width = width
        self.height = height

    def area(self):
        """返回面积。"""
        return self.width * self.height

    def perimeter(self):
        """返回周长。"""
        return 2 * (self.width + self.height)

    def is_square(self):
        """返回是否是正方形（宽等于高）。"""
        return self.width == self.height

    def __str__(self):
        return f"宽 {self.width}，高 {self.height}"


# ===== 测试：用元组列表存数据 =====
rectangles = [
    Rectangle(3, 4),
    Rectangle(5, 5),      # 正方形
    Rectangle(2, 8),
    Rectangle(6, 6),      # 正方形
    Rectangle(7, 3),
]

print(f"{'矩形':<20}{'面积':>10}{'周长':>10}{'是否正方形':>12}")
print("-" * 52)

square_count = 0
total_area = 0

for rect in rectangles:
    is_square = rect.is_square()
    if is_square:
        square_count += 1
    total_area += rect.area()
    print(
        f"{str(rect):<20}{rect.area():>10}{rect.perimeter():>12}"
        f"{'是' if is_square else '否':>12}"
    )

print("-" * 52)
print(f"总共{len(rectangles)} 个矩形，正方形有 {square_count} 个")
print(f"面积总和：{total_area}")

# 单个矩形的详细信息
print("\n--- 单个矩形 ---")
r = Rectangle(10, 4)
print(f"{r} 面积 {r.area()}，周长 {r.perimeter()}")
print("是正方形吗：", r.is_square())

Python# student_class.py —— 学生类

class Student:
    """学生。"""
    def __init__(self, name, scores=None):
        self.name = name
        # 注意：这里用 None 然后再造列表，避免直接引用外部列表
        self.scores = list(scores) if scores else []

    def add_score(self, score):
        """添加一次成绩。"""
        self.scores.append(score)
        return self.scores

    def average(self):
        """返回平均分，没有成绩时返回 0。"""
        if not self.scores:
            return 0
        return sum(self.scores) / len(self.scores)

    def highest(self):
        """返回最高分，没有成绩时返回 None。"""
        if not self.scores:
            return None
        return max(self.scores)

    def lowest(self):
        """返回最低分。"""
        if not self.scores:
            return None
        return min(self.scores)

    def is_pass(self):
        """平均分 60 分以上算及格。"""
        return self.average() >= 60

    def __str__(self):
        return f"{self.name} 的平均分是 {self.average():.2f}"


# ===== 测试 =====
student = Student("小明")
for score in [88, 92, 76, 95, 61]:
    student.add_score(score)

print(student)
print(f"最高分：{student.highest()}")
print(f"最低分：{student.lowest()}")
print(f"及格了吗：{'是' if student.is_pass() else '否'}")
print(f"全部成绩：{student.scores}")

# 边界情况：没有成绩
print("\n--- 没有成绩的情况 ---")
empty = Student("小刚")
print(f"平均分：{empty.average()}")
print(f"最高分：{empty.highest()}")
print(f"及格了吗：{empty.is_pass()}")

# ===== 副本语义演示 =====
print("\n--- 副本语义：构造时传入的列表是否被引用 ---")
original_scores = [70, 80]
student2 = Student("小红", original_scores)
student2.add_score(90)

print(f"student2 的成绩：{student2.scores}")     # [70, 80, 90]
print(f"原始列表：{original_scores}")            # [70, 80]  没变！

# 原因：构造函数里写了 list(scores)，造了一份副本。
# 如果去掉 list() 直接self.scores = scores，
# 那么student2.scores 和 original_scores 就是同一个对象，
# student2 加分会影响原始列表。

# 多名学生管理
print("\n--- 班级成绩单 ---")
students = [
    Student("小明", [88, 92, 76]),
    Student("小红", [95, 89, 99]),
    Student("小刚", [45, 60, 55]),
]

for s in students:
    print(f"  {s}  最高 {s.highest()} 分  "
          f"{'及格' if s.is_pass() else '不及格'}")

passed = [s.name for s in students if s.is_pass()]
print("及格名单：", passed)

Python# car_class.py —— 类属性与实例属性的实践

class Car:
    """汽车类。"""

    # ===== 类属性：所有实例共享 =====
    wheels = 4
    made_in = "中国"

    def __init__(self, make, model, color="白色"):
        # ===== 实例属性：每个对象独有 =====
        self.make = make
        self.model = model
        self.color = color
        self.odometer = 0

    def drive(self, miles):
        """行驶指定里程，累加总里程。"""
        self.odometer += miles
        return self.odometer

    def is_new(self):
        """判断是否还是新车（里程小于 100 公里）。"""
        return self.odometer < 100

    def show_info(self):
        """输出完整信息。"""
        status = "新车" if self.is_new() else "二手车"
        return (
            f"{self.color} {self.make} {self.model} | "
            f"里程 {self.odometer} 公里 | {status} | "
            f"{self.wheels} 轮 | {self.made_in}制造"
        )

    def __str__(self):
        return f"{self.make} {self.model}"


# ===== 测试 =====
car1 = Car("丰田", "卡罗拉", "红色")
car2 = Car("本田", "雅阁")           # color 用默认值"白色"

print("--- 初始状态 ---")
print("car1:", car1.show_info())
print("car2:", car2.show_info())

# 修改实例属性：只影响这一个对象
car1.color = "黑色"
print("\n--- 改 car1 的颜色 ---")
print("car1:", car1.color)      # 黑色
print("car2:", car2.color)      # 白色  没变！

# 行驶后里程变化
print("\n--- 行驶之后 ---")
car1.drive(150)
car2.drive(30)
print("car1:", car1.show_info())   # 二手车
print("car2:", car2.show_info())   # 还是新车

# 修改类属性：影响所有对象
print("\n--- 修改类属性 wheels = 6 ---")
Car.wheels = 6
print("car1:", car1.wheels)       # 6
print("car2:", car2.wheels)       # 6都变了

# 陷阱：不要通过实例改类属性
print("\n--- 陷阱演示 ---")
car1.wheels = 8                  # 这样做创建的是 car1 的实例属性
print("car1.wheels =", car1.wheels)    # 8
print("car2.wheels =", car2.wheels)    # 6  没跟着变
print("Car.wheels =", Car.wheels)      # 6  类属性也没变
print("结论：改类属性必须写 Car.wheels = X，写 car1.wheels = X 是错的")

# 恢复
Car.wheels = 4

Python# shapes.py —— 多态实战：统一管理不同形状

class Shape:
    """形状基类：定义所有形状共有的东西。"""

    def __init__(self, name):
        self.name = name

    def area(self):
        """计算面积 —— 由子类各自重写。"""
        print(f"警告：{self.name} 没有实现 area()，父类返回 None")
        return None

    def perimeter(self):
        """计算周长 —— 由子类各自重写。"""
        print(f"警告：{self.name} 没有实现 perimeter()，父类返回 None")
        return None

    def __str__(self):
        return f"{self.name}（{type(self).__name__}）"


class Rectangle(Shape):
    """矩形。"""
    def __init__(self, width, height):
        # 先调用父类的 __init__ 设置共有的 name 属性
        super().__init__("矩形")
        # 再设置自己的属性
        self.width = width
        self.height = height

    def area(self):
        """重写：矩形的面积。"""
        return self.width * self.height

    def perimeter(self):
        """重写：矩形的周长。"""
        return 2 * (self.width + self.height)


class Circle(Shape):
    """圆形。"""
    PI = 3.14159      # 类属性：圆周率

    def __init__(self, radius):
        super().__init__("圆形")
        self.radius = radius

    def area(self):
        """重写：圆的面积。"""
        return self.PI * self.radius ** 2

    def perimeter(self):
        """重写：圆的周长。"""
        return 2 * self.PI * self.radius


class Triangle(Shape):
    """三角形。"""
    def __init__(self, base, height):
        super().__init__("三角形")
        self.base = base
        self.height = height

    def area(self):
        return self.base * self.height / 2

    def perimeter(self):
        # 假设是等腰三角形，两腰长为 sqrt((base/2)^2 + height^2)
        side = ((self.base / 2) ** 2 + self.height ** 2) ** 0.5
        return self.base + 2 * side


class Square(Rectangle):
    """正方形 —— 继承矩形，只需要一个边长。"""
    def __init__(self, side):
        # 直接调用父类（矩形）的 __init__
        super().__init__(side, side)


# ===== 测试：多态的关键演示 =====
shapes = [
    Rectangle(3, 4),
    Circle(2),
    Triangle(6, 4),
    Square(5),
]

print(f"{'形状':<20}{'面积':>12}{'周长':>12}")
print("-" * 44)

total_area = 0
for shape in shapes:
    area = shape.area()          # 每种形状自己算自己的
    perimeter = shape.perimeter()
    total_area += area
    print(f"{str(shape):<20}{area:>12.2f}{perimeter:>12.2f}")

print("-" * 44)
print(f"{'总面积':<20}{total_area:>12.2f}")

# ===== 证明多态的价值 =====
print("\n--- 加一种新形状，上面的循环代码一个字都不用改 ---")


class Trapezoid(Shape):
    """梯形 —— 这是运行后新加的类型。"""
    def __init__(self, top, bottom, height):
        super().__init__("梯形")
        self.top = top
        self.bottom = bottom
        self.height = height

    def area(self):
        return (self.top + self.bottom) * self.height / 2


shapes.append(Trapezoid(3, 5, 4))
print(f"加入梯形后，总数：{len(shapes)}")
for shape in shapes[-1:]:
    print(f"{shape} 面积 {shape.area():.2f}")

# ===== 调用未重写的方法会怎样 =====
print("\n--- 父类未重写方法的行为 ---")
generic = Shape("神秘形状")
print(f"area() 返回：{generic.area()}")
print("这就是多态：同一段调用，不同的对象有不同的表现")

Python# hierarchy.py —— 三层继承

class Person:
    """人。"""
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def greet(self):
        """打招呼。"""
        print(f"你好，我是{self.name}")

    def birthday(self):
        """过生日。"""
        self.age += 1

    def __str__(self):
        return f"{self.name}，{self.age} 岁"


class Employee(Person):
    """员工 —— 继承 Person。"""
    def __init__(self, name, age, company, salary):
        # 先让父类设置共有的 name 和 age
        super().__init__(name, age)
        self.company = company
        self.salary = salary            # 年薪

    def get_monthly(self):
        """返回月薪。"""
        return self.salary / 12

    def get_daily(self):
        """返回日薪（按 22 个工作日算）。"""
        return self.salary / 12 / 22

    def raise_salary(self, amount):
        """加薪。"""
        self.salary += amount
        return self.salary

    def __str__(self):
        # 用 super() 复用父类的字符串，再追加自己的信息
        return f"{super().__str__()}，在 {self.company} 工作，年薪 {self.salary} 元"


class Manager(Employee):
    """经理 —— 继承 Employee。"""
    def __init__(self, name, age, company, salary, team_size):
        # 调用祖父类的初始化（super() 会自动找对正确的父类）
        super().__init__(name, age, company, salary)
        self.team_size = team_size

    def __str__(self):
        # 复用 Employee 的 __str__，再追加团队信息
        return f"{super().__str__()}，管理 {self.team_size} 人团队"

    def performance_review(self):
        """经理独有的方法。"""
        return f"{self.name} 对 {self.team_size} 名下属进行了考核"

class Book:
    """一本书。"""
    def __init__(self, title, author, available=True):
        self.title = title
        self.author = author
        self.available = available

    def borrow(self):
        """借出图书，成功返回 True。"""
        if not self.available:
            return False
        self.available = False
        return True

    def return_book(self):
        """归还图书。"""
        self.available = True
        return True

    def status(self):
        """返回可借状态文字。"""
        return "在架" if self.available else "已借出"

    def __str__(self):
        return f"《{self.title}》/ {self.author}"


class Library:
    """图书馆。"""
    def __init__(self, name="市图书馆"):
        self.name = name
        self.books = []

    def add_book(self, book):
        """添加一本书。"""
        self.books.append(book)
        print(f"已入库：{book}")

    def find_book(self, title):
        """按书名查找，找到返回 Book 对象，找不到返回 None。"""
        for book in self.books:
            if book.title == title:
                return book
        return None

    def borrow_book(self, title):
        """按书名借书。"""
        book = self.find_book(title)
        if book is None:
            print(f"未找到《{title}》")
            return False
        if book.borrow():
            print(f"成功借出《{book.title}》")
            return True
        print(f"《{book.title}》已被借出，无法再次借出")
        return False

    def return_book(self, title):
        """按书名还书。"""
        book = self.find_book(title)
        if book is None:
            print(f"未找到《{title}》")
            return False
        book.return_book()
        print(f"已归还《{book.title}》")
        return True

    def list_available(self):
        """列出所有在架图书。"""
        available = [book for book in self.books if book.available]
        print(f"--- {self.name}在架图书（{len(available)} 本）---")
        for book in available:
            print(f"  {book} / {book.author}")
        if not available:
            print("  （暂无在架图书）")
        return available

    def statistics(self):
        """统计图书状态。"""
        total = len(self.books)
        on_shelf = sum(1 for book in self.books if book.available)
        return {
            "馆名": self.name,
            "总数": total,
            "在架": on_shelf,
            "借出": total - on_shelf,
        }

class Owner:
    """账户持有人。"""
    def __init__(self, name, email):
        self.name = name
        self.email = email

    def __str__(self):
        return f"{self.name} <{self.email}>"


class BankAccount:
    """一个银行账户。"""
    def __init__(self, account_number, owner, balance=0.0):
        self.account_number = account_number
        self.owner = owner          # 组合：持有一个 Owner 对象
        self.balance = balance
        self.history = []

    def deposit(self, amount):
        """存款，金额必须大于 0。"""
        if amount <= 0:
            raise ValueError("存款金额必须大于 0")
        self.balance += amount
        self.history.append(f"存款 {amount:.2f} 元")
        return self.balance

    def withdraw(self, amount):
        """取款，余额必须充足。"""
        if amount <= 0:
            raise ValueError("取款金额必须大于 0")
        if amount > self.balance:
            raise ValueError(f"余额不足，当前余额 {self.balance:.2f} 元")
        self.balance -= amount
        self.history.append(f"取款 {amount:.2f} 元")
        return self.balance

    def transfer_to(self, other, amount):
        """转账到另一个账户。"""
        if other is self:
            raise ValueError("不能转账到自己")
        self.withdraw(amount)      # 余额不足时会自动报错
        other.deposit(amount)
        self.history.append(f"转账 {amount:.2f} 元给 {other.account_number}")
        other.history.append(f"收到来自 {self.account_number} 的 {amount:.2f} 元")
        return True

    def statement(self):
        """输出对账单。"""
        lines = [
            "=" * 46,
            f"账户号码：{self.account_number}",
            f"户主：  {self.owner}",
            f"余额：    {self.balance:.2f} 元",
            "-" * 46,
            "交易记录：",
        ]
        if not self.history:
            lines.append("  （暂无交易）")
        for i, record in enumerate(self.history, 1):
            lines.append(f"  {i}. {record}")
        lines.append("=" * 46)
        return "\n".join(lines)

    def __str__(self):
        return f"账户 {self.account_number}（{self.owner.name}）余额 {self.balance:.2f} 元"


class Bank:
    """银行 —— 组合多个账户。"""
    def __init__(self, name):
        self.name = name
        self.accounts = []
        self._next_number = 1001      # 自动递增的账户号码

    def open_account(self, owner, initial=0.0):
        """开户，返回新建的账户对象。"""
        account = BankAccount(self._next_number, owner, initial)
        self._next_number += 1       # 下一个账户号码
        self.accounts.append(account)
        if initial > 0:
            account.history.append(f"开户存入 {initial:.2f} 元")
        print(f"{self.name} 已为 {owner.name} 开户，账号 {account.account_number}")
        return account

    def find_account(self, owner_name):
        """按户主名字找账户，找到返回对象，找不到返回 None。"""
        for account in self.accounts:
            if account.owner.name == owner_name:
                return account
        return None

    def total_deposit(self):
        """所有账户余额之和。"""
        return sum(account.balance for account in self.accounts)

    def total_customers(self):
        """客户数量。"""
        return len(self.accounts)

    def richest_account(self):
        """余额最多的账户，没有账户时返回 None。"""
        if not self.accounts:
            return None
        richest = self.accounts[0]
        for account in self.accounts[1:]:
            if account.balance > richest.balance:
                richest = account
        return richest

    def print_all(self):
        """列出所有账户。"""
        print(f"--- {self.name} 全部账户（{len(self.accounts)} 个）---")
        for account in self.accounts:
            print(f"  {account}")

Python# lottery.py —— 分档抽奖，已中奖的人不再参与下一档
import random

students = [f"学员{i:02d}" for i in range(1, 31)]
random.seed(2026)

first = random.sample(students, 1)
rest = [s for s in students if s not in first]        # 排除已中奖

second = random.sample(rest, 3)
rest = [s for s in rest if s not in second]

third = random.sample(rest, 6)

print("一等奖", first)
print("二等奖", second)
print("三等奖", third)

输出：

示例一等奖 ['学员04']
二等奖 ['学员12', '学员18', '学员22']
三等奖 ['学员05', '学员09', '学员24', '学员28', '学员21', '学员16']

为什么不用 random.choices：choices 是有放回抽样，同一个人可能既中一等奖又中二等奖。凡涉及「名额」，一律 sample。

另一种更短的写法是先抽 10 个再切片，结果顺序不同但同样正确：

Python# （片段）一次抽满再分档
winners = random.sample(students, 10)
first, second, third = winners[:1], winners[1:4], winners[4:]

class Car:
    def __init__(self, make, model, year):
        self.make = make
        self.model = model
        self.year = year

    def describe(self):
        return f"{self.year} {self.make} {self.model}"


class ElectricCar(Car):
    def __init__(self, make, model, year, battery=75):
        super().__init__(make, model, year)
        self.battery = battery

    def describe(self):
        return super().describe() + f"（电池 {self.battery} kWh）"


if __name__ == "__main__":
    print(Car("比亚迪", "汉", 2024).describe())
    print(ElectricCar("特斯拉", "Model 3", 2024).describe())

Python# （片段）my_cars.py
import car
from car import ElectricCar

print(car.Car("大众", "高尔夫", 2020).describe())          # ① 模块名.类名
print(ElectricCar("蔚来", "ET5", 2023).describe())         # ② 直接用类名

③ 的答案：from car import * 会把 car.py 里所有不以 _ 开头的名字一次性倒进当前命名空间，问题有两个：

  • 看不出来源：读代码时看到 Car，不知道是本文件定义的还是从别处导进来的；
  • 会静默覆盖：如果 car.py 里恰好有个变量叫 random，import * 之后本文件的 random 就被替换掉了，后面 random.randint 直接报 AttributeError。显式导入只会覆盖你写出来的名字，import * 覆盖的是「你不知道的全部」。
