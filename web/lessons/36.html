示例 1：定义与调用函数

Python# function_basics.py —— 函数的定义与调用

def greet_user():
    """一个最简单的函数：打招呼。"""
    print("你好，欢迎使用 Python！")


greet_user()      # 调用：写函数名加括号
greet_user()      # 可以反复调用，每次都执行一遍

# 带参数的函数
def greet_name(name):
    """向指定的人打招呼。"""
    print(f"你好，{name}！")


greet_name("小明")
greet_name("小红")

# 带默认值的参数
def greet_with_title(name, title="同学"):
    """带称呼的问候，title 可以不填。"""
    print(f"你好，{title} {name}！")


greet_with_title("小明")              # 你好，同学 小明！
greet_with_title("李老师", "老师")     # 你好，老师 李老师！

# 多参数函数
def rectangle_area(width, height):
    """计算矩形面积。"""
    return width * height


print(f"面积：{rectangle_area(5, 3)}")        # 面积：15

示例 2：return 与 None

Python# return_demo.py —— return 的三个要点

# 有 return 和没 return 的差别
def with_return():
    return 42


def without_return():
    print("我执行了，但我什么都不返回")


print("with_return() 返回：", with_return())      # 42
print("without_return() 返回：", without_return())  # None

# return 之后立刻结束函数
def early_exit():
    return "这是第一个 return 的结果"
    print("这行永远不会执行")            # 死代码
    return "这行也永远不会执行"


print(early_exit())        # 这是第一个 return 的结果

# 一个函数可以有多个 return，但只会走一个
def classify_number(n):
    """判断正数、负数或零。"""
    if n > 0:
        return "正数"
    elif n < 0:
        return "负数"
    return "零"      # 注意：这行缩进和 if 平齐，不在 elif 里面


for value in [5, -3, 0]:
    print(f"{value} 是 {classify_number(value)}")

# 返回多个值：其实返回的是一个元组
def min_and_max(numbers):
    """返回最小值和最大值。"""
    return min(numbers), max(numbers)


smallest, largest = min_and_max([3, 9, 2, 7])
print(f"最小 {smallest}，最大 {largest}")

# 返回字典：适合"多个名字的东西"
def get_user():
    """返回用户信息的字典。"""
    return {"name": "小明", "age": 18}


user = get_user()
print(f"{user['name']}今年 {user['age']} 岁")

# 无 return 的隐式 None（容易踩坑）
def side_effect_only(numbers):
    """只打印不返回。"""
    for n in numbers:
        print(n)


result = side_effect_only([1, 2, 3])
print("返回值是：", result)      # None —— 所以别拿它参与运算

示例 3：三种参数写法

Python# parameters.py —— 位置参数、关键字参数、默认参数

# 1）位置参数：按顺序对应
def describe_pet(animal_type, pet_name):
    """位置参数：调用时的顺序必须和定义时一致。"""
    return f"这是一只{animal_type}，名字叫{pet_name}"


print(describe_pet("狗", "旺财"))       # 狗 / 旺财
print(describe_pet("猫", "咪咪"))       # 猫 / 咪咪
# print(describe_pet("旺财", "狗"))    # 参数顺序错了，含义就全错了

# 2）关键字参数：按名字对应，顺序可以变
print(describe_pet(pet_name="旺财", animal_type="狗"))   # 顺序变了但结果正确

# 3）默认参数：可以不填，用默认值
def make_pizza(size=12, with_cheese=True):
    """默认参数：有默认值的参数必须写在最后。"""
    info = f"{size} 英寸披萨"
    if with_cheese:
        info += "，加芝士"
    else:
        info += "，不加芝士"
    return info


print(make_pizza())                          # 12 英寸披萨，加芝士
print(make_pizza(16))                         # 16 英寸披萨，加芝士
print(make_pizza(16, False))                  # 16 英寸披萨，不加芝士
print(make_pizza(with_cheese=False))          # 12 英寸披萨，不加芝士（跳过第一个）

# 4）可变参数 *args：收集所有的位置参数
def sum_all(*numbers):
    """接收任意多个位置参数。"""
    return sum(numbers)


print(sum_all(1, 2, 3))           # 6
print(sum_all(1, 2, 3, 4, 5))     # 15
print(sum_all())                  # 0

# 5）可变参数 **kwargs：收集所有的关键字参数
def build_profile(**info):
    """接收任意多个关键字参数，打包成字典。"""
    return info


print(build_profile(name="小明", age=18, city="杭州"))
print(build_profile())             # {}

# 6）四种参数混合使用（顺序有强制规定）
def full_demo(a, b=2, *args, **kwargs):
    """必填参数 -> 默认参数 -> *args -> **kwargs，顺序不能乱。"""
    print(f"a={a}")
    print(f"b={b}")
    print(f"args={args}")
    print(f"kwargs={kwargs}")


full_demo(1, 3, 5, 7, name="x", age=1)
# 错误示范：必填参数放在默认参数后面会报 SyntaxError
# def wrong(a=1, b):    # SyntaxError
#     pass

示例 4：★ 参数传递的副本语义

Python# copy_semantics.py —— 函数里修改参数，到底影响不影响原数据

# ===== 情况1：传整数 -> 不影响原变量 =====
def add_one(n):
    n = n + 1         # 只是让参数 n 指向了一个新整数
    print(f"  函数内 n = {n}")


age = 10
add_one(age)
print(f"  函数外 age = {age}\n")     # 10  没变

# ===== 情况2：传字符串 -> 不影响 =====
def shout(word):
    word = word.upper()   # 返回了新字符串，参数指向了它
    print(f"  函数内 word = {word}")


name = "tom"
shout(name)
print(f"  函数外 name = {name}\n")   # tom  没变

# ===== 情况3：传列表 -> 会影响原列表！=====
def add_item(items):
    items.append("新元素")     # 直接改的是同一份列表
    print(f"  函数内 items = {items}")


fruits = ["苹果", "香蕉"]
add_item(fruits)
print(f"  函数外 fruits = {fruits}\n")   # ['苹果', '香蕉', '新元素']  变了！

# ===== 情况4：传字典 -> 同样会影响 =====
def set_name(profile):
    profile["name"] = "李四"    # 改的是同一个字典


user = {"name": "张三", "age": 20}
set_name(user)
print(f"  函数外 user = {user}\n")       # name 变成了李四

# ===== 三种修复方案 =====
fruits = ["苹果", "香蕉"]

# 方案1：函数内自己造副本（推荐）
def safe_append(items):
    items = items[:]      # 先复制一份，只动副本
    items.append("新元素")
    return items


result = safe_append(fruits)
print(f"方案1：原列表 {fruits}，新列表 {result}\n")

# 方案2：调用时传副本
def append_direct(items):
    items.append("新元素")


result = fruits[:]        # 调用方给副本
append_direct(result)
print(f"方案2：原列表 {fruits}，新列表 {result}\n")

# 方案3：用 + 产生新列表，天然不改原数据
def append_pure(items):
    return items + ["新元素"]


result = append_pure(fruits)
print(f"方案3：原列表 {fruits}，新列表 {result}")

示例 5：作用域与 global

Python# scope_demo.py —— 全局变量与局部变量

name = "全局变量"


def show_variables():
    local_name = "局部变量"
    print("  函数内可以看全局：", name)
    print("  函数内可以看局部：", local_name)


show_variables()
# print(local_name)      # NameError: 局部变量出了函数就不存在了

# ===== 修改全局变量的三种方式 =====
count = 0

# 方式1：直接赋值 -> 只是新建了一个局部变量，全局没变
def way_one():
    count = 100
    print(f"  方式1 函数内 count = {count}")


way_one()
print("方式1 之后全局 count =", count, "\n")    # 0  没变

# 方式2：global 声明 -> 真能改，但官方不推荐
def way_two():
    global count
    count = 100
    print(f"  方式2 函数内 count = {count}")


way_two()
print("方式2 之后全局 count =", count, "\n")    # 100  改了

# 方式3（最佳）：把数据传进去，把结果返回出来
def add_to_count(amount, current):
    """不碰全局状态，纯靠参数和返回值。"""
    return current + amount


total = 100
total = add_to_count(50, total)      # 显式接住返回值
print("方式3 total =", total)         # 150

# 结论：能用参数和返回值解决的，绝不用 global
# 因为 global 让函数的"隐藏副作用"变多，代码会变得难追踪

示例 6：函数是一等公民

Python# first_class.py —— 把函数当数据用

def add(a, b):
    return a + b


def subtract(a, b):
    return a - b


# 1）赋给变量
operation = add
print("operation(3, 4) =", operation(3, 4))        # 7

# 2）放进列表，按需要选
operations = [add, subtract]
print("逐个执行：", [op(10, 3) for op in operations])   # [13, 7]

# 3）当参数传给别的函数（sorted 的 key 就是这么用的）
scores = {"小明": 95, "小红": 88, "小刚": 72}
by_score = sorted(scores.items(), key=lambda item: item[1], reverse=True)
print("按分数排序：", by_score)

# 4）当返回值返回（"工厂函数"）
# 注意：sorted 的 key 参数要求函数只接收"一个"参数，
# 而比较函数天然要接收"两个"，所以要用 cmp_to_key 包一层适配。
import functools


def get_comparator(reverse=False):
    """返回一个"两两比较"函数：接收两个值，返回负/零/正。"""
    def compare(a, b):
        if reverse:
            return b - a
        return a - b

    return compare


asc = get_comparator()
desc = get_comparator(reverse=True)
print("升序：", sorted([3, 7, 1], key=functools.cmp_to_key(asc)))
print("降序：", sorted([3, 7, 1], key=functools.cmp_to_key(desc)))

# 5）lambda：一次性小函数，写法极简
square = lambda x: x ** 2
print("square(5) =", square(5))       # 25

# lambda 的本质就是个没有名字的普通函数
print("lambda 和 def 等价：", (lambda x: x ** 2)(5) == square(5))   # True

示例 7：默认参数的可变对象陷阱

Python# mutable_default.py —— 默认参数不要用可变对象（重要坑）

# ===== 错误写法：默认参数用了列表 =====
def add_task(task, todo=[]):      # 注意这个 []
    """危险！默认列表只创建一次，所有调用共享它。"""
    todo.append(task)
    return todo


print(add_task("买菜"))     # ['买菜']
print(add_task("做饭"))     # ['买菜', '做饭']  <- 上次的"买菜"还在！
print(add_task("睡觉"))     # ['买菜', '做饭', '睡觉']

# 原因：默认参数在函数**定义时**只求值一次，
# 之后每次调用都用那同一份列表对象。

# ===== 正确写法1：用 None 作为默认值 =====
def add_task(task, todo=None):
    """默认值为 None，每次调用时再创建新列表。"""
    if todo is None:
        todo = []# 每次都新建一个，互不影响
    todo.append(task)
    return todo


print()
print(add_task("买菜"))     # ['买菜']
print(add_task("做饭"))     # ['做饭']  干净了

# ===== 正确写法2：用不可变类型 =====
def add_tag(tag, tags=()):        # 元组不可变，安全
    return tags + (tag,)


print()
print(add_tag("a"))     # ('a',)
print(add_tag("b"))     # ('b',)  独立

# ===== 默认参数还有另一个用途：让函数"记住"状态 =====
# 这是少数"用可变默认值"是合理的场景
def make_counter(start=0):
    """返回一个能计数的函数，记住上次的结果。"""
    return start


# 更实用的闭包写法
def make_adder(n):
    """返回一个"加了 n 的函数"。"""
    def adder(x):
        return x + n

    return adder


add_5 = make_adder(5)
print()
print("add_5(10) =", add_5(10))      # 15
add_10 = make_adder(10)
print("add_10(10) =", add_10(10))    # 20  两者互不影响

示例 8：__name__ 的两种身份

把下面这段存成 module_demo.py 直接运行，会看到 __name__ 是 "__main__"；如果被别的文件 import module_demo，__name__ 就变成 "module_demo"，if 里面的三行不会执行。

Python# module_demo.py —— 直接运行时才跑自测
def area(w, h):
    """矩形面积。"""
    return w * h


def average(nums):
    """平均值；空列表返回 None，而不是 ZeroDivisionError。"""
    return sum(nums) / len(nums) if nums else None


print("__name__ =", __name__)          # 直接运行 → __main__

if __name__ == "__main__":
    print("面积  ", area(3, 4))
    print("平均  ", average([80, 90, 100]))
    print("空列表", average([]))

运行结果：

示例__name__ = __main__
面积   12
平均   90.0
空列表 None

把它当模块用的时候，调用方只要写：

Python# （片段）另一个文件里
import module_demo

print(module_demo.area(5, 6))          # 30
print(module_demo.average([]))         # None
# 上面那三行 print 不会再出现，__name__ 是 "module_demo"

这个模式的实际好处是：同一个文件既能当脚本跑，又能当库被导入，而且「跑起来会打印东西」和「被导入时安静」这两件事不用维护两份代码。单元 10 的 pytest 也会用同一种思路——测试文件被 pytest 收集，而不是被直接运行。
