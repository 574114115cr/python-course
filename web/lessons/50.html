2.8 JSON：让数据能被记住

程序里的字典和列表不能直接存进文件（它们不是纯文本）。json 模块负责翻译。

Pythonimport json

data = {"name": "小明", "age": 18, "skills": ["Python", "SQL"]}

# 字典/列表 -> JSON 字符串
text = json.dumps(data)
print(text)     # {"name": "\u5c0f\u660e", "age": 18, ...}
# 中文乱码了，加上 ensure_ascii=False
print(json.dumps(data, ensure_ascii=False, indent=2))

# JSON 字符串 -> 字典
back = json.loads('{"name": "小明", "age": 18}')
print(back["name"])       # 小明

JSON 与 Python 的对应关系（这张表要背下来）：

JSON Python
object（{}） dict
array（[]） list
string（"..."） str
number（没有引号） int / float
true / false True / False
null None

注意一个坑：JSON 的 true/false 和 Python 的 True/False 大小写不同。Python 的 json 模块能自动转换，但其他语言的 JSON 解析库往往要求严格的 JSON 语法（用小写 true）。所以自己手写 JSON 时要用小写。

读写 JSON 文件：

Python# 写
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)

# 读
with open("data.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)

encoding="utf-8" 必须写！ 不写的话，在 Windows 上读写中文可能报 UnicodeEncodeError 或 UnicodeDecodeError，或者写成乱码。这是中文用户最常踩的文件坑。

2.9 CSV：读写表格

CSV（逗号分隔值）是最常见的表格格式，可以用 Excel 打开：

Pythonimport csv

# 写
with open("students.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["姓名", "年龄", "城市"])        # 写表头
    writer.writerow(["小明", 18, "杭州"])           # 写一行
    writer.writerow(["小红", 22, "北京"])

# 读
with open("students.csv", "r", newline="", encoding="utf-8") as f:
    reader = csv.reader(f)
    for row in reader:
        print(row)          # ['姓名', '年龄', '城市']

两个必写的参数： - newline=""：没有它，Windows 上会写出空行（Python 会在 \r\n 之间再加一个 \n）。这是标准写法，记住就行。 - encoding="utf-8"：同上，中文必需。

示例 1：with open 读写

Python# file_basics.py —— 文件读写基础（用 with 语句）

# ===== 写文件 =====
filename = "示例文件.txt"

# "w" 模式：文件不存在就新建，存在就清空
with open(filename, "w", encoding="utf-8") as f:
    f.write("你好，Python！\n")
    f.write("这是第二行。\n")
    f.write("这是第三行。\n")

print(f"已写入文件：{filename}")

# 注意：write() 不会自动换行，必须自己写 \n

# ===== 读文件（全部读进来）=====
with open(filename, "r", encoding="utf-8") as f:
    content = f.read()          # 一次性读入整个文件

print("--- 全部内容 ---")
print(content)
print(f"文件里有 {len(content)} 个字符")

# ===== 追加内容 =====
with open(filename, "a", encoding="utf-8") as f:
    f.write("这是追加的第四行。\n")

with open(filename, "r", encoding="utf-8") as f:
    print("--- 追加后的内容 ---")
    print(f.read())

# ===== with 的价值：出块后文件自动关闭 =====
with open(filename, "r", encoding="utf-8") as f:
    first_line = f.readline()

#出了 with 块，f 已经关闭，不能再用了
# print(f.read())     # ValueError: I/O operation on closed file
# 但我们保存下来的变量还在
print("第一行：", repr(first_line))

示例 2：三种模式的区别

Python# modes_demo.py —— r / w / a 三种模式的区别

filename = "模式测试.txt"

# 先创建一个有内容的文件
with open(filename, "w", encoding="utf-8") as f:
    f.write("第一行\n第二行\n第三行\n")

def show(title):
    """读取并显示文件内容。"""
    try:
        with open(filename, "r", encoding="utf-8") as f:
            lines = f.readlines()
        print(f"  {title}: {len(lines)} 行 -> {[l.strip() for l in lines]}")
    except FileNotFoundError:
        print(f"  {title}: 文件不存在")


show("初始状态")

# "a" 追加：保留原有内容
with open(filename, "a", encoding="utf-8") as f:
    f.write("追加的行\n")
show("a 追加后")

# "w" 写入：清空原内容！
with open(filename, "w", encoding="utf-8") as f:
    f.write("全新的内容\n")
show("w 覆盖后")            # 只剩1 行，前面的都没了！

# "r" 读取不存在的文件 -> FileNotFoundError
try:
    with open("根本不存在.txt", "r", encoding="utf-8") as f:
        f.read()
except FileNotFoundError as error:
    print("r 读取不存在的文件：", error)

# "w" 和 "a" 都会在文件不存在时自动创建
with open("新建的文件.txt", "w", encoding="utf-8") as f:
    f.write("用 w 创建的新文件")
with open("新建的文件.txt", "a", encoding="utf-8") as f:
    f.write("\n用 a 追加的内容")

print("新建的文件是否存在于当前目录：是（w 和 a 都能自动创建）")

示例 3：四种读取方式

Python# read_ways.py —— 四种读取文件的方式

filename = "读取测试.txt"
with open(filename, "w", encoding="utf-8") as f:
    f.write("小明\n小红\n小刚\n")# 注意最后一行没有 \n

# ① read()：全部读成一个字符串
with open(filename, "r", encoding="utf-8") as f:
    content = f.read()
print("① read():")
print(f"   类型={type(content).__name__} 内容={content!r}")

# ② readline()：一次读一行
print("\n② readline():")
with open(filename, "r", encoding="utf-8") as f:
    print(f"   第1次={f.readline()!r}")
    print(f"   第2次={f.readline()!r}")
    print(f"   第3次={f.readline()!r}")     # 最后一行的 \n 就是 \n

# ③ readlines()：读成列表
print("\n③ readlines():")
with open(filename, "r", encoding="utf-8") as f:
    lines = f.readlines()
print(f"   类型={type(lines).__name__} 内容={lines}")

# ④ for 循环逐行读（推荐方式，能处理超大文件）
print("\n④ for 逐行读:")
with open(filename, "r", encoding="utf-8") as f:
    for line in f:
        print(f"   读到: {line.strip()}")   # strip() 去掉首尾空白和换行

# readline() 的换行符细节
print("\n--- 换行符的细节 ---")
with open(filename, "r", encoding="utf-8") as f:
    for line in f:
        print(f"   {line!r}  -> strip 后 {line.strip()!r}")

# 清理
import os
for name in [filename]:
    if os.path.exists(name):
        os.remove(name)
        print(f"已清理临时文件：{name}")

示例 4：异常处理

Python# exceptions_demo.py —— try/except/else/finally

# ===== 1）最基础的 try/except =====
print("=== 1. 基础异常处理 ===")
try:
    number = int("abc")
except ValueError:
    print("捕获到 ValueError：'abc' 不是有效整数")

# ===== 2）多种异常分别处理 =====
print("\n=== 2. 多种异常 ===")
def divide(a, b):
    try:
        return float(a) / float(b)
    except ValueError:
        return "错误：必须输入数字"
    except ZeroDivisionError:
        return "错误：除数不能为 0"


print(divide(10, 2))# 5.0
print(divide("x", 2))      # 错误：必须输入数字
print(divide(10, 0))       # 错误：除数不能为 0

# ===== 3）捕获异常对象，读取错误信息 =====
print("\n=== 3. 用 as 获取异常信息 ===")
try:
    int("abc")
except ValueError as error:
    print(f"异常信息：{error}")

try:
    with open("不存在的文件.txt", "r", encoding="utf-8") as f:
        f.read()
except FileNotFoundError as error:
    print(f"文件错误信息：{error}")

# ===== 4）else：没出错才执行 =====
print("\n=== 4. else 的作用 ===")
for text in ["100", "abc"]:
    try:
        value = int(text)
    except ValueError:
        print(f"  {text!r} -> 转换失败")
    else:
        # 只有 try 块完全成功才会走这里
        print(f"  {text!r} -> 转换成功：{value * 2}")

# ===== 5）finally：无论如何都执行 =====
print("\n=== 5. finally 的作用 ===")


def process(raw):
    """演示 finally：无论成功失败都会打印收尾日志。"""
    try:
        index = int(raw)
        return ["甲", "乙", "丙"][index]     # 成功：返回对应元素
    except ValueError:
        return "输入无效"
    except IndexError:
        return "下标超出范围"
    finally:
        # 无论成功失败都会执行，常用于释放资源、记录日志
        print(f"  finally: 正在处理 {raw!r}")


print("成功的情况：", process("1"))# 乙
print("输入无效：", process("abc"))
print("下标越界：", process("99"))   # 返回友好提示而不是崩溃

# ===== 6）一个常见错误：裸 except =====
print("\n=== 6. 为什么不要用裸 except ===")
# INTENTIONALLY-BROKEN 演示：裸 except 会捕获一切，包括 Ctrl+C
# try:
#     while True:
#         print("按 Ctrl+C 应该能停下来")
# except:
#     pass
# 现象：Ctrl+C 完全无效，必须用任务管理器强杀。
# 原因：KeyboardInterrupt 也是一个异常类，被裸 except 吞掉了。

# 正确做法：明确指定异常类型
try:
    value = int("abc")
except ValueError:
    print("正确做法：只捕获 ValueError，Ctrl+C 仍然有效")

# ===== 7）主动抛出异常 =====
print("\n=== 7）主动抛出异常 ===")
def withdraw(balance, amount):
    if amount <= 0:
        raise ValueError("取款金额必须大于 0")
    if amount > balance:
        raise ValueError(f"余额不足，只有 {balance} 元")
    return balance - amount


print("正常取款：", withdraw(1000, 200))
try:
    withdraw(1000, 5000)
except ValueError as error:
    print("取款失败：", error)

示例 5：异常处理的实用模式

Python# patterns.py —— 三个最常用的异常处理模式

# ===== 模式1：输入重试循环 =====
print("=== 模式1：输入重试 ===")


def read_number(prompt, max_attempts=3):
    """读取一个有效数字，最多尝试 max_attempts 次。"""
    for attempt in range(1, max_attempts + 1):
        raw = input(f"{prompt}（第 {attempt}/{max_attempts} 次）：")
        try:
            return float(raw)
        except ValueError:
            print(f"  '{raw}' 不是有效数字")
    raise ValueError("尝试次数用尽，仍然没有有效输入")


# 演示：不需要真的输入，用固定值模拟
def demo_retry():
    inputs = ["abc", "12.5"]
    for raw in inputs:
        try:
            value = float(raw)
            print(f"  成功：{value}")
            return value
        except ValueError:
            print(f"  '{raw}' 无效，重试")


demo_retry()

# ===== 模式2：安全地执行可能失败的操作 =====
print("\n=== 模式2：安全操作 ===")


def safe_divide(a, b):
    """安全除法，任何问题都返回一个可读的结果。"""
    try:
        return float(a) / float(b)
    except (ValueError, ZeroDivisionError) as error:
        return f"无法计算：{error}"


print(safe_divide(10, 2))
print(safe_divide(10, 0))
# 注意 (ValueError, ZeroDivisionError) 用括号包起来，表示"任意一种"

# ===== 模式3：文件读取的标准健壮写法 =====
print("\n=== 模式3：健壮的文件读取 ===")


def read_file_lines(filename, encoding="utf-8"):
    """安全读取文件，返回行列表，出错时返回 None。"""
    try:
        with open(filename, "r", encoding=encoding) as f:
            return [line.rstrip("\n") for line in f]
    except FileNotFoundError:
        print(f"  文件 {filename} 不存在")
        return None
    except UnicodeDecodeError:
        print(f"  文件 {filename} 编码不匹配，试试换个 encoding")
        return None


# 先创建一个文件测试
with open("健壮性测试.txt", "w", encoding="utf-8") as f:
    f.write("第一行\n第二行\n第三行")

print("读取结果：", read_file_lines("健壮性测试.txt"))
print("读取不存在的文件：", read_file_lines("没有这个文件.txt"))

import os
os.remove("健壮性测试.txt")

示例 6：JSON

Python# json_demo.py —— 用 JSON 保存和读取结构化数据

import json

# 准备一个包含各种类型的数据
data = {
    "name": "小明",
    "age": 18,
    "score": 92.5,
    "is_student": True,
    "hobbies": ["篮球", "编程", "音乐"],
    "address": {"city": "杭州", "zip": "310000"},
    "notes": None,
}

# ===== Python 对象 -> JSON 字符串 =====
# 默认中文会变成 \uXXXX，可读性差
print("--- 默认 dumps ---")
print(json.dumps(data)[:80])

# ensure_ascii=False 让中文正常显示
print("\n--- ensure_ascii=False ---")
text = json.dumps(data, ensure_ascii=False)
print(text)

# indent=2 格式化输出，人和机器都好看
print("\n--- indent=2（会保存到文件的样子）---")
pretty = json.dumps(data, ensure_ascii=False, indent=2)
print(pretty)

# ===== JSON 字符串 -> Python 对象 =====
# 反序列化，还原成字典
back = json.loads(text)
print("\n--- 解析回来 ---")
print(f"类型：{type(back).__name__}")
print(f"姓名：{back['name']}")
print(f"爱好：{back['hobbies']}")
print(f"城市：{back['address']['city']}")
print(f"备注：{back['notes']}")

# ===== 读写 JSON 文件 =====
filename = "data.json"

# 写
with open(filename, "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)
print(f"\n已写入 {filename}")

# 读
with open(filename, "r", encoding="utf-8") as f:
    loaded = json.load(f)

print(f"从文件读回来的姓名：{loaded['name']}")
print(f"类型还是 dict吗：{type(loaded).__name__}")
print(f"爱好列表：{loaded['hobbies']}")

# 验证往返一致

# ===== JSON 语法要点 =====

# 3）JSON 不支持注释
# 4）JSON 不支持元组和集合，元组会变成列表

示例 7：CSV

Python# csv_demo.py —— 读写 CSV 表格

import csv

filename = "students.csv"

# ===== 写 CSV =====
# newline="" 是必须写的，encoding="utf-8" 保证中文不乱码
with open(filename, "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["姓名", "年龄", "城市", "成绩"])     # 表头
    writer.writerow(["小明", 18, "杭州", 92.5])
    writer.writerow(["小红", 22, "北京", 88.0])
    writer.writerow(["小刚", 20, "上海", 76.5])

print(f"已写入 {filename}")

# ===== 读 CSV =====
with open(filename, "r", newline="", encoding="utf-8") as f:
    reader = csv.reader(f)
    for row in reader:
        print(f"  {row}")

print("\n--- 读成字典列表（更方便处理）---")
with open(filename, "r", newline="", encoding="utf-8") as f:
    # DictReader 把每行和表头对应起来
    reader = csv.DictReader(f)
    students = list(reader)

for student in students:
    print(f"  {student['姓名']}：{student['年龄']} 岁，"
          f"在 {student['城市']}，成绩 {student['成绩']}")

# 用字典形式处理数据非常顺手
print("\n--- 基于字典做统计 ---")
total_score = sum(float(s["成绩"]) for s in students)
print(f"总成绩：{total_score}")
print(f"平均分：{total_score / len(students):.2f}")
best = max(students, key=lambda s: float(s["成绩"]))
print(f"最高分：{best['姓名']}，{best['成绩']}")

# ===== 追加数据 =====
with open(filename, "a", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["小丽", 19, "广州", 95.0])

with open(filename, "r", newline="", encoding="utf-8") as f:
    print(f"\n追加后的行数：{sum(1 for _ in f)}")

import os
os.remove(filename)
