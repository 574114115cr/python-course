from calculator import add, subtract, multiply, divide

import pytest


# ===== 正常用例 =====
def test_add():
    """测试加法。"""
    assert add(2, 3) == 5
    assert add(0, 0) == 0
    assert add(-1, 1) == 0


def test_subtract():
    """测试减法。"""
    assert subtract(5, 3) == 2
    assert subtract(3, 5) == -2
    assert subtract(10, 10) == 0


def test_multiply():
    """测试乘法。"""
    assert multiply(3, 4) == 12
    assert multiply(0, 100) == 0
    assert multiply(-2, 3) == -6


def test_divide():
    """测试除法。"""
    assert divide(10, 2) == 5
    assert divide(9, 3) == 3
    assert divide(-10, 2) == -5


# ===== 边界用例 =====
def test_divide_by_one():
    """除以 1 的边界情况。"""
    assert divide(7, 1) == 7


def test_divide_result_is_float():
    """除法结果应该是浮点数。"""
    result = divide(10, 2)
    assert result == 5.0
    assert isinstance(result, float)


# ===== 异常用例：除零 =====
def test_divide_by_zero_raises():
    """除以0 应该抛出 ZeroDivisionError。"""
    with pytest.raises(ZeroDivisionError):
        divide(10, 0)

示例 3：断言消息与常见断言

Python# test_assertions.py —— 各种 assert 写法

import pytest


def test_equal():
    """相等断言。"""
    assert 2 + 2 == 4


def test_not_equal():
    """不等断言。"""
    assert 2 + 2 != 5


def test_with_message():
    """带消息的断言，失败时会显示消息。"""
    scores = [70, 100, 100]
    average = sum(scores) / len(scores)
    assert average == 90, f"平均分应该是 90，实际是 {average}"


def test_comparison_chain():
    """链式比较。"""
    age = 25
    assert 18 <= age < 60


def test_in_container():
    """判断元素是否在容器里。"""
    fruits = ["苹果", "香蕉", "橙子"]
    assert "苹果" in fruits
    assert "西瓜" not in fruits


def test_type_check():
    """类型断言。"""
    value = "42"
    assert isinstance(value, str)
    assert not isinstance(value, int)


def test_truthy():
    """真假值断言。"""
    assert not []        # 空列表是假，所以 not [] 是真
    assert [1, 2]        # 非空列表是真
    assert not ""        # 空字符串同样是假


def test_approx():
    """浮点数比较要用 approx，不能直接用 ==。"""
    result = 0.1 + 0.2
    # assert result == 0.3        # 失败！结果是 0.30000000000000004
    assert result == pytest.approx(0.3)


def test_multiple_asserts():
    """一个测试函数里可以写多个断言，全部通过才算通过。"""
    data = [1, 2, 3]
    assert len(data) == 3
    assert sum(data) == 6
    assert max(data) == 3
    assert min(data) == 1

示例 4：测试类

Python# test_bank.py —— 测试类：把相关测试归类

import pytest


class BankAccount:
    """被测的银行账户类。"""
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
        self.history = []

    def deposit(self, amount):
        """存款。"""
        if amount <= 0:
            raise ValueError("存款金额必须大于 0")
        self.balance += amount
        self.history.append(f"存款 {amount}")
        return self.balance

    def withdraw(self, amount):
        """取款。"""
        if amount > self.balance:
            raise ValueError("余额不足")
        self.balance -= amount
        self.history.append(f"取款 {amount}")
        return self.balance


class TestBankAccountDeposit:
    """测试类：只用来分组，不能有 __init__。"""

    def test_deposit_increases_balance(self):
        """存款后余额增加。"""
        account = BankAccount("小明", 100)
        account.deposit(50)
        assert account.balance == 150

    def test_deposit_returns_new_balance(self):
        """存款方法返回新的余额。"""
        account = BankAccount("小明", 100)
        result = account.deposit(50)
        assert result == 150

    def test_deposit_zero_raises(self):
        """存 0 元应该报错。"""
        account = BankAccount("小明", 100)
        with pytest.raises(ValueError):
            account.deposit(0)

    def test_deposit_negative_raises(self):
        """存负数应该报错。"""
        account = BankAccount("小明", 100)
        with pytest.raises(ValueError):
            account.deposit(-50)

    def test_history_records_transaction(self):
        """交易记录应该被保存。"""
        account = BankAccount("小明", 100)
        account.deposit(50)
        assert len(account.history) == 1
        assert "存款 50" in account.history[0]


class TestBankAccountWithdraw:
    """另一组测试，同样针对 BankAccount。"""

    def test_withdraw_reduces_balance(self):
        account = BankAccount("小明", 100)
        account.withdraw(30)
        assert account.balance == 70

    def test_withdraw_all_succeeds(self):
        """取光全部余额应该允许。"""
        account = BankAccount("小明", 100)
        account.withdraw(100)
        assert account.balance == 0

    def test_withdraw_too_much_raises(self):
        """取超额应该报错。"""
        account = BankAccount("小明", 100)
        with pytest.raises(ValueError):
            account.withdraw(150)

示例 5：参数化测试

Python# test_parametrize.py —— 用 parametrize 减少重复代码

import pytest


def is_prime(n):
    """判断 n 是否为素数。"""
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True


# ===== 不用 parametrize：写 4 个几乎一样的函数 =====
def test_prime_2():
    assert is_prime(2) is True


def test_prime_4():
    assert is_prime(4) is False


def test_prime_17():
    assert is_prime(17) is True


def test_prime_1():
    assert is_prime(1) is False


# ===== 用 parametrize：等价但更紧凑 =====
@pytest.mark.parametrize("n", [2, 3, 5, 7, 11, 13])
def test_is_prime_true(n):
    """这些数都是素数。"""
    assert is_prime(n) is True


@pytest.mark.parametrize("n", [0, 1, 4, 6, 9, 15, 100])
def test_is_prime_false(n):
    """这些数都不是素数。"""
    assert is_prime(n) is False


# ===== 多参数：一次测多组组合 =====
def divide(a, b):
    """返回 a 除以 b。"""
    return a / b


@pytest.mark.parametrize("a,b,expected", [
    (10, 2, 5.0),
    (9, 3, 3.0),
    (7, 1, 7.0),
    (-10, 2, -5.0),
    (0, 5, 0.0),
])
def test_divide(a, b, expected):
    """多组数据共用一个测试函数。"""
    assert divide(a, b) == expected


# ===== 参数化 + 异常测试 =====
@pytest.mark.parametrize("b", [0, -0.0])
def test_divide_by_zero(b):
    """除以 0 应该报错。"""
    with pytest.raises(ZeroDivisionError):
        divide(10, b)

示例 6：测试夹具（简单版）

Python# test_fixture.py —— 用 setup_method 准备测试数据

import pytest


class ShoppingCart:
    """被测的购物车。"""
    def __init__(self):
        self.items = []

    def add(self, name, price):
        self.items.append({"name": name, "price": price})

    def remove(self, name):
        self.items = [item for item in self.items if item["name"] != name]

    def total(self):
        return sum(item["price"] for item in self.items)

    def count(self):
        return len(self.items)


class TestShoppingCart:
    """用 setup_method 为每个测试准备一个全新的购物车。"""

    def setup_method(self):
        """pytest 会在每个测试函数之前自动调用这个方法。"""
        self.cart = ShoppingCart()
        self.cart.add("苹果", 5.0)
        self.cart.add("香蕉", 3.0)

    def test_cart_is_empty_before_add(self):
        """测试：初始购物车是空的。"""
        empty = ShoppingCart()
        assert empty.count() == 0
        assert empty.total() == 0.0

    def test_setup_added_two_items(self):
        """测试：setup_method 确实准备好数据了。"""
        assert self.cart.count() == 2
        assert self.cart.total() == 8.0

    def test_add_item(self):
        self.cart.add("橙子", 4.0)
        assert self.cart.count() == 3
        assert self.cart.total() == 12.0

    def test_remove_item(self):
        self.cart.remove("苹果")
        assert self.cart.count() == 1
        assert self.cart.total() == 3.0

    def test_remove_nonexistent(self):
        """删除不存在的商品不应该报错，也不应该改变购物车。"""
        self.cart.remove("西瓜")
        assert self.cart.count() == 2
