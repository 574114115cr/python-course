Python# notepad.py —— 记事本：追加与读取

NOTES_FILE = "notes.txt"


def add_note(text):
    """用 'a' 模式追加一行，不会覆盖原有内容。"""
    with open(NOTES_FILE, "a", encoding="utf-8") as f:
        f.write(text + "\n")      # write 不会自动换行，必须自己加 \n


def read_all_notes():
    """读取全部内容。"""
    with open(NOTES_FILE, "r", encoding="utf-8") as f:
        return f.read()


def count_lines():
    """统计行数。"""
    with open(NOTES_FILE, "r", encoding="utf-8") as f:
        # 方式A：读成列表再数
        lines = f.readlines()
        return len(lines)


def read_all_with_loop():
    """用 for 循环逐行读，也能数行数（大文件推荐）。"""
    count = 0
    with open(NOTES_FILE, "r", encoding="utf-8") as f:
        for line in f:
            count += 1
    return count


# ===== 测试 =====
# 先清空文件
with open(NOTES_FILE, "w", encoding="utf-8") as f:
    f.write("")                  # w 模式清空文件

add_note("第一条：学习文件读写")
add_note("第二条：理解 with 语句")
add_note("第三条：掌握 try/except")

print("--- 全部内容 ---")
print(read_all_notes())
print(f"总行数：{count_lines()}（readlines 方式）")
print(f"总行数：{read_all_with_loop()}（for 循环方式）")

# ===== r / w / a 三种模式对比实验 =====
print("\n=== 模式对比实验 ===")

add_note("第四条：模式对比")

# 错误示范：想追加却用了 w，原数据全没了
# add_note 内部用的是 a，这里手动用 w 演示危害
with open(NOTES_FILE, "r", encoding="utf-8") as f:
    before = len(f.readlines())
print(f"用 w 之前：{before} 行")

with open(NOTES_FILE, "w", encoding="utf-8") as f:
    f.write("只有这一行\n")      # w 模式清空并重写！

with open(NOTES_FILE, "r", encoding="utf-8") as f:
    after = len(f.readlines())
print(f"用 w 之后：{after} 行")

# 结论：
# 'r' 读取，文件不存在会报 FileNotFoundError
# 'w' 写入，文件不存在则创建，存在则【清空】
# 'a' 追加，文件不存在则创建，存在则在末尾添加
# 所以"只想追加内容"时必须用 'a'，用 'w' 会无声地丢失原有数据

FILENAME = "four.txt"

with open(FILENAME, "r", encoding="utf-8") as f:
    line1 = f.readline()
    line2 = f.readline()
print(f"② readline() 第一次 -> {line1!r}")
print(f"② readline() 第二次 -> {line2!r}")
print("   注意：每行末尾都带\\n 换行符，需要 strip() 才干净")

# ===== ③ readlines()：返回列表 =====
with open(FILENAME, "r", encoding="utf-8") as f:
    lines = f.readlines()
print(f"\n③ readlines() -> 类型 {type(lines).__name__}，值 {lines}")
print(f"   列表长度：{len(lines)}")

# ===== ④ for 循环逐行读（推荐处理大文件）=====
collected = []
with open(FILENAME, "r", encoding="utf-8") as f:
    for line in f:
        collected.append(line)
print(f"\n④ for 循环    -> 收集到 {collected}")
print(f"   与 readlines() 结果相同：{collected == lines}")

# ===== 关系总结 =====
print("\n=== 关系总结 ===")
print("1. read() 返回 str，readlines() 返回 list")
print("2. for line in f 收集起来的结果 == readlines() 的结果")
print("3. read() 把整个文件当成一个大字符串（含 \\n）")
print("4. readline() 每次只前进一行，是'流式'读取")
print("5. 文件大时 read() 可能内存爆，readlines() 也是，for 循环最安全")

# 更安全的逐行处理（不把所有行都存起来）
print("\n=== 逐行处理，不存列表 ===")
count = 0
with open(FILENAME, "r", encoding="utf-8") as f:
    for line in f:
        name = line.strip()          # strip() 去掉换行符
        if name:
            count += 1
            print(f"   第{count} 行：{name}")
# 这种方式不管文件多大，内存里始终只有一行

Python# safe_int.py —— 用 try/except 保护类型转换

def safe_int(text):
    """把字符串安全地转成整数。

    成功返回整数，失败返回 None。
    """
    try:
        return int(text)
    except ValueError:
        # int() 在无法转换时抛出 ValueError
        return None


# 测试各种输入
test_cases = ["123", "abc", "12.5", "", "  42  ", "-17", "1e5"]

print(f"{'输入':<12}{'repr':<12}{'结果':<12}")
print("-" * 36)
for text in test_cases:
    result = safe_int(text)
    print(f"{text:<12}{text!r:<12}{result!r:<12}")

print("\n结果分析：")
print('"123"   -> 123    成功')
print('"  42  " -> 42     成功（int 会自动忽略首尾空白）')
print('"-17"   -> -17    成功（负数也行）')
print('"abc"   -> None   ValueError：不是合法整数')
print('"12.5"  -> None   ValueError：带小数点的字符串不能直接转 int')
print('""      -> None   ValueError：空字符串无法转换')
print('"1e5"   -> None   ValueError：科学计数法要先 float 再 int')

# ===== 对比：如果不用 try/except =====
print("\n=== 不用 try 的后果 ===")
# print(int("abc"))    # ValueError: invalid literal for int() with base 10: 'abc'
# 程序会直接崩溃终止，后面所有代码都不执行
# try/except 的价值：把"崩溃"变成"返回一个可处理的失败标记"

# ===== 进阶：需要区分失败原因时用 else =====
def convert_with_reason(text):
    """成功时同时返回结果和说明，失败时说明原因。"""
    try:
        value = int(text)
    except ValueError as error:
        return None, f"转换失败：{error}"
    else:
        return value, "转换成功"


print("\n=== 用 else 区分成功与失败 ===")
for text in ["88", "abc"]:
    value, message = convert_with_reason(text)
    print(f"  {text!r} -> value={value}, {message}")

Python# score_file.py —— 读写成绩文件并统计

FILENAME = "scores.txt"

# ===== 写入成绩 =====
def write_scores(data):
    """把成绩数据写进文件（覆盖模式）。"""
    with open(FILENAME, "w", encoding="utf-8") as f:
        for name, score in data:
            f.write(f"{name},{score}\n")
    print(f"已写入 {len(data)} 条成绩")


# ===== 读取成绩 =====
def read_scores():
    """读回成绩列表，返回 [(姓名, 成绩), ...]。"""
    result = []
    with open(FILENAME, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()# 去掉换行符
            if not line:
                continue                # 跳过空行
            name, score_text = line.split(",")
            # 关键：读回来的是字符串，必须转成数字才能计算
            result.append((name, int(score_text)))
    return result


# ===== 统计 =====
def analyze(data):
    """计算平均分、最高分、及格人数。"""
    if not data:
        return None

    scores = [score for name, score in data]
    average = sum(scores) / len(scores)
    highest_name, highest_score = max(data, key=lambda item: item[1])
    passed = len([s for s in scores if s >= 60])

    return {
        "总人数": len(data),
        "平均分": average,
        "最高分": highest_name,
        "最高分值": highest_score,
        "及格人数": passed,
        "及格率": passed / len(data) * 100,
    }


# ===== 主流程 =====
# 1）显示原始文件内容
with open(FILENAME, "w", encoding="utf-8") as f:
    f.write("姓名,成绩\n")

scores_data = [
    ("小明", 92),
    ("小红", 78),
    ("小刚", 65),
    ("小丽", 88),
    ("小强", 55),
]
write_scores(scores_data)

print("\n--- 文件原始内容 ---")
with open(FILENAME, "r", encoding="utf-8") as f:
    print(f.read(), end="")

# 2）读回来分析

print(f"读回来的数据：{loaded}")

stats = analyze(loaded)
print("\n--- 统计分析 ---")
for key, value in stats.items():

    else:
        print(f"  {key}: {value}")

import os

Python# settings_json.py —— 用 JSON 做配置文件

import json
import os

FILENAME = "settings.json"

DEFAULT_SETTINGS = {
    "theme": "dark",
    "font_size": 14,
    "autosave": True,
    "languages": ["python", "english"],
}


def load_settings(filename=FILENAME):
    """读取配置文件，不存在时返回默认配置。"""
    try:
        with open(filename, "r", encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        print(f"  {filename} 不存在，返回默认配置")
        return DEFAULT_SETTINGS.copy()
    except json.JSONDecodeError:
        print(f"  {filename} 内容不是合法 JSON，返回默认配置")
        return DEFAULT_SETTINGS.copy()


def save_settings(settings, filename=FILENAME):
    """保存配置到文件。"""
    with open(filename, "w", encoding="utf-8") as f:
        # ensure_ascii=False 保证中文可读，indent=2 让人也能看
        json.dump(settings, f, ensure_ascii=False, indent=2)
    print(f"  已保存到 {filename}")


# ===== 主流程 =====
print("=== 1. 首次读取（文件不存在）===")
settings = load_settings()
print(f"  读取结果：{settings}")

print("\n=== 2. 修改并保存 ===")
settings["font_size"] = 16                    # 修改字号
settings["languages"].append("japanese")     # 往列表里追加语言
settings["last_modified"] = "2026-10-14"      # 新增一个键
save_settings(settings)

print("\n=== 3. 文件里的实际内容 ===")
with open(FILENAME, "r", encoding="utf-8") as f:
    print(f.read())

print("=== 4. 重新读取，验证修改已保存 ===")
settings2 = load_settings()
print(f"  font_size = {settings2['font_size']}")           # 16  修改生效了
print(f"  languages = {settings2['languages']}")            # 包含新加的 japanese
print(f"  last_modified = {settings2['last_modified']}")   # 新键也保存了

Python# score_manager.py —— CSV 成绩管理系统

import csv
import os

FILENAME = "scores.csv"
FIELDS = ["姓名", "数学", "英语", "总分"]


def init_file(filename=FILENAME):
    """创建文件并写入表头（覆盖已有文件）。"""
    with open(filename, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(FIELDS)
    print(f"已初始化 {filename}")


def add_student(name, math, english, filename=FILENAME):
    """追加一名学生，总分自动计算。"""
    total = math + english
    with open(filename, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow([name, math, english, total])
    print(f"已添加：{name}（总分 {total}）")


def read_all(filename=FILENAME):
    """返回所有学生数据（字典列表），文件不存在时返回空列表。"""
    try:
        with open(filename, "r", newline="", encoding="utf-8") as f:
            return list(csv.DictReader(f))
    except FileNotFoundError:
        print(f"  {filename} 不存在，返回空列表")
        return []


def class_average(subject, filename=FILENAME):
    """返回某科目的班级平均分。"""
    students = read_all(filename)
    if not students:
        return 0
    total = sum(float(s[subject]) for s in students)
    return total / len(students)


def ranking(filename=FILENAME):
    """返回按总分降序排好的学生列表。"""
    students = read_all(filename)
    return sorted(students, key=lambda s: float(s["总分"]), reverse=True)


def delete_student(name, filename=FILENAME):
    """删除指定学生：读出 -> 过滤 -> 重写文件。"""
    students = read_all(filename)
    remaining = [s for s in students if s["姓名"] != name]

    if len(remaining) == len(students):
        print(f"  未找到学生 {name}")
        return False

    # 重写整个文件
    with open(filename, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(FIELDS)
        for student in remaining:
            writer.writerow([
                student["姓名"],
                student["数学"],
                student["英语"],
                student["总分"],
            ])

    print(f"  已删除 {name}，剩余 {len(remaining)} 人")
    return True


def show_all(filename=FILENAME):
    """格式化显示所有学生。"""
    students = read_all(filename)
    if not students:
        print("  （暂无数据）")
        return

    print(f"  {'姓名':<8}{'数学':>8}{'英语':>8}{'总分':>8}")
    print("  " + "-" * 32)
    for student in students:
        print(f"  {student['姓名']:<8}{student['数学']:>8}"
              f"{student['英语']:>8}{student['总分']:>8}")

attempts = 0

while attempts < MAX_ATTEMPTS:
    raw = input("请输入 1-100 的整数：")
    attempts += 1

        value = int(raw)
    except ValueError as error:

            break
        else:
            print(f"  {value} 不在 1-100 范围内，请重新输入")
    finally:
        print(f"  （第 {attempts}/{MAX_ATTEMPTS} 次尝试结束）")

Python# todo.py —— 待办事项管理器（综合练习）

import json
import os

TASKS_FILE = "tasks.json"


class Task:
    """一个待办任务。"""
    PRIORITY_ORDER = {"高": 0, "中": 1, "低": 2}

    def __init__(self, title, priority="中", done=False):
        self.title = title
        self.priority = priority
        self.done = done

    def complete(self):
        """标记为已完成。"""
        self.done = True
        return self

    def to_dict(self):
        """转成字典，用于 JSON 序列化。"""
        return {
            "title": self.title,
            "priority": self.priority,
            "done": self.done,
        }

    @classmethod
    def from_dict(cls, data):
        """从字典还原成 Task 对象（反向操作）。"""
        return cls(
            title=data["title"],
            priority=data.get("priority", "中"),
            done=data.get("done", False),
        )

    def __str__(self):
        mark = "[已完成]" if self.done else "[待办]"
        return f"{mark} {self.title}（优先级：{self.priority}）"


class TodoManager:
    """待办事项管理器。"""

    def __init__(self, filename=TASKS_FILE):
        self.filename = filename
        self.tasks = []
        self.load()

    # ===== 文件读写 =====
    def load(self):
        """从 JSON 文件加载任务。文件不存在或损坏都不报错。"""
        try:
            with open(self.filename, "r", encoding="utf-8") as f:
                data = json.load(f)
            self.tasks = [Task.from_dict(item) for item in data]
            print(f"  已加载 {len(self.tasks)} 个任务")
        except FileNotFoundError:
            self.tasks = []
            print(f"  {self.filename} 不存在，从空列表开始")
        except json.JSONDecodeError:
            self.tasks = []
            print(f"  {self.filename} 内容损坏，从空列表开始")
        except OSError as error:
            self.tasks = []
            print(f"  读取失败：{error}")

    def save(self):
        """把所有任务保存到 JSON 文件。"""
        try:
            data = [task.to_dict() for task in self.tasks]
            with open(self.filename, "w", encoding="utf-8") as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
            print(f"  已保存 {len(data)} 个任务到 {self.filename}")
        except OSError as error:
            print(f"  保存失败：{error}")

    # ===== 任务操作 =====
    def add(self, title, priority="中"):
        """添加一个任务。"""
        if any(task.title == title for task in self.tasks):
            print(f"  任务 '{title}' 已存在，跳过")
            return False
        self.tasks.append(Task(title, priority))
        print(f"  已添加：{title}")
        return True

    def complete(self, title):
        """标记指定任务为已完成。"""
        for task in self.tasks:
            if task.title == title:
                task.complete()
                print(f"  已完成：{title}")
                return True
        print(f"  未找到任务：{title}")
        return False

    def delete(self, title):
        """删除指定任务。"""
        original_count = len(self.tasks)
        self.tasks = [t for t in self.tasks if t.title != title]
        if len(self.tasks) < original_count:
            print(f"  已删除：{title}")
            return True
        print(f"  未找到任务：{title}")
        return False

    def list_all(self, show_done=True):
        """列出任务，可选择是否显示已完成的。"""
        visible = [
            task for task in self.tasks
            if show_done or not task.done
        ]
        # 按优先级排序：高 -> 中 -> 低
        visible.sort(key=lambda t: Task.PRIORITY_ORDER.get(t.priority, 9))

        if not visible:
            print("  （没有任务）")
            return []

        print(f"  共 {len(visible)} 个任务：")
        for task in visible:
            print(f"    {task}")
        return visible

    def stats(self):
        """统计任务数量。"""
        total = len(self.tasks)
        done = len([t for t in self.tasks if t.done])
        return {
            "总数": total,
            "已完成": done,
            "待办": total - done,
            "完成率": f"{done / total * 100:.1f}%" if total else "0%",
        }

六、本单元验收自测

1. 简答题：

Pythonwith open("test.txt", "w") as f:
    f.write("第一行\n")
with open("test.txt", "w") as f:
    f.write("第二行\n")
with open("test.txt", encoding="utf-8") as f:
    print(f.read())

输出什么？

要点：只输出 第二行。第二次用 "w" 模式打开文件时，文件被清空，所以第一次写入的"第一行"被彻底丢弃。"w" 是"清空后重写"，不是"写进去"。如果想在末尾追加必须用 "a"。

2. 简答题：with open(...) 的两个主要好处是什么？为什么 f.read() 的内容要存到变量里再用？

要点：① 进入 with 块时打开文件，离开块时自动关闭，而且即使块内代码抛异常也保证会关闭（普通写法忘了 close() 会泄漏文件句柄）。② 出块后 f 本身失效（访问会报 ValueError: I/O operation on closed file）。所以必须在块内把内容读进 content 这样的变量，出了块用 content。

3. 判断题：写 try: ... except: （不带任何异常类型）是可以的推荐做法。

要点：错误，应该尽量避免。裸 except: 会捕获一切异常，包括 KeyboardInterrupt（导致 Ctrl+C 失效）和 SystemExit，还会静默吞掉真正的代码 bug（NameError、TypeError 等），让问题在别处才暴露，排查成本极高。正确做法是明确指定类型，如 except ValueError: 或 except (ValueError, KeyError):。

4. 简答题：为什么读写文件时几乎总要写 encoding="utf-8"？CSV 的 newline="" 又是为了什么？

要点：encoding="utf-8" 是为了保证中文正确读写。不指定时 Python 会用操作系统默认编码（Windows 上是 GBK/cp936），会导致中文写出乱码，或读取时报 UnicodeDecodeError。newline="" 是 csv 模块的标准要求：不写它，在 Windows 上 Python 会在 \r\n 之间再插入一个 \n，导致文件里出现空行。CSV 文件推荐搭配 encoding="utf-8-sig"，这样 Excel 打开时中文不乱码。

5. 简答题：JSON 和 Python 的类型是怎么对应的？json.dumps(data) 不加 ensure_ascii=False 会有什么现象？

要点：对应关系是 object → dict、array → list、string → str、number → int/float、true/false → True/False、null → None。不加 ensure_ascii=False 时，中文字符会被转义成 \u5c0f\u660e 这样的形式——程序能正常读回，但人完全看不懂，文件体积也更大。注意 JSON 用小写 true/false/null，与 Python 的 True/False/None 不同。
