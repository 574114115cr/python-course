基础档

练习 5-1【个人信息字典】 创建一个字典 me，包含 name、age、height、city 四个键，然后： 1. 输出字典长度 2. 取出 name 和 city 的值 3. 新增一个键 hobby 4. 修改 age 的值 5. 删除 height 键 6. 输出最终字典 每一步都输出结果，观察长度什么时候变化。

练习 5-2【通讯录】 创建 contacts = {"小明": "13800138000", "小红": "13900139000"} 1. 用 .get() 取"小刚"的电话，键不存在时返回 "未知" 2. 用方括号取"小红"的电话（已知存在） 3. 用 in 判断"小刚"是否在通讯录里 4. 新增"小刚"及其电话 5. 用 for 循环输出所有联系人和电话（用 .items()） 6. 删除"小红"

练习 5-3【成绩单】 scores = {"语文": 90, "数学": 85, "英语": 78} 1. 找出最高分的科目和分数（用 max() 配合 items() 和 key，或自己写循环） 2. 输出不及格的科目（<60 分，用推导式或循环筛选） 3. 输出所有科目名（.keys()） 4. 输出平均分

进阶档

练习 5-4【词频统计器】 给定一段英文文本（自己写一句，比如 "the quick brown fox jumps over the lazy dog the end"），用字典统计每个单词出现的次数。 要求： - 文本先转小写（.lower()） - 按空格 split 成单词列表 - 用循环 + if in 计数（这是最基础的写法，务必掌握） - 最后输出统计结果 进阶：不用空格 split 的思路，先把标点替换成空格。

Python# me_dict.py —— 字典的增删改查全流程

me = {
    "name": "小明",
    "age": 18,
    "height": 1.75,
    "city": "杭州",
}

print("初始字典：", me)
print("长度：", len(me))# 4

# 2）取出值
print("name：", me["name"])
print("city：", me["city"])

# 3）新增键：长度会 +1
me["hobby"] = "篮球"
print("新增 hobby 后长度：", len(me))    # 5

# 4）修改已有键的值：长度不变（关键知识点）
me["age"] = 19
print("修改 age 后长度：", len(me))# 仍是 5
print("现在的字典：", me)

# 5）删除键：长度会 -1
del me["height"]
print("删除 height 后长度：", len(me))# 4
print("最终字典：", me)

# 总结这张表：新增长度加1，修改长度不变，删除长度减1

Python# contacts.py —— 通讯录的基本操作

contacts = {
    "小明": "13800138000",
    "小红": "13900139000",
}

# 1）安全取不存在的键
print("小刚的电话：", contacts.get("小刚", "未知"))   # 未知

# 2）方括号取已知存在的键
print("小红的电话：", contacts["小红"])    # 13900139000

# 3）用 in 判断是否存在
print("小刚在通讯录里吗：", "小刚" in contacts)     # False
print("小明在通讯录里吗：", "小明" in contacts)     # True

# 4）新增
contacts["小刚"] = "13700137000"
print("新增后：", contacts)

# 5）遍历所有联系人和电话（用 items）
print("--- 通讯录 ---")
for name, phone in contacts.items():
    print(f"{name}: {phone}")

# 6）删除
del contacts["小红"]
print("删除小红后：", contacts)

# 顺带演示：也可以按名字筛选（比如只找出 13 开头的号码）
print("--- 号码以 138 开头的人 ---")
for name, phone in contacts.items():
    if phone.startswith("138"):
        print(f"{name}: {phone}")

Python# score_report.py —— 成绩单分析

scores = {"语文": 90, "数学": 85, "英语": 78, "物理": 55, "化学": 92}

# 1）最高分的科目和分数
#max() 的 key 参数告诉它"按什么标准比大小"
best_subject = max(scores, key=scores.get)
print(f"最高分：{best_subject}，{scores[best_subject]} 分")    # 化学，92 分

# 最直接的对比：也可以排序
sorted_items = sorted(scores.items(), key=lambda item: item[1], reverse=True)
print("第一名到第三名：")
for subject, score in sorted_items[:3]:
    print(f"  {subject}: {score} 分")

# 2）不及格的科目（<60 分）
failed = [subject for subject, score in scores.items() if score < 60]
print("不及格：", failed)      # ['物理']

# 3）所有科目名
print("所有科目：", list(scores.keys()))
print("科目数量：", len(scores))

# 4）平均分
total = sum(scores.values())
average = total / len(scores)
print(f"总分：{total}，平均分：{average:.2f}")

# 顺带：最高分和最低分的差
print("极差：", max(scores.values()) - min(scores.values()))

Python# word_count.py —— 用字典统计词频

text = "the quick brown fox jumps over the lazy dog the end"
print("原文：", text)

# 1）转小写，避免 The 和 the 被当成不同的词
text = text.lower()

# 2）按空格切分成单词列表
words = text.split(" ")          # ['the', 'quick', 'brown', ...]
print("切分结果：", words)

# 3）用字典计数（最基础也最重要的写法）
counts = {}
for word in words:
    if word in counts:
        counts[word] += 1     # 已经见过，计数加一
    else:
        counts[word] = 1      # 第一次见，计数为一

print("词频统计：", counts)
# {'the': 3, 'quick': 1, 'brown': 1, 'fox': 1, 'jumps': 1,
#  'over': 1, 'lazy': 1, 'dog': 1, 'end': 1}

# 4）找出出现次数最多的词
most_common_word = max(counts, key=counts.get)
print(f"出现最多的词：{most_common_word}，出现了 {counts[most_common_word]} 次")

# ===== 进阶版本：处理带标点的文本 =====
messy = "Hello, world! Hello Python, python is fun. Is python fun?"
print("\n原始文本：", messy)

# 先把所有常见标点替换成空格，这样 split 就能切干净
cleaned = messy
for symbol in [",", "!", ".", "?", ";", ":"]:
    cleaned = cleaned.replace(symbol, " ")
print("清洗后：", cleaned)

counts2 = {}
for word in cleaned.lower().split():
    counts2[word] = counts2.get(word, 0) + 1

print("词频：", counts2)
# 进阶写法：counts2[word] = counts2.get(word, 0) + 1
# 含义是"取这个键的值，没有就用 0；然后加 1 再存回去"。
# 这一行等价于上面的 if/else 两种情况，是最常用的计数写法。

Python# class_data.py —— 嵌套字典的综合提取

class_data = {
    "class1": {
        "teacher": "李老师",
        "students": [
            {"name": "小明", "score": 92},
            {"name": "小红", "score": 78},
            {"name": "小刚", "score": 85},
        ],
    },
    "class2": {
        "teacher": "王老师",
        "students": [
            {"name": "小丽", "score": 88},
            {"name": "小强", "score": 95},
        ],
    },
}

# 1）每个班的老师和学生人数
print("--- 班级概况 ---")
for class_name, info in class_data.items():
    print(
        f"{class_name}: 老师{info['teacher']}，"
        f"共{len(info['students'])} 名学生"
    )

# 2）所有学生中分数最高的
highest_name = ""
highest_score = -1
for class_name, info in class_data.items():
    for student in info["students"]:
        if student["score"] > highest_score:
            highest_score = student["score"]
            highest_name = student["name"]
print(f"\n最高分：{highest_name}，{highest_score} 分")

# 3）总分最高的班级
print("\n--- 班级总分 ---")
best_class = ""
best_total = -1
for class_name, info in class_data.items():
    total = 0
    for student in info["students"]:
        total += student["score"]
    average = total / len(info["students"])
    print(f"{class_name}: 总分 {total}，平均 {average:.1f}")
    if total > best_total:
        best_total = total
        best_class = class_name
print(f"总分最高的是 {best_class}（{best_total} 分）")

# 4）所有学生的平均分
all_scores = []
for class_name, info in class_data.items():
    for student in info["students"]:
        all_scores.append(student["score"])
overall = sum(all_scores) / len(all_scores)
print(f"\n全校平均分：{overall:.2f}")     # 87.60

# 5）安全地取不存在的班级
teacher = class_data.get("class3", {}).get("teacher", "班级不存在")
print("\nclass3 的老师：", teacher)

# 用推导式做同样的事，更短
all_names = [
    student["name"]
    for class_name, info in class_data.items()
    for student in info["students"]
]
print("所有学生：", all_names)

Python# reverse_lookup.py —— 字典的反向查找

scores = {"小明": 95, "小红": 88, "小刚": 72, "小丽": 91}

# 1）字典推导式构造反向字典 {分数: 姓名}
name_by_score = {score: name for name, score in scores.items()}
print("正向（姓名 -> 分数）：", scores)
print("反向（分数 -> 姓名）：", name_by_score)

# 2）查询 91 分是谁
target = 91
print(f"{target} 分是谁：", name_by_score.get(target, "查无此人"))   # 小丽

# 3）找出分数最高的学生的姓名
# 错误思路：max(name_by_score) 拿到的是最大的"键"，也就是 95
# 那是分数，不是姓名
top_name = max(scores, key=scores.get)
print(f"最高分学生：{top_name}，{scores[top_name]} 分")   # 小明，95 分
# key=scores.get 的意思是"比较时用 scores 里对应的值当标准"

# 最低分学生
low_name = min(scores, key=scores.get)
print(f"最低分学生：{low_name}，{scores[low_name]} 分")

# 4）按分数从高到低排好的姓名列表
ordered_names = [
    name
    for name, score in sorted(scores.items(), key=lambda item: item[1], reverse=True)
]
print("排名：", ordered_names)    # ['小明', '小丽', '小红', '小刚']

# 用 lambda 解释一下：item 就是一个 (姓名, 分数) 的元组，
# item[1] 就是分数，我们按分数排序。
# 如果写成 key=scores.get 也可以，那是因为 sorted 前要先转成列表对齐名字，
# 但用 lambda 更通用、更清楚。

Python# login_system.py —— 简易登录系统

users = {
    "admin": "123456",
    "root": "root888",
    "user": "abc123",
}

# 允许重试 3 次
for attempt in range(1, 4):
    print(f"--- 第 {attempt} 次尝试（共 3 次机会）---")
    username = input("请输入用户名：")
    password = input("请输入密码：")

    # 第一步：判断用户名是否存在
    if username not in users:
        print(f"该用户不存在。\n可用用户名：{list(users.keys())}")

    # 第二步：用户名存在，再判断密码
    elif users[username] == password:
        print(f"登录成功，欢迎 {username}！")
        break# 登录成功，用 break 跳出循环

    # 第三步：密码错误
    else:
        print("密码错误")

    # 如果三次都失败，会走到这里
else:
    print("三次机会已用完，账户可能被锁定")

Python# inventory.py —— 遍历、判断、统计
inventory = {"苹果": 12, "香蕉": 0, "橙子": 5, "榴莲": 0}

for name, qty in inventory.items():
    if qty > 0:
        print(f"{name} 有货（{qty}）")
    else:
        print(f"{name} 缺货")

out = [name for name, qty in inventory.items() if qty == 0]
print("缺货", out)

total = sum(inventory.values())
in_stock = len([q for q in inventory.values() if q > 0])
print(f"总数 {total}，有货品类 {in_stock}")

橙子 有货（5）
榴莲 缺货
缺货 ['香蕉', '榴莲']
总数 17，有货品类 2

inventory.items() 同时给出键和值，是遍历字典最好用的方式；只要键用 for k in inventory，只要值用 for v in inventory.values()。三个都记不住时，先想「我要不要用键」，要就用 items()。

Python# scores.py —— 列表套字典的统计
students = [
    {"name": "张三", "scores": [88, 92, 76]},
    {"name": "李四", "scores": [59, 64, 70]},
    {"name": "王五", "scores": [100, 95, 98]},
]

avgs = {}
for s in students:
    avg = sum(s["scores"]) / len(s["scores"])
    avgs[s["name"]] = avg
    tag = "及格" if avg >= 60 else "不及格"
    print(f"{s['name']} {avg:.1f} 分 · {tag}")

class_avg = sum(avgs.values()) / len(avgs)
print(f"班级平均 {class_avg:.1f} 分")

print("最高分", max(avgs, key=avgs.get))

输出：

示例张三 85.3 分 · 及格
李四 64.3 分 · 及格
王五 97.7 分 · 及格
班级平均 82.4 分
最高分 王五

三处细节：

  • 先把每人的平均分存进 avgs 字典，后面算班级平均和找最高分都直接复用，不用再遍历一次原始数据。
  • max(avgs, key=avgs.get) 是「按值找键」的惯用法：max 遍历的是键，key=avgs.get 告诉它「比较时用对应的值」。
  • :.1f 只在打印时四舍五入，avgs 里存的仍是完整精度——所以班级平均用的是 85.333…，不是 85.3。
