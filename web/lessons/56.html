def count_words(text):
    """返回文本中的单词数，以空格分隔。"""
    if not text.strip():
        return 0
    return len(text.split())


def reverse(text):
    """返回反转后的字符串。"""
    return text[::-1]


def is_palindrome(text):
    """判断是否回文（忽略大小写、空格和常见标点）。"""
    # 清洗：只保留字母和数字，统一小写
    cleaned = "".join(
        char.lower()
        for char in text
        if char.isalnum()
    )
    return cleaned == cleaned[::-1]

Python# divider.py —— 被测模块

def divide(a, b):
    """返回 a 除以 b 的结果。

    除数为 0 时会抛出 ZeroDivisionError。
    """
    return a / b


def divide_safe(a, b, default=0.0):
    """安全除法：除数为 0 时返回默认值。"""
    try:
        return a / b
    except ZeroDivisionError:
        return default

Python# book.py —— 被测模块

class Book:
    """一本书。"""
    def __init__(self, title, price, available=True):
        self.title = title
        self.price = price
        self.available = available

    def borrow(self):
        """借出图书。已被借出时抛出 ValueError。"""
        if not self.available:
            raise ValueError(f"《{self.title}》已被借出，无法再次借出")
        self.available = False
        return self

    def return_book(self):
        """归还图书。"""
        self.available = True
        return self

    def apply_discount(self, rate):
        """打折，rate 是 0-1 之间的折扣率。"""
        if not 0 <= rate <= 1:
            raise ValueError("折扣率必须在 0 到 1 之间")
        self.price = self.price * rate
        return self.price

Python# cart.py —— 被测模块

class ShoppingCart:
    """购物车。"""
    def __init__(self):
        self.items = []

    def add(self, name, price):
        """添加商品。price 必须大于 0。"""
        if price <= 0:
            raise ValueError("价格必须大于 0")
        self.items.append({"name": name, "price": price})
        return self

    def remove(self, name):
        """移除指定商品。商品不存在时不报错，返回是否真的移除了。"""
        original = len(self.items)
        self.items = [item for item in self.items if item["name"] != name]
        return len(self.items) < original

    def total(self):
        """返回总价。"""
        return sum(item["price"] for item in self.items)

    def count(self):
        """返回商品件数。"""
        return len(self.items)

    def clear(self):
        """清空购物车。"""
        self.items = []
        return self

Python# vote_analyzer.py —— 被测模块

def analyze_votes(votes):
    """分析投票结果。

    参数:
        votes: 候选人名字组成的列表

    返回:
        {
            "winner": 票数最多的候选人名字；平票或无票时为 None,
            "total": 总票数,
            "counts": {候选人: 票数},
            "tied": 是否平票,
        }
    """
    # 统计票数
    counts = {}
    for name in votes:
        counts[name] = counts.get(name, 0) + 1

    total = len(votes)

    # 空列表的边界情况
    if not counts:
        return {
            "winner": None,
            "total": 0,
            "counts": {},
            "tied": False,
        }

    # 找出最高票数
    highest_count = max(counts.values())

    # 找出所有达到最高票数的候选人
    leaders = [name for name, count in counts.items() if count == highest_count]

    # 只有一个人领先才有winner，否则是平票
    winner = leaders[0] if len(leaders) == 1 else None

    return {
        "winner": winner,
        "total": total,
        "counts": counts,
        "tied": len(leaders) > 1,
    }

七、本单元验收自测

1. 简答题：下面这个测试为什么不会被 pytest 执行？

Python# test_calc.py
def check_add(a, b):
    assert a + b == 5

要点：因为函数名不以 test_ 开头。pytest 按命名约定自动发现测试，函数必须命名成 test_* 才会被收集。如果命名不对，pytest 会显示 collected 0 items——不报错，但什么都不跑，这是最容易浪费时间的问题之一。改名为 test_add 即可。

2. 简答题：为什么测试文件应该单独放在 test_calculator.py，而不是直接写在 calculator.py 里？

要点：三个理由：① 测试代码可以被独立运行、独立删除，不会影响程序正常运行；② 被测代码保持"纯粹"，不用为了配合测试而加入奇怪的辅助代码；③ 一看文件名就能分清哪些是功能、哪些是测试。标准结构是两个文件并列：calculator.py 放函数定义，test_calculator.py 放测试。

3. 判断题：用 assert 写测试时，写 assert result is not None 就够了，不需要知道 result 具体等于多少。

要点：错误。这是一种"弱断言"，很多错误的实现也能通过它（比如函数返回 "abc"、返回 0、返回 False）。正确的做法是明确写出期望值：assert add(2, 3) == 5。测试的价值在于能区分"正确的实现"和"错误的实现"，只断言非空几乎做不到这一点。

4. 简答题：测试除法函数时，除了 divide(10, 2) == 5 这种正常情况，还必须测哪些边界情况？用什么语法写？

要点：至少要测：① 除零 divide(10, 0) —— 这是最重要的，必须用 with pytest.raises(ZeroDivisionError): 断言它抛异常；② 除以 1 divide(7, 1) == 7；③ 负数相除 divide(-10, 2) == -5；④ 除不尽 divide(10, 3) 的小数结果（用 pytest.approx 比较）；⑤ 如果函数要求整数输入，还要测小数输入。核心是异常路径也必须被测试——正常路径通常能工作，bug 都藏在异常路径里。

5. 简答题：pytest -v 输出 3 passed、1 failed，-k "divide"、-x 这两个参数分别有什么用？

要点：-v（verbose）用详细模式显示每个测试的名字和 PASSED/FAILED 状态，方便定位是哪一条失败了。-k "divide" 是按测试名字筛选，只运行名字里含 divide 的测试，在测试文件变多后能大幅缩短反馈时间。-x 是遇到第一个失败就停止（fail-fast），适合"CI 里快速发现问题"的场景。加上 pytest test_bank.py::TestBook::test_borrow 这样的完整路径，可以精确运行单个测试。

八、常见错误清单（务必逐条自查）

错误 1：pytest 报 no tests ran

症状：明明写了测试，一条都没跑。 原因：文件名不是 test_*.py，或函数名不是 test_ 开头。 改法：文件 test_xxx.py、函数 def test_xxx():；pytest -q --collect-only 先看有没有被收集到。

错误 2：断言写反

症状：代码明明有 bug，测试却绿了。 原因：assert 结果 == 3 里条件为真才通过；写成 assert 结果 != 3 就是反的。 改法：写完故意改坏一行实现，确认测试变红——不会红的就是无效测试。

错误 3：浮点数直接用 ==

症状：0.1 + 0.2 == 0.3 是 False。 原因：二进制浮点有精度误差。 改法：math.isclose(a, b) 或 abs(a-b) < 1e-9；金额用 Decimal。

错误 4：测试之间互相依赖

症状：单独跑通过，一起跑失败（或反过来）。 原因：共用了全局变量/同一个文件，执行顺序变了结果就变。 改法：每个测试自己准备数据（fixture 里创建临时目录），不依赖别的测试留下的状态。

错误 5：只测「正常路径」

症状：覆盖率挺高，但用户一输入空值就崩。 原因：只测了理想输入，没测边界与非法值。 改法：每个函数至少三条：正常、边界（空/0/最大）、非法（类型错/负数）——用 pytest.raises 测异常。
