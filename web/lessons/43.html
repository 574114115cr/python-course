示例 1：定义类与 __init__

Python# class_basics.py —— 类的定义与对象创建

class Dog:
    """一只狗。

    属性：name（名字）、age（年龄）
    方法：bark（叫）、birthday（过生日）
    """
    def __init__(self, name, age=0):
        # __init__ 在对象创建时自动执行一次
        self.name = name      # 实例属性：每个对象各有一份
        self.age = age

    def bark(self):
        """让狗叫一声。"""
        print(f"{self.name} 说：汪汪！")

    def birthday(self):
        """年龄加一。"""
        self.age += 1
        print(f"{self.name} 生日快乐，现在 {self.age} 岁了！")


# 创建对象：像调用函数一样，但造出来的是一个"实例"
dog1 = Dog("旺财", 3)
dog2 = Dog("咪咪")# age 用默认值 0

# 访问属性：不用括号
print("dog1 的名字：", dog1.name)
print("dog1 的年龄：", dog1.age)
print("dog2 的年龄：", dog2.age)

# 调用方法：用括号
dog1.bark()
dog2.bark()

# 方法可以修改属性
dog1.birthday()
dog1.birthday()
print("过完两个生日后 dog1 的年龄：", dog1.age)   # 5

# 两个对象的数据互相独立
print("dog2 的年龄没变：", dog2.age)          # 0

# 查看对象类型
print("dog1 的类型：", type(dog1))
print("类型名：", type(dog1).__name__)         # Dog

示例 2：self 到底是什么

Python# self_demo.py —— 理解 self 的本质

class Cat:
    def __init__(self, name):
        self.name = name

    def speak(self):
        # self 就是"我自己"
        print(f"{self.name} 说：喵")

    def compare(self, other):
        """比较两只猫。self 是当前这只，other 是传入的那只。"""
        print(f"self 的名字：{self.name}")
        print(f"other 的名字：{other.name}")


cat1 = Cat("小白")
cat2 = Cat("小黑")

cat1.speak()        # Python 实际执行的是 Cat.speak(cat1)，所以 self=cat1
cat2.speak()        # 这里 self=cat2
cat1.compare(cat2)  # self=cat1，other=cat2

# 手动验证：普通函数 vs 方法
def manual_speak(self):
    print(f"{self.name} 说：喵")


# 直接调用方法的两种写法完全等价
manual_speak(cat1)      # 手动把 cat1 传进去
cat1.speak()            # Python 自动帮你传了 cat1

print("两种调用方式结果相同")

# 演示：同一个类造多个对象，self 各不相同
cats = [Cat("咪咪1"), Cat("咪咪2"), Cat("咪咪3")]
print("--- 三只猫依次叫 ---")
for cat in cats:
    cat.speak()

# 常见误解澄清：self 不是"关键字"，也不是"类名"
# 它就是函数的第一个普通参数，只是约定俗成写成 self
# 你完全可以叫它别的名字（虽然不要这么做）
print("\n--- 用别的名字也能工作（但请不要这样写）---")


class Weird:
    def __init__(this):       # 用 this 代替 self
        this.value = 42

    def show(this):
        print("value =", this.value)


w = Weird()
w.show()       # 能正常工作！value = 42
# 但所有 Python 代码都用 self，保持一致才让别人看得懂

示例 3：__str__ 与属性方法对比

Python# str_demo.py —— 让对象能被读懂地打印

class Book:
    def __init__(self, title, author, price, pages):
        self.title = title
        self.author = author
        self.price = price
        self.pages = pages

    def __str__(self):
        """定义对象转成字符串时的样子。"""
        return f"《{self.title}》/ {self.author}，{self.price} 元，{self.pages} 页"


book1 = Book("Python编程", "埃里克·马瑟斯", 89, 480)
book2 = Book("流畅的Python", "阿尔·拉玛", 139, 700)

# 没有 __str__ 时的默认输出：毫无意义的内存地址
class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y


p = Point(3, 5)
print("未定义 __str__ 的输出：")
print(p)      # <__main__.Point object at 0x000001...>  看不懂

# 有了 __str__ 之后
print("\n定义 __str__ 后的输出：")
print(book1)  # 《Python编程》/ 埃里克·马瑟斯，89 元，480 页
print(book2)  # 《流畅的Python》/ 阿尔·拉玛，139 元，700 页

# __str__ 会自动用在这些地方
print(f"这本书是 {book1}")     # f-string 内部会调用 __str__
print("列表里也是：", [book1])   # 列表打印时会调用元素的 __str__

# 顺带定义一个实用方法：每页多少钱
class BookWithMethod(Book):
    """在 Book 基础上加一个方法（这里用继承演示，单元后面会详讲）。"""

    def price_per_page(self):
        """计算每页价格，保留两位小数。"""
        return self.price / self.pages


book3 = BookWithMethod("流畅的Python", "阿尔·拉玛", 139, 700)
print(f"\n《{book3.title}》每页 {book3.price_per_page():.2f} 元")

# 属性和方法一起用
def show_book(book):
    """打印书籍信息。"""
    print(f"书名：{book.title}")        # 读属性，不写括号
    print(f"作者：{book.author}")
    if hasattr(book, "price_per_page"):    # 检查有没有这个方法
        print(f"每页价格：{book.price_per_page():.2f} 元")


print("\n--- 通用打印函数 ---")
show_book(book1)       # Book 类没有 price_per_page 方法
show_book(book3)       # BookWithMethod 有

示例 4：类属性与实例属性

Python# class_vs_instance.py —— 两种属性的区别

class Car:
    """汽车类。"""

    # 类属性：所有实例共享
    wheels = 4
    made_in = "中国"

    def __init__(self, make, model, color):
        # 实例属性：每个实例独有
        self.make = make
        self.model = model
        self.color = color
        self.odometer = 0

    def drive(self, miles):
        """行驶指定里程。"""
        self.odometer += miles
        return self.odometer

    def show(self):
        return f"{self.color} {self.make} {self.model}，已行驶 {self.odometer} 公里"


car1 = Car("丰田", "卡罗拉", "红色")
car2 = Car("本田", "雅阁", "蓝色")

print(car1.show())
print(car2.show())

# 修改实例属性：只影响这一个对象
car1.color = "黑色"
print("\n改后 car1：", car1.color)      # 黑色
print("car2 不受影响：", car2.color)      # 蓝色

# 修改类属性：影响所有对象
print("\n所有车的轮子数：", car1.wheels, car2.wheels)
Car.wheels = 6
print("改类属性后：", car1.wheels, car2.wheels)   # 6 6 都变了

# 警告：不要通过实例去改类属性（这会意外创建一个实例属性，掩盖真值）
car1.wheels = 8
print("\n通过 car1 改轮子数后：")
print("  car1.wheels =", car1.wheels)      # 8
print("  car2.wheels =", car2.wheels)      # 6  没变
print("  Car.wheels =", Car.wheels)        # 6  类属性也没变
# 结论：要用 Car.wheels = X 改类属性，不要用 car1.wheels = X

# 判断属性归属的方法
print("\n--- 属性归属判断 ---")
print("wheels 在 car1 自己的字典里吗：", "wheels" in car1.__dict__)   # False
print("color 在 car1 自己的字典里吗：", "color" in car1.__dict__)     # True
print("wheels 是类属性：", "wheels" in Car.__dict__)                # True

示例 5：继承

Python# inheritance.py —— 继承：子类复用父类的属性和方法

class Animal:
    """动物基类：定义所有动物共有的东西。"""
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def __str__(self):
        return f"{self.name}，{self.age} 岁"

    def speak(self):
        """动物都会发声，但发声方式各不相同。"""
        print(f"{self.name} 发出了一声不明的声音")

    def birthday(self):
        """过生日。"""
        self.age += 1
        print(f"{self.name} 生日快乐，现在 {self.age} 岁了")


class Dog(Animal):
    """狗继承动物，自己追加"汪汪"和"fetch"。"""
    def __init__(self, name, age, breed):
        super().__init__(name, age)      # 先让父类设置 name 和 age
        self.breed = breed               # 再设置子类自己的属性

    def speak(self):                     # 重写父类的方法
        print(f"{self.name}（{self.breed}）说：汪汪汪！")

    def fetch(self):                     # 子类独有的方法
        print(f"{self.name} 去捡球了")


class Cat(Animal):
    """猫继承动物。"""
    def __init__(self, name, age, indoor=True):
        super().__init__(name, age)
        self.indoor = indoor

    def speak(self):                     # 重写
        print(f"{self.name} 说：喵~")


# 使用
dog = Dog("旺财", 3, "柴犬")
cat = Cat("咪咪", 2)

print("--- 继承来的属性 ---")
print(dog.name, dog.age, dog.breed)     # name/age 来自父类，breed 来自子类
print(cat.name, cat.age, cat.indoor)

print("\n--- 重写的方法 ---")
dog.speak()
cat.speak()

print("\n--- 继承来的方法 ---")
dog.birthday()
print(str(dog))                        # 用的是父类的 __str__

print("\n--- 子类独有的方法 ---")
dog.fetch()
# cat.fetch()                          # AttributeError：猫没有这个方法

print("\n--- super() 复用父类的 __str__ ---")


class Puppy(Dog):
    """小狗继承狗。"""
    def __str__(self):
        parent = super().__str__()      # 拿到父类生成的字符串
        return f"【幼犬】{parent}"


puppy = Puppy("豆豆", 1, "比熊")
print(puppy)      # 【幼犬】豆豆，1 岁
puppy.speak()     # 继承了狗的 speak

示例 6：方法重写与多态

Python# polymorphism.py —— 重写与多态

class Vehicle:
    """交通工具抽象基类。"""
    def __init__(self, name):
        self.name = name

    def move(self):
        """怎么移动 —— 由子类各自决定。"""
        raise NotImplementedError("子类必须重写这个方法")

    def fuel(self):
        """用什么能源。"""
        return "未定义"


class Car(Vehicle):
    def move(self):
        print(f"{self.name} 在公路上行驶")

    def fuel(self):
        return "汽油"


class ElectricCar(Vehicle):
    def move(self):
        print(f"{self.name} 在充电桩旁边安静地待着")

    def fuel(self):
        return "电能"


class Ship(Vehicle):
    def move(self):
        print(f"{self.name} 在水上航行")

    def fuel(self):
        return "柴油"


# ===== 多态：完全不需要判断类型 =====
vehicles = [
    Car("卡车"),
    ElectricCar("特斯拉"),
    Ship("泰坦尼克"),
]

print("--- 同一段代码，不同表现 ---")
for vehicle in vehicles:
    vehicle.move()                 # 每辆车各自"表现"不同

print("\n--- 统计各类型的数量（靠 isinstance 判断）---")
for vehicle in vehicles:
    print(f"{vehicle.name} 用的能源是 {vehicle.fuel()}")

print("\n--- 多态的核心价值：加新类型不用改老代码 ---")
vehicles.append(Ship("泰坦尼克号"))
print("现在有", len(vehicles), "个交通工具，上面的循环代码一个字都不用改")

print("\n--- 父类的 move 提醒：子类必须重写 ---")
v = Vehicle("抽象交通工具")
# v.move()      # NotImplementedError: 子类必须重写这个方法
print("父类的 fuel 默认返回：", v.fuel())   # 未定义

示例 7：类的组合与实际项目结构

Python# composition.py —— 组合：用类装类

class Engine:
    """发动机。"""
    def __init__(self, horsepower, fuel_type):
        self.horsepower = horsepower
        self.fuel_type = fuel_type

    def __str__(self):
        return f"{self.horsepower}马力 {self.fuel_type}发动机"


class Wheel:
    """轮胎。"""
    def __init__(self, size=17):
        self.size = size


class CarWithParts:
    """汽车：组合了发动机和轮胎。"""
    def __init__(self, make, model, engine, wheels=None):
        self.make = make
        self.model = model
        self.engine = engine            # 组合：一个对象装着另一个对象
        self.wheels = wheels or [Wheel() for _ in range(4)]

    def __str__(self):
        return f"{self.make} {self.model}，搭载{self.engine}，{len(self.wheels)} 个轮胎"

    def horsepower(self):
        """返回马力数。"""
        return self.engine.horsepower


# 使用
engine = Engine(150, "汽油")
car = CarWithParts("丰田", "卡罗拉", engine)
print(car)
print("马力：", car.horsepower())

# 修改组合的对象，汽车也跟着变（因为是同一个对象）
engine.horsepower = 200
print("改发动机后：", car.horsepower())     # 200

# 组合 vs 继承的经验法则：
# - "是一个"（is-a）关系 -> 用继承：狗 是一 种 动物
# - "有一个"（has-a）关系 -> 用组合：车 有 一个 发动机
# 拿不准时优先用组合，它更灵活、更不容易出继承耦合问题。

print("\n--- 一个更完整的实际例子 ---")


class BankAccount:
    """银行账户。"""
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
        self.history = []

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("存款金额必须大于 0")
        self.balance += amount
        self.history.append(f"存款 {amount}")
        return self.balance

    def withdraw(self, amount):
        if amount > self.balance:
            raise ValueError("余额不足")
        self.balance -= amount
        self.history.append(f"取款 {amount}")
        return self.balance

    def __str__(self):
        return f"{self.owner} 的账户余额：{self.balance} 元"


account = BankAccount("小明", 1000)
print(account)
account.deposit(500)
account.withdraw(200)
print(account)
print("操作记录：", account.history)

# try/except 属于单元 9，但这里先感受一下错误处理
try:
    account.withdraw(99999)
except ValueError as error:
    print("操作失败：", error)

示例 8：随机点名器（seed 可复现）

random 的四个函数放在一个例子里对比，用 seed(7) 固定结果，所以你跑出来应该和下面完全一致。

Python# roll_call.py —— 随机点名：抽一个、抽三个、打乱顺序
import random

names = ["张三", "李四", "王五", "赵六", "钱七", "孙八"]

random.seed(7)                          # 固定随机序列 → 结果可复现
print("抽一个  ", random.choice(names))
print("抽三个  ", random.sample(names, 3))     # 不放回，三人不重复

random.shuffle(names)                   # 原地打乱，不要接返回值
print("排队    ", names)

运行结果：

示例抽一个   王五
抽三个   ['李四', '赵六', '张三']
排队     ['王五', '赵六', '李四', '孙八', '钱七', '张三']

三处值得注意：

  1. seed(7) 之后所有结果都固定。删掉这一行，每次运行结果都不一样——写测试时这会让人误以为代码坏了。
  2. sample 抽出的三个人不会重复，choices 会（有放回）。点名、抽奖用 sample，模拟掷骰子用 choices。
  3. shuffle 之后 names 本身变了，因为它是原地修改；如果还要保留原始顺序，先 order = names[:] 复制一份再打乱。
