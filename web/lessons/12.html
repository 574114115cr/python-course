Python# reference_demo.py —— 引用陷阱的三种写法

import copy

# ===== 原写法：会互相影响 =====
cars = ["a", "b"]
other = cars# 注意：这里没有复制，只是多贴了一张标签
other.append("c")

print("实际输出：")
print(cars, other)          # ['a', 'b', 'c'] ['a', 'b', 'c']
print(cars is other)        # True

# 原因：把变量想象成贴在笔记本封面上的标签。
# other = cars 相当于给同一本笔记本再贴一张标签，
# 不管从哪张标签往本子上写字，本子都还是那一本。

# ===== 写法一：切片复制（浅拷贝）=====
cars = ["a", "b"]
other = cars[:]# 创建了一个全新的列表
other.append("c")
print("切片拷贝：", cars, other)      # ['a', 'b'] ['a', 'b', 'c']

# ===== 写法二：copy.copy() =====
cars = ["a", "b"]
other = copy.copy(cars)
other.append("c")
print("copy.copy：", cars, other)    # ['a', 'b'] ['a', 'b', 'c']

# ===== 写法三：完全独立（deepcopy）=====
cars = ["a", "b"]
other = copy.deepcopy(cars)
other.append("c")
print("deepcopy：", cars, other)     # ['a', 'b'] ['a', 'b', 'c']

Python# scores_report.py —— 成绩分析（只用内置函数，不用循环）

scores = [88, 92, 79, 95, 60, 85, 91]

# 1）最高分与最低分
print("最高分：", max(scores))          # 95
print("最低分：", min(scores))          # 60

# 2）前三名：sorted 产生新列表，所以原scores 不受影响
top3 = sorted(scores, reverse=True)[:3]
print("前三名：", top3)                 # [95, 92, 91]
print("原列表没变：", scores)

# 3）平均分：sum 求和后除以元素个数
average = sum(scores) / len(scores)
print(f"平均分：{average:.2f}")# 84.29

# 4）低于 85 分的分数：用列表推导式过滤
low_scores = [s for s in scores if s < 85]
print("低于 85 分的：", low_scores)      # [79, 60]

# 顺带演示：如果不用推导式，手动构造也能做
low_manual = []
for s in scores:
    if s < 85:
        low_manual.append(s)
print("低分（手动版）：", low_manual)

# 5）最高分和最低分的差
print("极差：", max(scores) - min(scores))          # 35

Python# stack_sim.py —— 用列表实现"后进先出"的栈

cart = []

# 入栈：用 append 放到末尾
cart.append("手机")
cart.append("耳机")
cart.append("充电器")
print("加入商品后：", cart)        # ['手机', '耳机', '充电器']

# 出栈：用 pop 从末尾取出，取出的值存入变量
popped = cart.pop()
print("结账取出：", popped)        # 充电器（最后加入的先出来）
print("剩余：", cart)              # ['手机', '耳机']

# 再弹一次
popped2 = cart.pop()
print("再次取出：", popped2)       # 耳机
print("剩余：", cart)              # ['手机']

# 统计一共弹出了几个：初始长度 3，现在长度 1，所以弹出了 2 个
initial = 3
print("弹出总数：", initial - len(cart))      # 2

# 空列表再 pop 会报 IndexError，所以要先检查
if cart:
    cart.pop()
    print("弹出后：", cart)
else:
    print("购物车已经是空的，不能再弹出")

Python# dedupe_analysis.py —— 不用 set，用列表方法做计数与查位置

nums = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3]

# 1）长度
print("列表长度：", len(nums))            # 10

# 2）统计 1 出现几次
print("1 出现了：", nums.count(1), "次")   # 2 次

# 3）找出 3 所在的所有位置
#思路：index() 只返回第一个位置，所以找到后把它替换成
#      一个绝不会出现在列表里的标记（这里是 0），然后继续往后找。
positions = []
temp = nums[:]# 用副本，不破坏原始数据

i = 0
while i < len(temp):                     # while 循环属于单元 6，先照抄即可
    if temp[i] == 3:
        positions.append(i)
        temp[i] = 0                     # 打标记，避免重复找到
    i += 1

print("3 出现的位置：", positions)        # [0, 9]
print("原数据没被破坏：", nums)

# 4）升序与降序（sorted 不修改原列表）
print("升序：", sorted(nums))             # [1, 1, 2, 3, 3, 4, 5, 5, 6, 9]
print("降序：", sorted(nums, reverse=True))

# 5）反转，且不破坏原列表
print("反转副本：", nums[::-1])
print("原列表：", nums)               # [3, 1, 4, 1, 5, 9, 2, 6, 5, 3]

# 如果允许原地修改，才用 reverse()
flipped = nums[:]
flipped.reverse()
print("原地反转：", flipped)

参考答案 2-10

Python# matrix.py —— 两种造矩阵的写法
good = [[0] * 3 for _ in range(2)]     # 每行都是新建的列表
bad = [[0] * 3] * 2                    # 复制的是同一个列表的引用

good[0][0] = 1
bad[0][0] = 1

print("good", good)
print("bad ", bad)
print("bad[0] is bad[1] →", bad[0] is bad[1])

输出：

示例good [[1, 0, 0], [0, 0, 0]]
bad  [[1, 0, 0], [1, 0, 0]]
bad[0] is bad[1] → True

原因：[[0] * 3] * 2 先造出一个内层列表 [0, 0, 0]，再用 * 2 把这个引用复制两份。所以 bad[0]、bad[1] 指向同一个对象，改哪一行都是改同一份数据。推导式 [[0] * 3 for _ in range(2)] 每轮循环都执行一次 [0] * 3，得到两个独立列表，才是正确的二维初始化。
