Python# temperature.py —— 温度转换函数

def celsius_to_fahrenheit(celsius):
    """把摄氏温度转换成华氏温度。

    参数:
        celsius: 摄氏温度，数字或数字字符串
    返回:
        华氏温度（float）
    """
    # 用 float() 转换，兼容用户输入字符串的情况
    return float(celsius) * 9 / 5 + 32


def fahrenheit_to_celsius(fahrenheit):
    """把华氏温度转换成摄氏温度。"""
    return (float(fahrenheit) - 32) * 5 / 9


# 测试：输出对照表
print(f"{'摄氏':>8} | {'华氏':>8}")
print("-" * 21)
for celsius in [0, 100, 37, -40]:
    fahrenheit = celsius_to_fahrenheit(celsius)
    print(f"{celsius:>8} | {fahrenheit:>8.1f}")

# 反向验证：来回转换应该回到原点
print("\n--- 反向转换验证 ---")
for celsius in [0, 100, 37]:
    back = fahrenheit_to_celsius(celsius_to_fahrenheit(celsius))
    print(f"{celsius} -> {celsius_to_fahrenheit(celsius):.1f} -> {back:.2f}")

# 特殊值说明：-40 摄氏正好也是 -40 华氏，这是唯一一个"两温度刻度数值相同"的点
print("\n-40 摄氏 =", celsius_to_fahrenheit(-40), "华氏")

Python# make_profile.py —— 必填 + 默认 + **extras 的组合

def make_profile(name, age=18, city="未知", **extras):
    """创建用户档案字典。

    必填：name
    选填：age（默认 18）、city（默认"未知"）
    其余任意关键字参数会进入 extras
    """
    profile = {
        "name": name,
        "age": age,
        "city": city,
    }
    # 把其余的关键字参数也加进去
    for key, value in extras.items():
        profile[key] = value
    return profile


# 测试各种调用方式

# {'name': '小明', 'age': 18, 'city': '未知'}

Python# average.py —— 可变参数求平均值

def average(*numbers):
    """返回所有传入数字的平均值，没有参数时返回 0。

    用法:average(1, 2, 3) -> 2.0
    """
    if not numbers:              # 空元组是"假值"
        return 0
    return sum(numbers) / len(numbers)


# 测试
print(average(10, 20, 30))       # 20.0
print(average(1, 2, 3, 4, 5))     # 3.0
print(average(100))               # 100.0
print(average())                  # 0   空输入不崩

# numbers 实际上是一个元组，可以用列表方法以外的方式处理
def average_verbose(*numbers):
    """同样的功能，但展示 numbers 的真实类型。"""
    print(f"  numbers 的类型是 {type(numbers).__name__}，内容是 {numbers}")
    if len(numbers) == 0:
        return 0
    return sum(numbers) / len(numbers)


print()
average_verbose(10, 20, 30)
print("平均值：", average_verbose(80, 90, 100))

# 进阶：把一个已有列表展开成可变参数

Python# longest_word.py —— 找出最长单词

def longest_word_loop(words):
    """用普通循环找最长单词，长度相同返回第一个。"""
    if not words:               # 空列表要提前处理
        return ""

    longest = words[0]          # 假设第 0 个最长
    for word in words:
        if len(word) > len(longest):     # 严格大于，所以同长度保留先来的
            longest = word
    return longest


def longest_word_max(words):
    """用内置 max 找最长单词。"""
    if not words:
        return ""
    return max(words, key=len)


# 测试
words = ["apple", "banana", "watermelon", "kiwi"]
print("输入：", words)
print("循环版：", longest_word_loop(words))     # watermelon
print("max 版：", longest_word_max(words))      # watermelon
print("两者一致：", longest_word_loop(words) == longest_word_max(words))

# 边界情况：长度相同时都应返回第一个
tie = ["apple", "mango", "pear"]              # mango 和 pear 都是 5 个字母
print()
print("并列时循环版：", longest_word_loop(tie))    # mango
print("并列时 max 版：", longest_word_max(tie))     # mango
# max 也返回第一个，这是 Python 的稳定行为

# 空列表
print()
print("空列表：", repr(longest_word_loop([])), repr(longest_word_max([])))

# 顺带演示：所有单词都返回
def all_longest_words(words):
    """返回所有达到最长长度的单词。"""
    if not words:
        return []
    max_len = max(len(w) for w in words)
    return [w for w in words if len(w) == max_len]


print("所有最长的：", all_longest_words(tie))    # ['mango', 'pear']

Python# text_analyzer.py —— 文本统计分析函数

def analyze_text(text):
    """分析一段文本，返回包含各项统计信息的字典。

    返回的字典包含：
        字符数     不含空格的字符总数
        单词数     按空格切分后的单词数量
        最长单词   长度最大的单词
        各单词次数 {单词: 出现次数}
    """
    # 切分成单词列表
    words = text.split()

    # 统计各单词出现次数（单元 5 的计数方法）
    counts = {}
    for word in words:
        counts[word] = counts.get(word, 0) + 1

    # 找最长单词
    longest = ""
    for word in words:
        if len(word) > len(longest):
            longest = word

    return {
        "字符数": len(text.replace(" ", "")),
        "单词数": len(words),
        "最长单词": longest,
        "各单词次数": counts,
    }


def print_report(text):
    """调用 analyze_text 并格式化输出所有信息。"""
    result = analyze_text(text)

    print("=" * 40)
    print("文本分析报告")
    print("=" * 40)
    print(f"原文：{text}")
    print(f"字符数（不含空格）：{result['字符数']}")
    print(f"单词数：{result['单词数']}")
    print(f"最长单词：{result['最长单词']}（{len(result['最长单词'])} 个字母）")
    print("-" * 40)
    print("词频统计：")
    for word, count in sorted(result["各单词次数"].items(), key=lambda x: -x[1]):
        print(f"  {word:<10} {count} 次")


# 测试
text = "the quick brown fox jumps over the lazy dog"
print_report(text)

# 进阶：统计出现次数最多的词

print(f"出现最多的词是 '{top_word}'，出现了 {result['各单词次数'][top_word]} 次")

Python# multi_return.py —— 多返回值与解包

def analyze_numbers(numbers):
    """返回 (总和, 平均值, 最大值, 最小值, 偶数个数) 组成的元组。"""
    total = sum(numbers)
    average = total / len(numbers)
    largest = max(numbers)
    smallest = min(numbers)
    even_count = 0
    for n in numbers:
        if n % 2 == 0:
            even_count += 1
    # 逗号分隔多个值，Python 会自动打包成元组返回
    return total, average, largest, smallest, even_count


def count_vowels(text):
    """返回 (元音数量, 元音列表) 组成的元组。"""
    vowels = "aeiouAEIOU"
    found = []
    for char in text:
        if char in vowels:
            found.append(char)
    return len(found), found


# ===== 解包接收全部结果 =====
data = [3, 8, 12, 5, 20, 7]

total, average, largest, smallest, even_count = analyze_numbers(data)
print(f"总和：{total}")
print(f"平均值：{average:.2f}")
print(f"最大值：{largest}")
print(f"最小值：{smallest}")
print(f"偶数个数：{even_count}")

# 元组也可以整体接收，然后按下标访问
result = analyze_numbers(data)
print(f"\n下标访问：总和 {result[0]}，最大 {result[2]}")

# 星号解包：只要中间几个，中间用 * 忽略
total, _, largest, *_ = analyze_numbers(data)
print(f"只要总和和最大值：{total}，{largest}")

# 元音统计

Python# timer_decorator.py —— 用"函数返回函数"实现计时器

import time


def make_timer(func):
    """把一个函数包成"会自动打印耗时"的版本。"""
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)      # 真正干活
        end = time.time()
        print(f"  [{func.__name__}] 耗时 {end - start:.6f} 秒")
        return result                      # 别忘了把结果传出去
    return wrapper


# 被包装的函数：故意做点耗时的活
def slow_sum(n):
    """累加 1 到 n 的和，n 大一点才有感。"""
    total = 0
    for i in range(1, n + 1):
        total += i
    return total


def quick_answer():
    """一个几乎不耗时的函数。"""
    return 42


# ===== 使用 =====
timed_slow_sum = make_timer(slow_sum)
timed_answer = make_timer(quick_answer)

print("调用被包装的慢函数：")
result1 = timed_slow_sum(1000000)
print("  返回值：", result1)

print("\n调用被包装的快函数：")
result2 = timed_answer()
print("  返回值：", result2)

# ===== 观察问题 =====
print("\n--- 问题：被包装后函数名变了 ---")
print("timed_slow_sum 的 __name__ 是：", timed_slow_sum.__name__)
print("原函数 slow_sum 的 __name__ 是：", slow_sum.__name__)
# 打印栈信息或报错时，会显示 wrapper 这个没有意义的名字

# ===== 修复：手动把名字复制过去 =====
timed_slow_sum.__name__ = slow_sum.__name__
print("\n修复后 timed_slow_sum 的 __name__ 是：", timed_slow_sum.__name__)

# Python 内置的functools.wraps 就是自动做这件事的工具：
import functools


def make_timer_pro(func):
    """用 functools.wraps 保留原函数的名字和文档。"""
    @functools.wraps(func)         # 这一行等价于上面的手动赋值 + 复制文档
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"  [{func.__name__}] 耗时 {end - start:.6f} 秒")
        return result
    return wrapper


print("\n用 wraps 版本再试一次：")
timed_slow_sum2 = make_timer_pro(slow_sum)
timed_slow_sum2(100000)
print("  名字是：", timed_slow_sum2.__name__)
print("  文档是：", timed_slow_sum2.__doc__)

# 思考题答案：
# 保留原函数名很重要，因为报错信息、帮助文档、日志都会用到它。
# 如果用户看到一个叫 wrapper 的函数名在报错里，他会完全不知道
# 自己调用的是哪个函数。这正是 @装饰器 语法糖要解决的问题。

参考答案 7-9

Python# （片段）geometry.py —— 只放函数，不放副作用
def area(w, h):
    """矩形面积。"""
    return w * h


def perimeter(w, h):
    """矩形周长。"""
    return 2 * (w + h)


def is_square(w, h):
    """宽高相等即正方形。"""
    return w == h

Python# （片段）main.py —— 与 geometry.py 同目录
from geometry import area, perimeter, is_square

w, h = 3, 4
print("面积", area(w, h))
print("周长", perimeter(w, h))
print("是正方形？", is_square(w, h))

输出：

示例面积 12
周长 14
是正方形？ False

关键点：geometry.py 里一个 print 都没有。如果它在顶层写了 print("加载成功")，那么每次 import geometry 都会打印一次——导入本该是安静的。

参考答案 7-10

Python# （片段）geometry.py —— 加自测块
def area(w, h):
    return w * h


def perimeter(w, h):
    return 2 * (w + h)


def is_square(w, h):
    return w == h


if __name__ == "__main__":
    # 直接运行 python geometry.py 时才会执行这三行
    assert area(3, 4) == 12
    assert perimeter(3, 4) == 14
    assert is_square(4, 4) is True
    print("geometry.py 自测通过")

Python# （片段）main.py —— 长函数名用别名
from geometry import area, is_square
from geometry import perimeter as peri

w, h = 3, 4
print("面积", area(w, h))
print("周长", peri(w, h))          # 少写 7 个字符，但读起来仍然清楚
print("是正方形？", is_square(w, h))

③ 的答案：把 geometry.py 改名成 random.py 后，同目录下任何 import random 都会先导入你自己这个文件（Python 按 sys.path 顺序找，当前目录排在最前），于是 random.choice、random.shuffle 全部消失，标准库的 random 就再也用不上了。这类「影子模块」是新手最莫名其妙的一类 bug：报错信息只会说 module 'random' has no attribute 'choice'，看着像标准库坏了。
