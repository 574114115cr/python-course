示例 2：datetime 解析与格式化

Python# datetime_demo.py —— strptime / strftime 练习

from datetime import datetime, timedelta

# ===== 1. strptime：字符串 -> 日期对象（parse 解析）=====
samples = [
    ("2021-07-01", "%Y-%m-%d"),
    ("2021/07/01", "%Y/%m/%d"),
    ("2021-07-01 14:30:05", "%Y-%m-%d %H:%M:%S"),
    ("July 1, 2021", "%B %d, %Y"),
]

print("--- strptime 解析 ---")
for text, fmt in samples:
    date_obj = datetime.strptime(text, fmt)
    print(f"{text!r:24} + {fmt!r:22} -> {date_obj}")

# ===== 2. 格式码必须和字符串严格匹配，否则 ValueError =====
print("\n--- 格式不匹配会怎样 ---")
try:
    datetime.strptime("2021-07-01", "%Y/%m/%d")     # 字符串是 -，格式写 /
except ValueError as error:
    print("解析失败：", error)

# ===== 3. strftime：日期对象 -> 字符串（format 格式化）=====
first_date = datetime.strptime("2021-07-01", "%Y-%m-%d")
print("\n--- strftime 格式化 ---")
print(first_date.strftime("%Y年%m月%d日"))            # 2021年07月01日
print(first_date.strftime("%B"))                      # July（月份名）
print(first_date.strftime("%A"))                      # Thursday（星期名）

# ===== 4. 日期对象能做算术，字符串不能 =====
tomorrow = first_date + timedelta(days=30)
print(f"\n{first_date.date()} 往后 30 天是 {tomorrow.date()}")
print("字符串直接相加:", "2021-07-01" + "30")          # 2021-07-0130，没有意义

import csv
from datetime import datetime
from pathlib import Path

# ===== 1. 造一份"有缺失"的死亡谷风格 CSV =====
# 表头：STATION, NAME, DATE, TMAX, TMIN, TOBS
# 故意让 2021-03-05 这天的 TMAX/TMIN 缺失
rows = [
    ["STATION", "NAME", "DATE", "TMAX", "TMIN", "TOBS"],
    ["USC00042319", "DEATH VALLEY, CA US", "2021-01-01", "71", "51", "56"],
    ["USC00042319", "DEATH VALLEY, CA US", "2021-01-02", "69", "48", "52"],
    ["USC00042319", "DEATH VALLEY, CA US", "2021-03-05", "", "", ""],
    ["USC00042319", "DEATH VALLEY, CA US", "2021-01-03", "74", "53", "58"],
]

path = Path("death_valley_2021_simple.csv")
with open(path, "w", newline="", encoding="utf-8") as f:
    csv.writer(f).writerows(rows)

# ===== 2. 教材同款：提取 + 缺失值处理 =====
lines = path.read_text(encoding="utf-8").splitlines()
reader = csv.reader(lines)
header_row = next(reader)

print("表头:", header_row)       # 先打印表头找列号（好习惯）

dates, highs, lows = [], [], []
missing_days = 0
for row in reader:
    current_date = datetime.strptime(row[2], "%Y-%m-%d")
    try:
        high = int(row[3])
        low = int(row[4])
    except ValueError:
        # 空字符串转 int 会抛 ValueError：这一天没测到数据
        print(f"Missing data for {current_date}")
        missing_days += 1
    else:
        # 只有两个值都转换成功，才一起收进三个列表
        # 保证 dates/highs/lows 永远等长、一一对应
        dates.append(current_date)
        highs.append(high)
        lows.append(low)

print(f"\n共 {len(dates)} 天有效数据，跳过 {missing_days} 天缺失数据")
print(f"三个列表等长吗？dates={len(dates)}, highs={len(highs)}, lows={len(lows)}")

# ===== 3. 为什么 else 不能省？ =====
# 如果把 append 写进 try：遇到 TMAX 缺失的这天，
# high = int("") 先抛异常，而 dates 已经在异常前 append 过了
# 结果 dates 比 highs 长 -> 画图时报错或整列错位

path.unlink()        # 清理

示例 4：matplotlib 画温度区间图

Python# plot_temperatures.py —— plot + fill_between 画温度区间带

import csv
import random
from datetime import datetime
from pathlib import Path

import matplotlib.pyplot as plt

random.seed(42)        # 固定随机种子，每次运行生成的数据一样

# ===== 1. 准备数据：生成 30 天气温样本（2021 年 7 月）=====
path = Path("weather_data/sample_july.csv")
path.parent.mkdir(exist_ok=True)

with open(path, "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["STATION", "NAME", "DATE", "TAVG", "TMAX", "TMIN"])
    for day in range(1, 31):
        # 模拟气温：最高 15~22 之间波动，最低 = 最高 - 5~9
        high = random.randint(15, 22)
        low = high - random.randint(5, 9)
        writer.writerow(
            ["USW00025333", "SITKA AIRPORT, AK US", f"2021-07-{day:02d}", "", high, low]
        )
print(f"已生成 {path}")

# ===== 2. 读数据：日期字符串用 strptime 转成日期对象 =====
lines = path.read_text(encoding="utf-8").splitlines()
reader = csv.reader(lines)
next(reader)           # 跳过表头

dates, highs, lows = [], [], []
for row in reader:
    dates.append(datetime.strptime(row[2], "%Y-%m-%d"))
    highs.append(int(row[4]))
    lows.append(int(row[5]))

# ===== 3. 画图 =====
plt.style.use("seaborn-v0_8")         # 皮肤；这行删掉也能跑
fig, ax = plt.subplots(figsize=(10, 6))

ax.plot(dates, highs, c="red", alpha=0.5, label="High")
ax.plot(dates, lows, c="blue", alpha=0.5, label="Low")
ax.fill_between(dates, highs, lows, facecolor="blue", alpha=0.1)

ax.set_title("Daily High and Low Temperatures - July 2021", fontsize=20)
ax.set_ylabel("Temperature (C)", fontsize=14)
ax.legend()                            # 显示图例
fig.autofmt_xdate()                    # 日期标签倾斜，防止重叠

plt.savefig("温度区间图.png", dpi=150, bbox_inches="tight")
print("图片已保存为 温度区间图.png")

# 教材里这一步是 plt.show()，会弹出窗口；
# savefig 存成文件，画出来的图完全一样，无图形界面的环境也能跑

示例 5：plotly 画全球地震散点图

Python# eq_world_map.py —— plotly 散点地图

import json
import random
from pathlib import Path

import plotly.express as px

random.seed(16)

# ===== 1. 生成 12 条模拟地震（结构和 USGS GeoJSON 一致）=====
# 真实项目里把这一步换成 path.read_text() 读下载的 geojson 文件
# 地震带中心（经度, 纬度）：夏威夷、日本、秘鲁、阿拉斯加、巴新
BELTS = [(-156.5, 19.3), (142.8, 37.5), (-74.3, -14.6), (-176.7, 51.8), (147.0, -6.0)]

features = []
for i in range(1, 13):
    lon0, lat0 = random.choice(BELTS)
    mag = round(random.uniform(1.0, 5.8), 1)
    features.append({
        "type": "Feature",
        "properties": {"mag": mag, "title": f"M {mag} - simulated quake #{i}"},
        "geometry": {"type": "Point",
                     "coordinates": [round(lon0 + random.uniform(-4, 4), 2),
                                     round(lat0 + random.uniform(-4, 4), 2),
                                     round(random.uniform(5, 60), 1)]},
        "id": f"sim{i}",
    })

eq_data = {"type": "FeatureCollection",
           "metadata": {"title": "Simulated M1+ Earthquakes, Past 30 Days", "count": 12},
           "features": features}

path = Path("eq_data_30_day_m1.geojson")
path.write_text(json.dumps(eq_data, ensure_ascii=False, indent=2), encoding="utf-8")

# ===== 2. 提取三个列表 =====
all_eq_dicts = json.loads(path.read_text(encoding="utf-8"))["features"]

mags, lons, lats, eq_titles = [], [], [], []
for eq_dict in all_eq_dicts:
    mags.append(eq_dict["properties"]["mag"])
    lons.append(eq_dict["geometry"]["coordinates"][0])   # 经度
    lats.append(eq_dict["geometry"]["coordinates"][1])   # 纬度
    eq_titles.append(eq_dict["properties"]["title"])

# ===== 3. 画图 =====
title = "Global Earthquakes"
fig = px.scatter_geo(
    lat=lats, lon=lons, size=mags, title=title,
    color=mags,
    color_continuous_scale="Viridis",
    labels={"color": "Magnitude"},
    projection="natural earth",
    hover_name=eq_titles,
)

fig.write_html("全球地震散点图.html")
print("已生成 全球地震散点图.html，用浏览器打开即可交互查看")

path.unlink()        # 清理

示例 6：不下载文件也能练 CSV——StringIO 造表 + 按年统计（csv_stringio.py）

2.2 节的「三件套」需要真文件，rw_visual 那类脚本更是非得有 weather_data/*.csv 才能跑。练手阶段不必等下载：io.StringIO 能把一段文本伪装成文件对象，csv.DictReader 照样能读。这个示例用 6 行假数据把「读表头 → 处理缺失值 → 按年统计 → 转成 JSON」整条链路走一遍——换成真文件时，只需把 StringIO 那行换成 open(...)。

Python# csv_stringio.py —— 不下载文件也能练：用 StringIO 造表 + 按年统计
import csv
import io
from datetime import datetime

CSV_TEXT = """STATION,NAME,DATE,TMAX,TMIN
USC00042319,DEATH VALLEY,2021-01-01,71,51
USC00042319,DEATH VALLEY,2021-07-11,128,91
USC00042319,DEATH VALLEY,2021-12-31,64,40
USC00042319,DEATH VALLEY,2022-06-17,,88
USC00042319,DEATH VALLEY,2022-07-09,127,90
USW00025333,SITKA,2022-07-05,64,52
"""

stream = io.StringIO(CSV_TEXT)          # 文件对象？StringIO 假装自己是文件
reader = csv.DictReader(stream)          # ★ 用表头当键，不用记列号
print("表头：", reader.fieldnames)
rows = list(reader)
print(f"数据行：{len(rows)} 条；第一行：{rows[0]}")

by_year = {}                             # 年份 -> {"tmax": 最高温, "when": 哪天}
missing = 0
for row in rows:
    when = datetime.strptime(row["DATE"], "%Y-%m-%d")
    try:
        high = int(row["TMAX"])
    except (TypeError, ValueError):
        missing += 1                     # 空字段 -> None，int(None) 抛 TypeError
        print(f"  跳过 {row['DATE']}：TMAX 缺失")
        continue
    best = by_year.setdefault(when.year, {"tmax": high, "when": when})
    if high > best["tmax"]:
        by_year[when.year] = {"tmax": high, "when": when}

print(f"\n=== 每年最高温（共 {len(by_year)} 年，跳过 {missing} 条缺数据）===")
for year in sorted(by_year):
    item = by_year[year]
    print(f"{year}：{item['tmax']}°F 出现在 {item['when']:%Y-%m-%d}")

print("\n=== 两个站点各有多少条 ===")
stations = {}
for row in rows:
    stations[row["NAME"]] = stations.get(row["NAME"], 0) + 1
for name, count in sorted(stations.items()):
    print(f"  {name:<14}{count} 条")

print("\n=== 换成 json 输出（plotly 那边要的就是这种结构）===")
import json

payload = [{"year": year, "tmax": item["tmax"],
            "date": item["when"].strftime("%Y-%m-%d")}
           for year, item in sorted(by_year.items())]
print(json.dumps(payload, ensure_ascii=False))

运行结果（实跑输出）：

示例表头： ['STATION', 'NAME', 'DATE', 'TMAX', 'TMIN']
数据行：6 条；第一行：{'STATION': 'USC00042319', 'NAME': 'DEATH VALLEY', 'DATE': '2021-01-01', 'TMAX': '71', 'TMIN': '51'}
  跳过 2022-06-17：TMAX 缺失

=== 每年最高温（共 2 年，跳过 1 条缺数据）===
2021：128°F 出现在 2021-07-11
2022：127°F 出现在 2022-07-09

=== 两个站点各有多少条 ===
  DEATH VALLEY  5 条
  SITKA         1 条

=== 换成 json 输出（plotly 那边要的就是这种结构）===
[{"year": 2021, "tmax": 128, "date": "2021-07-11"}, {"year": 2022, "tmax": 127, "date": "2022-07-09"}]

三条解读：

  1. DictReader 让「列号硬编码」这个坑彻底消失。2.2 节说别写 row[4]，DictReader 用表头当键，row["TMAX"] 永远不会因为列序调整而取错；代价是每行多一个 dict，4000 行的表也就几 MB，值得。空字段读出来是空字符串 ""（不是 None），int("") 抛的是 ValueError。
  2. by_year.setdefault(...) 与「先赋值再比」两行配合，才不漏第一年。少了 setdefault 那行，best["tmax"] 会 KeyError；少了后面那个 if，每年只会记住第一天。这类「累加式最大值」写错时不报错，只是结果偏向第一个样本——所以一定要像本示例一样把每年结果打印出来核对。
  3. 把 CSV 多加一列 TMIN 之后再加一列 TAVG 会怎样：统计逻辑一行都不用改，因为 DictReader 只按名字取；但如果换成 csv.reader + row[4]，加一列就让所有取值整体错位一列，页面照样出图，数字全错。这就是教材反复强调「先取表头」的原因。

四、动手练习

基础档

练习 16-1【表头定位器】 写 header_locator.py： 1. 生成一份 CSV：表头 ["城市", "日期", "最高温", "最低温"]，5 行数据（同一城市 5 天，日期形如 2021-07-01） 2. 读取文件，用 enumerate 打印「列号: 列名」 3. 写函数 find_column(header, name)：输入表头列表和列名，返回列号；找不到返回 -1 4. 用找到的列号提取数据，按 2021-07-01 北京 最高 35 最低 26 的格式打印 5. 验收：把表头顺序打乱成 ["日期", "城市", "最低温", "最高温"] 再跑一遍，结果依然正确——因为你是按名字找列，不是硬编码

练习 16-2【日期格式转换器】 写 date_converter.py： 1. 函数 parse_date(text)：依次尝试 "%Y-%m-%d"、"%Y/%m/%d"、"%Y年%m月%d日" 三种格式解析 2. 成功返回 (True, 日期对象)，三种都失败返回 (False, None) 3. 主程序用 for 测试："2021-07-01"、"2021/07/01"、"2021年7月1日"、"07-01-2021"、"乱七八糟" 4. 解析成功的再用 strftime 输出成 2021年07月01日 星期四（提示：%A 的输出和系统语言有关，想稳定可以用 weekday() 自己映射） 5. 验收：前三个成功、后两个失败且程序不崩溃；能说清失败分别败在哪个字符上

Python# date_converter.py —— 多格式日期解析

from datetime import datetime

FORMATS = ["%Y-%m-%d", "%Y/%m/%d", "%Y年%m月%d日"]
WEEKDAYS = ["星期一", "星期二", "星期三", "星期四", "星期五", "星期六", "星期日"]

def parse_date(text):
    """依次尝试多种格式解析日期。

    成功返回 (True, 日期对象)，全部失败返回 (False, None)。
    """
    for fmt in FORMATS:
        try:
            return True, datetime.strptime(text, fmt)
        except ValueError:
            continue                     # 这种格式不行，换下一种
    return False, None

test_inputs = ["2021-07-01", "2021/07/01", "2021年7月1日", "07-01-2021", "乱七八糟"]

for text in test_inputs:
    ok, date_obj = parse_date(text)
    if ok:
        weekday = WEEKDAYS[date_obj.weekday()]      # weekday(): 周一=0
        pretty = f"{date_obj.year}年{date_obj.month:02d}月{date_obj.day:02d}日 {weekday}"
        print(f"{text!r:12} -> 解析成功: {date_obj.date()}, 格式化: {pretty}")
    else:
        print(f"{text!r:12} -> 解析失败（三种格式都不匹配）")

# 失败原因分析：
# "07-01-2021"：格式要求年份在前，它把 07 当年份、01 当月、2021 当日 -> 日越界
# "乱七八糟"：没有数字，三种格式全部 ValueError

Python# death_valley_year.py —— 全年温度图：读数据 -> 洗数据 -> 画图

import csv
import math
import random
from datetime import datetime, timedelta
from pathlib import Path

import matplotlib.pyplot as plt

random.seed(2021)

# ===== 1. 生成 365 天死亡谷风格数据 =====
# 用正弦曲线造"夏热冬暖"的基准温度，再加随机波动；约 5% 的天数缺数据
start = datetime(2021, 1, 1)
rows = [["STATION", "NAME", "DATE", "TMAX", "TMIN"]]

for day in range(365):
    date_text = (start + timedelta(days=day)).strftime("%Y-%m-%d")
    base = 30 + 10 * math.sin(2 * math.pi * (day - 200) / 365)   # 40（夏）~ 20（冬）
    high = int(base + random.uniform(-4, 4))
    low = high - random.randint(9, 14)
    if random.random() < 0.05:                                   # 5% 概率缺数据
        rows.append(["USC00042319", "DEATH VALLEY, CA US", date_text, "", ""])
    else:
        rows.append(["USC00042319", "DEATH VALLEY, CA US", date_text, high, low])

path = Path("weather_data/death_valley_2021_full.csv")
path.parent.mkdir(exist_ok=True)
with open(path, "w", newline="", encoding="utf-8") as f:
    csv.writer(f).writerows(rows)
print(f"已生成 {path}")

# ===== 2. 读取 + 缺失值处理 =====
lines = path.read_text(encoding="utf-8").splitlines()
reader = csv.reader(lines)
next(reader)

dates, highs, lows = [], [], []
for row in reader:
    current_date = datetime.strptime(row[2], "%Y-%m-%d")
    try:
        high = int(row[3])
        low = int(row[4])
    except ValueError:
        print(f"Missing data for {current_date}")
    else:
        dates.append(current_date)
        highs.append(high)
        lows.append(low)

print(f"有效天数: {len(dates)}")
assert len(dates) == len(highs) == len(lows), "三个列表不等长，肯定有 bug"

# ===== 3. 画图 =====
fig, ax = plt.subplots(figsize=(12, 6))
ax.plot(dates, highs, c="red", alpha=0.5, label="High")
ax.plot(dates, lows, c="blue", alpha=0.5, label="Low")
ax.fill_between(dates, highs, lows, facecolor="blue", alpha=0.1)

ax.set_title(f"Death Valley 2021 ({len(dates)} valid days)", fontsize=20)
ax.set_ylabel("Temperature (F)", fontsize=14)
ax.legend()
fig.autofmt_xdate()

plt.savefig("死亡谷2021.png", dpi=150, bbox_inches="tight")
print("图片已保存为 死亡谷2021.png")

path.unlink()        # 清理

    quakes.append({

        "title": eq_dict["properties"]["title"],
        "lon": eq_dict["geometry"]["coordinates"][0],
        "lat": eq_dict["geometry"]["coordinates"][1],
    })

# ===== 3. 震级最大的 3 次 =====
print("--- 震级最大的 3 次地震 ---")
top3 = sorted(quakes, key=lambda q: q["mag"], reverse=True)[:3]

# ===== 4. 按震级分组统计 =====

print(f"\nM1 以下: {below_1} 次")

    else:
        status = "坐标可疑！"

六、常见错误清单（务必逐条自查）

错误 1：★ 把 coordinates[1] 当成经度（经纬度写反）

GeoJSON 的坐标是 [经度, 纬度, 深度]。写成 lat = coordinates[0]、lon = coordinates[1]，程序不报错，但所有点跑到镜像位置。自检方法：纬度绝对值不应超过 90。每次写完 coordinates，默念「先 lon 后 lat」。

错误 2：★ 硬编码列号 row[4]

数据源一更新（列顺序变了、多了几列），程序要么 IndexError，要么安静地取错列。正确做法：先 enumerate 打印表头，按列名建字典 {name: index}，再取数。

错误 3：strptime 格式码和字符串不匹配

"2021-07-01" 配 "%Y/%m/%d"、"2021/7/1" 配 "%Y-%m-%d"（少前导零也麻烦），都会抛 ValueError: time data ... does not match format。排查方法：把字符串和格式码并排写在纸上，一个字符一个字符对。

错误 4：把 append 写进 try 块

某天 TMAX 缺失时，int("") 抛异常，但异常前的 dates.append 已经执行，lows 也可能转换成功——三个列表错位，画图时报一个看不懂的错，或者更糟：画出一张安静的错图。记住：append 放 else，保证「要么三个都进，要么都不进」。

错误 5：plt.show() 在无图形界面的环境卡住

远程服务器、WSL、某些 IDE 里 plt.show() 会报错或让程序挂起。本讲义统一用 plt.savefig("图.png", dpi=150, bbox_inches="tight") 存文件，哪里都能跑。教材未涉及，补充说明。

错误 6：细节两件套——样式名和 CSV 参数

plt.style.use("seaborn") 在新版 matplotlib 会报 ValueError，要写 "seaborn-v0_8"（样式只是皮肤，删掉也行）；写 CSV 忘加 newline="" 会在 Windows 上多出空行，忘加 encoding="utf-8" 中文可能变乱码。

七、本单元验收自测

1. 简答题：读取 CSV 时，为什么教材不直接写 row[4] 取最高温，而要先用 enumerate 打印表头？

要点：真实数据的列顺序不保证。今天 DATE 在第 2 列，数据源一更新可能跑到第 3 列。硬编码列号的程序会安静地取错列或直接 IndexError。先用 enumerate 建立「列名 → 列号」的映射再取数，数据怎么变都不会错。这是处理外部数据的基本修养：永远不要信任你没验证过的数据结构。

2. 判断题：GeoJSON 的 coordinates 是 [纬度, 经度, 深度]。

要点：错误。是 [经度, 纬度, 深度]——经度在前。coordinates[0] 是 lon（东西向，±180 内），coordinates[1] 是 lat（南北向，±90 内）。这个错误不报错、只静默地错，是最危险的一类 bug。自检：纬度超过 90 就一定写反了。

3. 简答题：datetime.strptime("2021-07-01", "%Y-%m-%d") 和 .strftime("%Y年%m月%d日") 分别在做什么？格式码和字符串不匹配会怎样？

要点：strptime 是 parse（解析）：字符串 → 日期对象，第二个参数是「格式说明书」；不匹配抛 ValueError。strftime 是 format（格式化）：日期对象 → 字符串，格式码描述「想要什么样的输出」，一般不抛错，只会得到不符合预期的输出。记忆：p=parse 解析，f=format 格式化。
