2.7 edit_entry 的双重保护

Entry 自己没有 owner 字段——它的归属是从 topic 继承的（教材的设计：只需把最高层数据关联到用户，低层数据自动跟着归属）。所以保护编辑页面时，要顺着「条目 → 主题 → 用户」追两层：

Python# learning_logs/views.py（片段）—— edit_entry() 的归属检查
entry = Entry.objects.get(id=entry_id)
topic = entry.topic
if topic.owner != request.user:
    raise Http404

类比：进房间前要验两道——先查这个房间号（entry）在哪一层（topic），再查这一层属于哪位住户（owner）。只验房间号不够：房间号对了，但那是别人家的楼层。

new_entry 同样要防（教材练习 19.4）：用户如果在 URL 里塞别人的 topic_id（比如 /new_entry/1/），就能往别人的主题里塞条目。保存前先问一句「这主题是你的吗」，不是就 404。

示例 1：用纯 Python 看懂「字段白名单」

Python# whitelist_simulation.py —— fields 白名单为什么是安全闸门

# 数据库里一条主题记录的真实长相
topic_data = {"id": 1, "text": "Python", "owner_id": 1, "date_added": "2026-10-15"}

# 表单声明的白名单：只允许用户填 text
FORM_FIELDS = ["text"]

# 恶意用户提交的表单数据：除了 text，还偷偷塞了 owner_id
submitted = {"text": "Django", "owner_id": 999}

# ===== 有白名单时：只取白名单内的字段，其余统统丢弃 =====
cleaned = {key: submitted[key] for key in FORM_FIELDS if key in submitted}
print("白名单清洗后的数据：", cleaned)
print("=> owner_id 被丢弃，用户无法把数据挂到别人名下")

# ===== 反面：把 owner_id 也加进白名单 =====
FORM_FIELDS_BAD = ["text", "owner_id"]
cleaned_bad = {key: submitted[key] for key in FORM_FIELDS_BAD if key in submitted}
print("\n白名单含 owner_id 时：", cleaned_bad)
print("=> 恶意提交的 owner_id=999 被接受，这条数据就归别人了！")

# 结论：fields = ['text'] 这行不是省事，是安全闸门
print("\nowner_id 从来不由用户填写，而由系统根据 request.user 盖章：")
topic_data.update(cleaned)
topic_data["owner_id"] = 1      # 系统盖章：当前登录用户的 ID
print("最终存入数据库的记录：", topic_data)

示例 2：用纯 Python 看懂 request.user

Python# current_user_simulation.py —— request.user 到底是什么


class User:
    """已登录的用户对象。"""

    def __init__(self, username, user_id):
        self.username = username
        self.id = user_id
        self.is_authenticated = True


class AnonymousUser:
    """未登录用户的替身：Django 不给 None，而是给一个匿名对象。"""

    username = ""
    is_authenticated = False


# 已登录的请求：request.user 是真实的 User 对象
logged_in_request = {"user": User("eric", 1)}

# 未登录的请求：request.user 是 AnonymousUser
anonymous_request = {"user": AnonymousUser()}

for label, request in [("已登录", logged_in_request), ("未登录", anonymous_request)]:
    user = request["user"]
    print(f"{label}: is_authenticated={user.is_authenticated}, "
          f"username={user.username!r}")

print("\n=> 视图里写 request.user 永远拿得到对象：")
print("   登录了是 User，没登录是 AnonymousUser，不会 AttributeError")
print("   所以 filter(owner=request.user) 对未登录用户是安全的——")
print("   查不到任何数据，而不是崩溃")

示例 3：用纯 Python 模拟 @login_required 门禁

Python# login_required_simulation.py —— 装饰器是怎么当上门禁的


def login_required(view_func):
    """模拟 Django 的 @login_required：没登录就不放行。"""

    def wrapper(request, *args, **kwargs):
        user = request.get("user")
        if user is None or not user.get("is_authenticated"):
            print("  门禁：未登录 -> 重定向到登录页（settings.LOGIN_URL）")
            return "重定向到 /accounts/login/"
        return view_func(request, *args, **kwargs)

    return wrapper


@login_required
def topics(request):
    """受保护的主题列表视图。"""
    return "主题列表页面（只含当前用户的主题）"


# 场景一：已登录用户
print("已登录用户访问：")
print(" ", topics({"user": {"is_authenticated": True, "username": "eric"}}))

# 场景二：未登录用户
print("\n未登录用户访问：")
print(" ", topics({"user": {"is_authenticated": False}}))

print("\n=> 装饰器不改变视图本身的代码，只是在门口加了一道检查：")
print("   忘写 @login_required，等于门禁没上岗，谁都能推门进去")

示例 4：用纯 Python 看懂数据隔离

Python# data_isolation_simulation.py —— .all() 和 .filter(owner=...) 的生死差别

# 内存「数据库」：三个主题，分属两个用户
DATABASE = [
    {"id": 1, "text": "Python", "owner": "eric"},
    {"id": 2, "text": "Django", "owner": "eric"},
    {"id": 3, "text": "私密日记", "owner": "willie"},
]


def all_topics():
    """错误示范：Topic.objects.all() —— 不管是谁的，全都捞出来。"""
    return list(DATABASE)


def filter_topics(owner):
    """正确写法：Topic.objects.filter(owner=request.user)。"""
    return [t for t in DATABASE if t["owner"] == owner]


print("=== 危险：用 .all() ===")
for t in all_topics():
    print(f"  {t['owner']} 的主题: {t['text']}")
print("  => eric 登录后看到了 willie 的「私密日记」——数据裸奔！")

print("\n=== 安全：filter(owner=request.user) ===")
for t in filter_topics("eric"):
    print(f"  {t['owner']} 的主题: {t['text']}")
print("  => eric 只看得到自己的两条")
for t in filter_topics("willie"):
    print(f"  {t['owner']} 的主题: {t['text']}")
print("  => willie 登录则只看得到自己的")

# 三层防护的另外两层：详情页和编辑页
print("\n=== 详情层/编辑层：查到之后再比对归属 ===")
topic = DATABASE[2]          # willie 的主题
current_user = "eric"
if topic["owner"] != current_user:
    print(f"  {current_user} 请求主题「{topic['text']}」-> raise Http404")
    print("  => 不是你的东西，当成不存在（连「存在但不给你看」都不透露）")

Python# learning_logs/forms.py —— 把模型变成表单（ModelForm）
from django import forms

from .models import Topic, Entry


class TopicForm(forms.ModelForm):
    class Meta:
        model = Topic
        fields = ["text"]                     # 白名单：只暴露 text
        labels = {"text": ""}                 # 不生成字段标签


class EntryForm(forms.ModelForm):
    class Meta:
        model = Entry
        fields = ["text"]
        labels = {"text": ""}
        widgets = {"text": forms.Textarea(attrs={"cols": 80})}

示例 6：learning_logs/views.py（项目文件，第 19 章完成版）

Python# learning_logs/views.py —— 表单三视图 + 归属保护
from django.shortcuts import render, redirect
from django.contrib.auth.decorators import login_required
from django.http import Http404

from .models import Topic, Entry
from .forms import TopicForm, EntryForm


def index(request):
    """学习笔记的主页（不保护：谁都能看）。"""
    return render(request, "learning_logs/index.html")


@login_required
def topics(request):
    """只显示当前用户的主题。"""
    topics = Topic.objects.filter(owner=request.user).order_by("date_added")
    context = {"topics": topics}
    return render(request, "learning_logs/topics.html", context)


@login_required
def topic(request, topic_id):
    """显示单个主题及其所有条目。"""
    topic = Topic.objects.get(id=topic_id)
    # 确认请求的主题属于当前用户
    if topic.owner != request.user:
        raise Http404

    entries = topic.entry_set.order_by("-date_added")
    context = {"topic": topic, "entries": entries}
    return render(request, "learning_logs/topic.html", context)


@login_required
def new_topic(request):
    """添加新主题。"""
    if request.method != "POST":
        # 未提交数据：创建一个新表单
        form = TopicForm()
    else:
        # POST 提交的数据：对数据进行处理
        form = TopicForm(data=request.POST)
        if form.is_valid():
            new_topic = form.save(commit=False)
            new_topic.owner = request.user
            new_topic.save()
            return redirect("learning_logs:topics")

    # 显示空表单或指出表单数据无效
    context = {"form": form}
    return render(request, "learning_logs/new_topic.html", context)


@login_required
def new_entry(request, topic_id):
    """在特定主题中添加新条目。"""
    topic = Topic.objects.get(id=topic_id)

    if request.method != "POST":
        # 未提交数据：创建一个空表单
        form = EntryForm()
    else:
        # POST 提交的数据：对数据进行处理
        form = EntryForm(data=request.POST)
        if form.is_valid():
            new_entry = form.save(commit=False)
            new_entry.topic = topic
            new_entry.save()
            return redirect("learning_logs:topic", topic_id=topic_id)

    # 显示空表单或指出表单数据无效
    context = {"topic": topic, "form": form}
    return render(request, "learning_logs/new_entry.html", context)


@login_required
def edit_entry(request, entry_id):
    """编辑既有条目。"""
    entry = Entry.objects.get(id=entry_id)
    topic = entry.topic
    # 双重保护：条目 -> 主题 -> 当前用户
    if topic.owner != request.user:
        raise Http404

    if request.method != "POST":
        # 初次请求：使用当前的条目填充表单
        form = EntryForm(instance=entry)
    else:
        # POST 提交的数据：对数据进行处理
        form = EntryForm(instance=entry, data=request.POST)
        if form.is_valid():
            form.save()
            return redirect("learning_logs:topic", topic_id=topic.id)

    context = {"entry": entry, "topic": topic, "form": form}
    return render(request, "learning_logs/edit_entry.html", context)

示例 7：accounts 应用与项目配置（项目文件，完整版）

Python# accounts/views.py —— 注册页面（Django 不提供注册视图，自己写）
from django.shortcuts import render, redirect
from django.contrib.auth import login
from django.contrib.auth.forms import UserCreationForm


def register(request):
    """注册新用户。"""
    if request.method != "POST":
        # 显示空的注册表单
        form = UserCreationForm()
    else:
        # 处理填写好的表单
        form = UserCreationForm(data=request.POST)
        if form.is_valid():
            new_user = form.save()
            # 让用户自动登录，再重定向到主页
            login(request, new_user)
            return redirect("learning_logs:index")

    # 显示空表单或指出表单无效
    context = {"form": form}
    return render(request, "registration/register.html", context)

Python# accounts/urls.py —— 登录/注销用自带视图，注册用自己的
"""为应用程序 accounts 定义 URL 模式"""

from django.urls import path, include

from . import views

app_name = "accounts"

urlpatterns = [
    # 包含默认的身份验证 URL
    path("", include("django.contrib.auth.urls")),

]
