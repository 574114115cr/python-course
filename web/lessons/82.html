fig, ax = plt.subplots()
ax.scatter(x_values, y_values, s=40, c=y_values, cmap=plt.cm.Blues)
ax.set_title("校正之后：四周有留白，点是完整的")
ax.axis([0, 1100, 0, 1_100_000])   # 顺序是 [xmin, xmax, ymin, ymax]

示例 4：RandomWalk 类（画一次 + 模拟多次）

Python# random_walk_demo.py —— 教材 RandomWalk 类：单次轨迹 + 500 次叠加
# 跑完可以删掉生成的 png 文件

import random

import matplotlib.pyplot as plt


class RandomWalk:
    """模拟一只小虫的随机游走：每一步往四个方向之一移动一格。"""

    def __init__(self, num_points=5000):
        self.num_points = num_points       # 走多少步
        self.x_values = [0]                # 属性：位置列表，从原点出发
        self.y_values = [0]

    def step(self):
        """走一步，然后在列表末尾追加新位置。"""
        x_step = random.choice([-1, 0, 1])
        y_step = random.choice([-1, 0, 1])
        # ★ and 不是 or：只有"横竖都不动"才重来
        if x_step == 0 and y_step == 0:
            return False
        self.x_values.append(self.x_values[-1] + x_step)
        self.y_values.append(self.y_values[-1] + y_step)
        return True

    def fill_walk(self):
        """反复调用 step，直到攒够 num_points 个点。"""
        while len(self.x_values) < self.num_points:
            self.step()


plt.style.use("seaborn-v0_8")
random.seed(42)        # 固定种子：每次运行结果一样，bug 可复现

# ===== 1. 画一次游走：连线 + 标出终点 =====
walk = RandomWalk(1000)
walk.fill_walk()
print("点数:", len(walk.x_values), " 终点:", walk.x_values[-1], walk.y_values[-1])

fig, ax = plt.subplots(figsize=(8, 6))
ax.plot(walk.x_values, walk.y_values, linewidth=1, color="royalblue")   # 连线看轨迹
ax.scatter(walk.x_values[-1], walk.y_values[-1], s=120, c="red", label="终点")
ax.set_title("一次随机游走（1000 步）")
ax.legend()
ax.axis([0, 1000 * 1.05, 0, 1000 * 1.05])       # 校正绘图：留边距
fig.savefig("random_walk_一次.png", dpi=120, bbox_inches="tight")

# ===== 自检：拒绝原地不动之后，实际步数必须刚好等于 num_points =====
assert len(walk.x_values) == 1000, f"点数不对：{len(walk.x_values)}"
print("自检通过：没有原地踏步的浪费")

# ===== 2. 模拟 500 次游走并设置样式 =====
num_walks, num_points = 500, 5000
fig, ax = plt.subplots(figsize=(12, 8))
for i in range(num_walks):
    w = RandomWalk(num_points)
    w.fill_walk()
    # 用「第几次游走」当颜色值：先画的淡、后画的深，形成历史层次
    # 注意 c 必须和点一样长，所以写成 [i] * len(...) 这样的列表
    ax.scatter(w.x_values, w.y_values, s=1,
               c=[i] * len(w.x_values), cmap=plt.cm.Blues,
               alpha=0.1, edgecolors="none")

ax.set_title(f"{num_walks} 次随机游走（每次 {num_points} 步）", fontsize=16)
ax.axis([0, num_points * 1.05, 0, num_points * 1.05])
fig.savefig("random_walk_500.png", dpi=150, bbox_inches="tight")
print("两张图都已保存")

示例 5：Die 类 + 两个骰子 + 两种面数对比

Python# die_plotly.py —— Plotly 模拟掷骰子（单骰 / 双六面 / 六面+二十面）
# 跑完可以删掉生成的 html 文件

import random

import plotly.express as px
import plotly.graph_objects as go


class Die:
    """一枚 n 面骰子，results 记住每一次的点数。"""

    def __init__(self, num_sides=6):
        self.num_sides = num_sides
        self.results = []

    def roll(self):
        result = random.randint(1, self.num_sides)   # 含 1 和 num_sides 两端
        self.results.append(result)


random.seed(42)

# ===== 1. 单个六面骰：results.count() 数频次 =====
die = Die()
for _ in range(1000):
    die.roll()

x_values = [str(v) for v in range(1, die.num_sides + 1)]
y_values = [die.results.count(v) for v in range(1, die.num_sides + 1)]
print("点数:", x_values, "次数:", y_values)

# ★ 自检：频次之和必须等于总投掷次数，否则说明漏统计了某个类别
assert sum(y_values) == len(die.results), "频次对不上，肯定漏了一个类别"

fig = px.bar(x=x_values, y=y_values, labels={"y": "出现次数"})
fig.update_layout(title="掷 1000 次六面骰子的点数分布", xaxis_title="骰子点数")
fig.update_traces(marker_color="royalblue", marker_line_color="black")
fig.add_hline(y=len(die.results) / die.num_sides, line_dash="dash", line_color="red",
              annotation_text="理论平均")
fig.write_html("die_六面.html")

# ===== 2. 两个六面骰：zip 把两条列表拉链配对，列表推导逐对相加 =====
d1, d2 = Die(), Die()
for _ in range(1000):
    d1.roll()
    d2.roll()

sums = [a + b for a, b in zip(d1.results, d2.results)]
freq = [sums.count(v) for v in range(2, 13)]          # 2~12 共 11 种
assert sum(freq) == len(sums), "点数和统计对不上"
print("点数和频次:", dict(zip(range(2, 13), freq)))

fig2 = px.bar(x=list(range(2, 13)), y=freq, labels={"x": "点数和", "y": "出现次数"})
fig2.update_layout(title="掷两个六面骰子 1000 次：点数和分布（钟形）")
fig2.update_traces(marker_color="royalblue", marker_line_color="black")
fig2.add_hline(y=len(sums) / 11, line_dash="dash", line_color="red",
               annotation_text="理论平均")
fig2.write_html("die_两个六面.html")

# ===== 3. 两个面数不同的骰子：六面 vs 二十面（graph_objects 画两组柱子）=====
d6, d20 = Die(6), Die(20)
for _ in range(2000):
    d6.roll()
    d20.roll()

fig3 = go.Figure()
fig3.add_trace(go.Bar(x=list(range(1, 21)),
                      y=[d6.results.count(v) / len(d6.results) for v in range(1, 21)],
                      name="六面骰子", marker_color="royalblue", opacity=0.8))
fig3.add_trace(go.Bar(x=list(range(1, 21)),
                      y=[d20.results.count(v) / len(d20.results) for v in range(1, 21)],
                      name="二十面骰子", marker_color="tomato", opacity=0.8))
fig3.update_layout(barmode="group",
                   title="六面 vs 二十面：面数越多，分布越平（2000 次）",
                   xaxis_title="点数", yaxis_title="频率")
fig3.write_html("die_六面vs二十面.html")
print("已生成三张 html：单个骰子平坦，叠加后中间隆起，面数越多越平坦")

示例 6：同一份坐标，画成折线还是散点——先把数据算出来（walk_and_dice.py）

示例 4 和 5 的重点都在 ax.plot() 上，容易让人以为「画图」才是难点。其实这张图对不对，全看 x_values / y_values 怎么来的。这个示例把画图那一步整个拿掉，只算数据：随机游走的坐标怎么一步步累加出来、掷骰子的结果怎么变成频次表。跑通它，你再回去看示例 4/5 的绘图代码，会发现那几十行只是「把两个列表交给 ax」。

Python# walk_and_dice.py —— 不画图，只算数据：坐标是怎么一步步走出来的
import random
from collections import Counter


def get_step():
    """一步的方向：x、y 各自随机取 -1 / 0 / 1。"""
    return random.choice([-1, 0, 1]), random.choice([-1, 0, 1])


class RandomWalk:
    """随机游走：x_values / y_values 两个列表就是全部数据。"""

    def __init__(self, num_points=5000):
        self.num_points = num_points
        self.x_values = [0]
        self.y_values = [0]

    def fill_walk(self):
        while len(self.x_values) < self.num_points:
            x_step, y_step = get_step()
            if x_step == 0 and y_step == 0:      # ★ and：两个都不动才重来
                continue
            self.x_values.append(self.x_values[-1] + x_step)
            self.y_values.append(self.y_values[-1] + y_step)


random.seed(42)                               # 固定种子，输出才可复现
walk = RandomWalk(500)
walk.fill_walk()
print(f"点数：{len(walk.x_values)}（num_points = 500）")
print("前十步坐标：", list(zip(walk.x_values[:10], walk.y_values[:10])))
print(f"终点：({walk.x_values[-1]}, {walk.y_values[-1]})")
print(f"x 的范围：{min(walk.x_values)} ~ {max(walk.x_values)}"
      f"；y 的范围：{min(walk.y_values)} ~ {max(walk.y_values)}")
stayed = sum(1 for a, b in zip(walk.x_values, walk.y_values)
             if (a, b) == (0, 0))
print(f"经过原点 {stayed} 次（概率模型不同，这个数会变）")

print("\n=== 同一份坐标，换个画法就是不同的图 ===")
print("折线 --> ax.plot(walk.x_values, walk.y_values)")
print("散点 --> ax.scatter(walk.x_values, walk.y_values, s=1)")
print("上色 --> c=range(len(walk.x_values)), cmap=plt.cm.Blues  # 越走越深")

print("\n=== 掷骰子：一堆结果怎么变成频次表 ===")
random.seed(7)
results = [random.randint(1, 6) for _ in range(600)]
counts = Counter(results)                    # 一行数完，等价于 results.count(v)
print("前 20 次结果：", results[:20])
print("频次表：", dict(sorted(counts.items())))
frequencies = [counts[v] for v in range(1, 7)]   # 按面数顺序取，缺的面补 0
print("按面数排列：", frequencies)
assert sum(frequencies) == len(results)      # ★ 分布类程序必加：总数必须对上
print(f"总数校验通过：sum(frequencies) = {len(results)}")
print(f"出现最多：{counts.most_common(1)[0]}；最少：{counts.most_common()[-1]}")
print(f"平均点数：{sum(results) / len(results):.3f}（理论 3.5）")

终点：(21, -18)
x 的范围：-6 ~ 23；y 的范围：-25 ~ 1
经过原点 3 次（概率模型不同，这个数会变）

折线 --> ax.plot(walk.x_values, walk.y_values)
散点 --> ax.scatter(walk.x_values, walk.y_values, s=1)

=== 掷骰子：一堆结果怎么变成频次表 ===
前 20 次结果： [3, 2, 4, 6, 1, 1, 5, 1, 3, 5, 1, 5, 2, 1, 1, 4, 4, 1, 2, 1]
频次表： {1: 107, 2: 107, 3: 93, 4: 99, 5: 109, 6: 85}
按面数排列： [107, 107, 93, 99, 109, 85]
总数校验通过：sum(frequencies) = 600
出现最多：(5, 109)；最少：(6, 85)
平均点数：3.418（理论 3.5）

三条解读：

  1. append(self.x_values[-1] + x_step) 是整套随机游走的心脏：每一步都在上一步的位置上加位移，所以数据不是「一群随机数」，而是一条连续路径。写成 append(x_step) 你会得到一团散沙（每次从原点跳），图还是画得出来，但已经完全不是随机游走了。
  2. Counter 和 list.count() 的差别在复杂度：Counter 一趟遍历 O(n) 出全部频次；[results.count(v) for v in range(1, 7)] 要扫 6 趟。600 次掷骰子看不出差别，600 万次就是 6 倍时间——示例 5 用 count() 是为了跟教材对齐，量大时换 Counter。
  3. 把 num_points 改成 5000、掷骰子改成 6000 会怎样（同种子实跑）：游走终点从 (21, -18) 变成 (122, 59)，x 的范围从 -6 ~ 23 变成 -6 ~ 146；骰子六面的频次从 85 ~ 109 收窄到 953 ~ 1038，平均点数从 3.418 变成 3.473。样本越大，随机越像规律——位移按 √n 增长，频次越来越贴近 1000 这条理论线。
