看答案与解析 （先自己写，卡住 15 分钟再看） 
Pythonline = "2026-10-26 08:30:15 INFO user=alice action=login"
date, clock, level, *rest = line.split()
fields = {}
for item in rest:
    key, _, value = item.partition("=")
    fields[key] = value
print(" | ".join([f"{date} {clock}", fields["user"], fields["action"], level]))
解析

    def __init__(self, screen, settings):

        self.settings = settings
        self.moving_right = False
        self.moving_left = False
        self.rect = pygame.Rect(0, 0, 60, 40)
        self.x = float(self.rect.centerx)

    def handle_event(self, event):
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_RIGHT:
                self.moving_right = True
            elif event.key == pygame.K_LEFT:
                self.moving_left = True
        elif event.type == pygame.KEYUP:
            if event.key == pygame.K_RIGHT:
                self.moving_right = False
            elif event.key == pygame.K_LEFT:
                self.moving_left = False

    def update(self):
        if self.moving_right and self.rect.right < self.screen.get_rect().right:
            self.x += self.settings.ship_speed
        if self.moving_left and self.rect.left > 0:
            self.x -= self.settings.ship_speed
        self.rect.centerx = self.x
